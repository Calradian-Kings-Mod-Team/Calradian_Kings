namespace = wb_companion_event

### These events are dependent on the particular companions they are made for. Grievances are intended as a way to bring the similar system into CK3.
### If, as an example, you have a companion who hates raiding, and you raid, you'll gain a grievance for them with a max of 3-5 before they declare their departure from your company, unable to be recruited again for years.
### These events will be sorted by the particular type of grievance
# Effects
scripted_effect wb_companion_event_0001_cleanup = {
	# Remove Lists
	every_in_list = {
		list = ck_no_casualty_list
		remove_from_list = ck_no_casualty_list
	}
	every_in_list = {
		list = ck_no_quest_list
		remove_from_list = ck_no_quest_list
	}
	every_in_list = {
		list = ck_no_food_list
		remove_from_list = ck_no_food_list
	}
	# Remove Flags
	if = {
		limit = {
			has_character_flag = ck_wb_food
		}
		remove_character_flag = ck_wb_food
	}
	if = {
		limit = {
			has_character_flag = ck_wb_quest
		}
		remove_character_flag = ck_wb_quest
	}
	if = {
		limit = {
			has_character_flag = ck_wb_casualty
		}
		remove_character_flag = ck_wb_casualty
	}
}
# Remove the appropriate grievances to give them a cooldown
scripted_effect wb_companion_event_0001_remove_flags = {
	every_courtier = {
		limit = {
			has_character_flag = ck_has_$GRIEVANCE$_objection
		}
		remove_character_flag = ck_has_$GRIEVANCE$_objection
		add_character_flag = {
			flag = ck_$GRIEVANCE$_objection_cooldown
			months = 30
		}
	}
}

wb_companion_event.1000 = { # Low Provisions - Objection
	hidden = yes
	trigger = {
		is_landless_adventurer = yes
		is_ai = no
		any_courtier = {
			has_character_flag = companion_objects_low_food
		}
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_character_flag = ck_addressed_food_flag
				}
			}
			every_courtier = {
				limit = {
					employer = root
					has_character_flag = companion_objects_low_food
					NOR = {
						has_character_flag = ck_has_food_objection
						has_character_flag = ck_food_objection_cooldown
					}
				}
				add_character_flag = {
					flag = ck_has_food_objection
					years = 10
				}
			}
		}
		else_if = {
			limit = {
				has_character_flag = ck_addressed_food_flag
			}
			every_courtier = {
				limit = {
					employer = root
					has_character_flag = companion_objects_food_failure
					has_character_flag = ck_food_objection_cooldown
				}
				add_opinion = {
					target = root
					modifier = broke_promise
				}
			}
			random_courtier = {
				limit = {
					employer = root
					has_character_flag = companion_objects_food_failure
					has_character_flag = ck_food_objection_cooldown
				}
				save_scope_as = ck_companion_food_scope
			}
			if = {
				limit = {
					NOT = {
						has_character_modifier = breaks_promises_to_followers
					}
				}
				add_character_modifier = {
					modifier = breaks_promises_to_followers
					years = 3
				}
			}
			send_interface_toast = {
				left_icon = scope:root
				right_icon = scope:ck_companion_food_scope
				title = wb_companion_broke_promise
			}
		}
	}
