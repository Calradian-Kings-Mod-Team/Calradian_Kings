			trigger = { scope:recruit = yes }
			desc = wb_claimant_event.0002.recruit
		}
		# desc = wb_claimant_event.0002.closing
	}
	theme = dungeon
	left_portrait = {
		character = scope:actor
		animation = aggressive_dagger
	}
	right_portrait = {
		character = scope:recipient
		outfit_tags = { prison } # the previous option already freed them
		triggered_animation = {
			trigger = {
				always = yes
			}
			animation = { crying fear delirium beg disbelief grief prisondungeon }
		}
		
	}

	immediate = { #setup from grand_ambitions.0202
		# save_scope_as = actor
		# save_scope_as = root_scope # for loc
		# primary_title = { save_scope_as = revoked }
		# scope:defender = {

		# 	add_character_flag = {

		# 		days = 5
		# 	}
		# }
	}

	option = {
		name = wb_claimant_event.0002.disfigured
		trigger = {
			scope:recipient = {
				NOT = { has_trait = disfigured }
			}
		}
		disfigure_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = disfigured }
	}
	option = {
		name = wb_claimant_event.0002.blind
		trigger = {
			scope:recipient = {
				NOT = { has_trait = blind }
			}
		}
		blind_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = blinded }
	}
	option = {
		name = wb_claimant_event.0002.one_eyed
		trigger = {
			scope:recipient = {
				NOR = {
					has_trait = blind
					has_trait = one_eyed
					scope:actor.faith = {
						trait_is_virtue = trait:one_eyed
					}
				}
			}
		}
		blind_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = blinded }
		scope:recipient = {
			random_list = {
				1 = { add_character_flag = ck_blind_right_eye }
				1 = { add_character_flag = ck_gouged_eye }
				1 = { add_character_flag = ck_removed_eyes }
			}
		}
	}
	option = {
		name = wb_claimant_event.0002.hunchbacked
		trigger = {
			scope:recipient = {
				NOT = { has_trait = hunchbacked }
			}
		}
		castrate_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = disfigured }
	}
	option = {
		name = wb_claimant_event.0002.eunuch
		trigger = {
			scope:recipient = {
				NOT = { has_trait = clubfooted }
			}
		}
		castrate_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = maimed }
	}
	option = {
		name = wb_claimant_event.0002.one_legged
		trigger = {
			scope:recipient = {
				NOT = {has_trait = one_legged }
			}
		}
		save_scope_value_as = {
			name = maim_type
			value = flag:leg
		}
		maimed_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = maimed }
	}
	option = {
		name = wb_claimant_event.0002.one_handed
		trigger = {
			scope:recipient = {
				NOT = { has_trait = maimed }
			}
		}
		save_scope_value_as = {
			name = maim_type
			value = flag:arm
		}
		maimed_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = maimed }
	}
	option = {
		name = wb_claimant_event.0002.eunuch
		trigger = {
			scope:recipient = {
				is_eunuch_trigger = no
			}
		}
		castrate_recipient_effect = yes
		torture_blind_castrate_disfigure_opinion_effect = { VERB = castrated }
	}
	after = {
		if = {
			limit= {
				scope:banish ?= yes
			}
			banish_effect = { BANISHER = scope:actor }
		}
		else = {
			scope:recipient = { release_from_prison = yes }
			if = {
				limit = {
					scope:recruit ?= yes
					scope:recipient.liege = scope:actor
					is_valid_to_hire_court_position_type = court_jester_court_position
				}
				court_position_grant_effect = {
					CANDIDATE = scope:recipient
					EMPLOYER = scope:actor
					POS = court_jester
				}
			}
		}

		# disfigure_recipient_effect = yes
		blind_castrate_and_disfigure_effect = yes #Stress & dread
		scope:recipient = {
			remove_trait_force_tooltip = whole_of_body
			if = {
				limit = {
					has_trait = infirm
				}
				add_trait_xp = {
					trait = infirm
					value = 25
				}
			}
		}
	}
}
