scripted_effect read_tome_category_effect = {
	# Add stress for every more you read
	if = {
		limit = {
			has_variable = ck_mahdi_read_tomes
			var:ck_mahdi_read_tomes > 0
		}
		add_stress = {
			value = {
				value = 7.5
				multiply = var:ck_mahdi_read_tomes
				multiply = var:ck_mahdi_read_tomes # To get an exponential rise
				round = yes
			}
		}
	}
	# Set the appropriate variable/flag for skill-type
	set_variable = {
		name = west_nahasa_0003_desired_skill
		value = flag:$SKILL$
	}
	# Check if the player is rereading the lore
	if = {
		limit = {
			exists = var:read_all_of_$SKILL$_cooldown
			has_variable = read_all_of_$SKILL$_cooldown
		}
		add_character_flag = rereading_skill_books
	}
	# Move onto the desired skill
	trigger_event = west_nahasa.0003

}

scripted_effect ck_west_nahasa_0003_pick_subject_effect = {
	# Track how many in each subject you read
	increment_variable_with_duration_and_max_effect = {
		VAR = num_read_subject_$SKILL$
		VAL = 1
		MAX = 6
		TIME = 150 # plenty of time to accumulate read subjects for modifiers and a unique trait
	}
	if = {
		limit = {
			has_variable_list = read_subject_0003_$SKILL$
			has_character_flag = rereading_skill_books
		}
		# Clear the list if the player is rereading
		clear_variable_list = read_subject_0003_$SKILL$
		# Then remove the flag to avoid the constant clearing 
		remove_character_flag = rereading_skill_books
	}
	# Save variable for loc purposes
	random_list = {
		1 = {
			trigger = {
				NOT = {
					is_target_in_variable_list = {
						name = read_subject_0003_$SKILL$
						target = flag:$SKILL$_1
					}
				}
			}
			# Add it to the read list
			add_to_variable_list = {
				name = read_subject_0003_$SKILL$
				target = flag:$SKILL$_1
			}
			# Pull up that subject
			set_variable = {
				name = current_subject_0003
				value = flag:$SKILL$_1
			}
		}
		1 = {
			trigger = {
				NOT = {
					is_target_in_variable_list = {

				limit = { # Limit to stop players from coming back and forth to refresh the cooldown

		VAL = 1

		TIME = 4 # in time for the cooldown, when the decision is taken, we'll remove the var if it exists
