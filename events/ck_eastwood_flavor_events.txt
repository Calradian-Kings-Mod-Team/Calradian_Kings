## Educational Advancements - Well-educated man offers to teach children to read
eastwood_flavor.0001 = {
	type = character_event
	title = eastwood_flavor.0001.t
	desc = {
		desc = eastwood_flavor.0001.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:fascinated_child_ck ?= {
						is_child_of = root
					}
				}
				desc = eastwood_flavor.0001.desc.my_child
			}
			triggered_desc = {
				trigger = {
					scope:fascinated_child_ck ?= {
						NOT = {
							is_child_of = root
						}
					}
				}
				desc = eastwood_flavor.0001.desc.random_child
			}
		}
	}
	theme = realm

	left_portrait = {
		trigger = { # Child must exists
			exists = scope:fascinated_child_ck
		}
		character = scope:fascinated_child_ck
		animation = happiness
	}
	right_portrait = {
		character = scope:wandering_educator_ck
		animation = debating
	}
	override_background = {
		reference = terrain_scope
	}

	trigger = {
		is_ai = no # Sooooo many heresiarchs across the world without this
		is_landed = yes
		highest_held_title_tier >= tier_county
		NOR = {
			any_courtier = { # Don't trigger if you have a courtier with the character flag
				has_character_flag = ew_flavor_0001_flag
			}
			has_character_flag = ew_flavor_0001_cooldown # Cooldown trigger
		}
	}

	immediate = {
		capital_province ?= {
			save_scope_as = background_terrain_scope
		}
		# Create the educated man & save him as a scope
		create_character = {
			template = ck_wandering_educator_template
			dynasty = none
			location = root.capital_province
			save_scope_as = wandering_educator_ck
			after_creation = {
				add_trait = scholar
				add_trait = heresiarch
			}
		}
		# Pull a random child from court, if one exists. Prioritize your own children first
		if = {
			limit = {
				any_child = {
					is_adult = no
					is_ai = yes
					age >= 10
				}
			}
			random_child = {
				limit = {
					is_adult = no
					is_ai = yes
					age >= 10
				}
				save_scope_as = fascinated_child_ck
			}
		}
		else_if = {
			limit = {
				any_courtier_or_guest = {
					is_adult = yes
					is_ai = yes
					age >= 10
				}
			}
			random_courtier_or_guest = {
				limit = {
					is_adult = no
					is_ai = yes
					age >= 10
				}
				save_scope_as = fascinated_child_ck
			}
		}
	}

	option = { # Let him stay
		name = eastwood_flavor.0001.a
		# Add neutral modifier to increase hostile scheme success against you and decreases county control
		add_character_modifier = {
			modifier = ck_hired_wandering_educator
			years = 2
		}
		# Child is happy at this news
		if = {
			limit = {
				exists = scope:fascinated_child_ck
			}
			reverse_add_opinion = {
				modifier = grateful_opinion
				target = scope:fascinated_child_ck
				opinion = 30
			}
			# Give the child two increases to a single stat
			scope:fascinated_child_ck = {
				random_list = {
					1 = {
						add_stewardship_skill = 1
					}
					1 = {
						add_learning_skill = 1
					}
					1 = {
						add_martial_skill = 1
					}
					1 = {
						add_diplomacy_skill = 1
					}
					1 = {
						add_intrigue_skill = 1
					}
				}
				if = {
					limit = {
						NOT = {
							has_trait = zealous
						}
					}
					custom_tooltip = {
						text = eastwood_flavor.0001.a.tt
						random_list = {
							75 = {
								show_chance = no
							}
							25 = {
								show_chance = no
								add_trait = zealous
							}
						}
					}
				}
				set_relation_friend = {
					target = scope:wandering_educator_ck
					reason = friend_old_mentor
				}
			}
		}
		# The man's preachings are causing discontent
		capital_county = {
			change_county_control = -15
			add_county_modifier = {
				modifier = ck_paid_wandering_education
				years = 2
			}
		}
		# Hire the man
		add_courtier = scope:wandering_educator_ck
		# If you have an empty tutor spot, fill it.
		if = {
			limit = {
				is_valid_to_hire_court_position_type = court_tutor_court_position
				NOT = { employs_court_position = court_tutor_court_position }
			}
			appoint_court_position = {
				recipient = scope:wandering_educator_ck
				court_position = court_tutor_court_position
			}
		}
		hidden_effect = {
			add_character_flag = {
				flag = ew_flavor_0001_cooldown
				years = 3
			}
			scope:wandering_educator_ck = {
				add_character_flag = ew_flavor_0001_flag
			}
		}
	}
	option = { # Kick him out (text depends on arrogance)
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = {
							has_trait = arrogant
						}
						desc = eastwood_flavor.0001.b.arrogant
					}
					triggered_desc = {
						trigger = {
							NOT = { has_trait = arrogant }
							has_trait = zealous
						}
						desc = eastwood_flavor.0001.b.zealous
					}
					triggered_desc = {
						trigger = {
							NOR = { 
								has_trait = arrogant
								has_trait = zealous
							}
						}
						desc = eastwood_flavor.0001.b.normal
					}
				}
			}
		}
		# Child is upset
		if = {
			limit = {
				exists = scope:fascinated_child_ck
			}
			reverse_add_opinion = {
				modifier = upset_opinion
				target = scope:fascinated_child_ck
				opinion = -20
			}
		}
		# With the man's preachings gone, the county is a bit more stable
		capital_county = {
			change_county_control = 5
		}
		# Make him disappear
		custom_tooltip = {
			text = eastwood_flavor.0001.b.tt
			scope:wandering_educator_ck = {
				silent_disappearance_effect = yes
			}
		}
		# Cooldown
		hidden_effect = {
			add_character_flag = {
				flag = ew_flavor_0001_cooldown
				years = 3
			}
		}
	}
	option = { # Pay him to shut his heretical mouth up, stewardship duel. If fail, he leaves
		name = eastwood_flavor.0001.c
		custom_tooltip = eastwood_flavor.0001.c.tt
		# Pay out gold, county gets positive modifier for increased development, you get a positive modifier for children education
		pay_short_term_gold = {
			target = scope:wandering_educator_ck
			gold = 100
		}
		# The man stops preaching and starts helping with development and tutoring
		capital_county = {
			add_county_modifier = {
				modifier = ck_paid_wandering_education
				years = 2
			}
		}
		add_character_modifier = {
			modifier = ck_hired_wandering_educator_positive
			years = 2
		}
		# Hire the man
		add_courtier = scope:wandering_educator_ck
		# If you have an empty tutor spot, fill it.
		if = {
			limit = {
				is_valid_to_hire_court_position_type = court_tutor_court_position
				NOT = { employs_court_position = court_tutor_court_position }
			}
			appoint_court_position = {
				recipient = scope:supporter
				court_position = court_tutor_court_position
			}
		}
		# Child is happy
		if = {
			limit = {
				exists = scope:fascinated_child_ck
			}
			reverse_add_opinion = {
				modifier = grateful_opinion
				target = scope:fascinated_child_ck
				opinion = 30
			}
			# Give the child two increases to a single stat
			scope:fascinated_child_ck = {
				random_list = {
					1 = {
						add_stewardship_skill = 2
					}
					1 = {
						add_learning_skill = 2
					}
					1 = {
						add_martial_skill = 2
					}
					1 = {
						add_diplomacy_skill = 2
					}
					1 = {
						add_intrigue_skill = 2
					}
				}
				set_relation_friend = {
					target = scope:wandering_educator_ck
					reason = friend_old_mentor
				}
			}
		}
		# All is forgiven?
		scope:wandering_educator_ck = {
			remove_trait_force_tooltip = heresiarch
		}
		hidden_effect = {
			add_character_flag = {
				flag = ew_flavor_0001_cooldown
				years = 3
			}
			scope:wandering_educator_ck = {
				add_character_flag = ew_flavor_0001_flag
			}
		}
	}
}

scripted_trigger eastwood_flavor_0002_nomad_trigger = {
	OR = {
		holder ?= { # If anyone has a herder or nomadic government
			OR = {
				government_has_flag = government_is_herder
				government_has_flag = government_is_nomadic
			}
		}
		culture = { # If the culture has one of the big nomadic cultures
			OR = {
				has_cultural_pillar = heritage_khuzic
				has_cultural_pillar = heritage_khiindai
				has_cultural_pillar = heritage_lir_ovus
				has_cultural_parameter = unlock_maa_horse_archers
			}
		}
	}
}
### Nomads Wish to Settle
eastwood_flavor.0002 = {
	type = character_event
	title = eastwood_flavor.0002.t
	desc = eastwood_flavor.0002.desc
	theme = nomads

	left_portrait = {
		character = root
		animation = thinking
		camera = camera_event_center
	}

	lower_right_portrait = {
		character = scope:ck_0002_nomad_2
		animation = personality_bold
	}
	lower_center_portrait = {
		character = scope:ck_0002_nomad_1
		animation = beg
	}
	lower_left_portrait = {
		character = scope:ck_0002_nomad_3
		animation = personality_rational
	}

	override_background = {
		reference = terrain_scope
	}
	
	trigger = {
		# is_ai = no ## Give AI a chance here but make it rare for them
		is_landed = yes
		highest_held_title_tier >= tier_county
		NOR = {
			has_character_flag = ew_flavor_0002_cooldown # Cooldown trigger
			
			government_has_flag = government_is_nomadic # Makes no sense you'd be settling other nomads as a nomad
		}
		any_held_title = {
			title_tier = county
			any_neighboring_county = {
				eastwood_flavor_0002_nomad_trigger = yes
			}
		}
		culture = { # Not a nomadic culture yourself
			NOR = {
				has_cultural_pillar = heritage_khuzic
				has_cultural_pillar = heritage_khiindai
				has_cultural_pillar = heritage_lir_ovus
				has_cultural_parameter = unlock_maa_horse_archers
			}
		}
	}

	immediate = {
		capital_county = {
			save_scope_as = root_capital_county
		}
		# Find that neighboring realm and scope it
		random_held_title = {
			title_tier = county
			limit = {
				any_neighboring_county = {
					eastwood_flavor_0002_nomad_trigger = yes
				}
			}
			random_neighboring_county = {
				limit = {
					OR = {
						holder = { # If anyone has a herder or nomadic government
							OR = {
								government_has_flag = government_is_herder
								government_has_flag = government_is_nomadic
							}
						}
						culture = { # If the culture has one of the big nomadic cultures
							OR = {
								has_cultural_pillar = heritage_khuzic
								has_cultural_pillar = heritage_khiindai
								has_cultural_pillar = heritage_lir_ovus
								has_cultural_parameter = unlock_maa_horse_archers
							}
						}
					}
				}
				save_scope_as = ew_0002_nomad_county
			}
		}
		faith = { save_scope_as = root_faith_ck }
		# Create three nomad characters from this county and give them the appropriate nomadic traits
		# First character
		create_character = {
			location = root.location
			template = ck_nomadic_wanderer_0002
			culture = scope:ew_0002_nomad_county.culture
			faith = scope:ew_0002_nomad_county.faith
			gender_female_chance = {
				if = {
					limit = { scope:ew_0002_nomad_county.faith = { has_doctrine = doctrine_gender_male_dominated } }
					add = 0
				}
				else_if = {
					limit = { scope:ew_0002_nomad_county.faith = { has_doctrine = doctrine_gender_female_dominated } }
					add = 100
				}
				else = {
					add = 0
				}
			}
			save_scope_as = ck_0002_nomad_1
		}
		# Second character
		create_character = {
			location = root.location
			template = ck_nomadic_wanderer_0002
			culture = scope:ew_0002_nomad_county.culture
			faith = scope:ew_0002_nomad_county.faith
			save_scope_as = ck_0002_nomad_2
		}
		# Third character
		create_character = {
			location = root.location
			template = ck_nomadic_wanderer_0002
			culture = scope:ew_0002_nomad_county.culture
			faith = scope:ew_0002_nomad_county.faith
			save_scope_as = ck_0002_nomad_3
		}
	}

	# Welcome them openly / Malus to control / Increased development / Cost ( 300 / Paid out in thirds to each )
	option = {
		name = eastwood_flavor_0002.a

		show_as_tooltip = {
			pay_short_term_gold = {
				target = scope:ck_0002_nomad_1
				gold = 300
			}
		}
		capital_county = {
			add_county_modifier = {
				modifier = ck_settled_nomads_2
				years = 2
			}
		}
		# Setup Unique Nomadic MAA's below:
		# Khuzic MAAs
		if = {
			limit = {
				scope:ck_0002_nomad_1.culture = {
					has_cultural_pillar = heritage_khuzic # unique maa's
				}
			}
			hidden_effect = {
				add_character_flag = ck_can_recruit_khuzic_maas
			}
		}
		# County Changes
		capital_county = {
			change_county_control = -10
			change_development_level = 3
		}
		# Add Courtiers
		add_courtier = scope:ck_0002_nomad_1
		add_courtier = scope:ck_0002_nomad_2
		add_courtier = scope:ck_0002_nomad_3
		# Inform Player of Recruitment Choice
		send_interface_message = {
			type = msg_troops_gained
			title = eastwood_flavor_0002.a.recruit_nomads
			left_icon = root
			right_icon = scope:ck_0002_nomad_1
		}
		# Payout Gold and Add Event Cooldown
		hidden_effect = {
			pay_short_term_gold = {
				target = scope:ck_0002_nomad_1
				gold = 100
			}
			pay_short_term_gold = {
				target = scope:ck_0002_nomad_2
				gold = 100
			}
			pay_short_term_gold = {
				target = scope:ck_0002_nomad_3
				gold = 100
			}
			add_character_flag = {
				flag = ew_flavor_0002_cooldown
				years = 20
			} 
			add_character_modifier = {
				modifier = ck_settled_nomads_character
				years = 2
			}
		}
	}
	# Welcome them but demand faith conversion / No Cost / Increased Development / No MAAs
	option = {
		name = eastwood_flavor_0002.b
		if = {
			limit = {
				scope:ck_0002_nomad_1.faith != root.faith
			}
			custom_tooltip = {
				text = eastwood_flavor_0002.b.tt
				scope:ck_0002_nomad_1 = {
					set_character_faith = root.faith
				}
				scope:ck_0002_nomad_2 = {
					set_character_faith = root.faith
				}
				scope:ck_0002_nomad_3 = {
					set_character_faith = root.faith
				}
				add_courtier = scope:ck_0002_nomad_1
				add_courtier = scope:ck_0002_nomad_2
				add_courtier = scope:ck_0002_nomad_3
			}
		}
		capital_county = {
			change_development_level = 3
			add_county_modifier = {
				modifier = ck_settled_nomads
				years = 2
			}
		}
		hidden_effect = {
			add_character_flag = {
				flag = ew_flavor_0002_cooldown
				years = 10
			} 
		}
	}
	# Kick them out / Longer cooldown / May cause trouble in a neighboring county, regardless of if it's yours
	option = {
		name = eastwood_flavor_0002.c
		duel = {
			target = scope:ck_0002_nomad_1
			skill = martial
			value = medium_skill_rating
			50 = {
				desc = eastwood_flavor_0002.c.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = eastwood_flavor_0002.c.success
					left_icon = root
					right_icon = scope:ck_0002_nomad_1
				}
			}
			50 = {
				desc = eastwood_flavor_0002.c.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = eastwood_flavor_0002.c.failure
					left_icon = root
					right_icon = scope:ck_0002_nomad_1
					add_prestige = medium_prestige_loss
					root.capital_county = {
						add_county_modifier = {
							modifier = ck_expelled_nomads
							months = 18
						}
						change_county_control = -15
					}
				}
			}
		}
		hidden_effect = {
			add_character_flag = {
				flag = ew_flavor_0002_cooldown
				years = 10
			} 
			scope:ck_0002_nomad_1 = {
				silent_disappearance_effect = yes
			}
			scope:ck_0002_nomad_2 = {
				silent_disappearance_effect = yes
			}
			scope:ck_0002_nomad_3 = {
				silent_disappearance_effect = yes
			}
		}
	}

}

## HAUNTED DREAMS  -  Get a nightmare from a defeat in battle
eastwood_flavor.0003 = {
	type = character_event
	title = eastwood_flavor.0003.t
	desc = eastwood_flavor.0003.desc
	theme = mental_health
	override_background = {
		reference = bedchamber
	}
	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		is_adult = yes
		is_ai = no
		has_character_flag = ck_battle_wipe_nightmare_flag
	}

	immediate = {
		play_music_cue = "mx_cue_stress"
	}

	option = {
		name = eastwood_flavor.0003.a

		if = {
			limit = {
				faith = {
					NOR = {
						has_doctrine = tenet_pacifism
						has_doctrine = tenet_dharmic_pacifism
					}
				}
			}
			custom_tooltip = {
				text = eastwood_flavor.0003.a.tt_discount
				add_character_flag = { # Add character flag for discount on pacifism tenets
					flag = eastwood_flavor_3_a_flag
					years = 5
				}
			}
		}

		if = {
			limit = {
				faith = {
					trait_is_virtue = brave
				}
			}
			add_piety = medium_piety_loss
		}
			
		stress_impact = {
			zealous = medium_stress_impact_gain
		}

		add_character_modifier = {
			modifier = pacifist_mindset_modifier
			years = 5
		}

		# Add the cooldown
		hidden_effect = {
			add_character_flag = {
				flag = ck_battle_wipe_nightmare_cooldown
				years = 8
			}
		}
	}
	option = {
		name = eastwood_flavor.0003.b
		
		if = {
			limit = {
				faith = {
					trait_is_virtue = brave
				}
			}
			add_piety = medium_piety_gain
		}
		if = { # To hell with pacifism I guess? If you had it
			limit = {
				has_character_modifier = pacifist_mindset_modifier
			}
			remove_character_modifier = pacifist_mindset_modifier
		}
		
		# If your faith is Pacifist, gain stress for being zealous
		if = {
			limit = {
				faith = {
					OR = {
						has_doctrine = tenet_pacifism
						has_doctrine = tenet_dharmic_pacifism
					}
				}
			}
			add_piety = major_piety_loss
			stress_impact = {
				zealous = massive_stress_impact_gain
			}
		}

		# Add the cooldown
		hidden_effect = {
			add_character_flag = {
				flag = ck_battle_wipe_nightmare_cooldown
				years = 8
			}
		}
	}
	option = {
		name = eastwood_flavor.0003.c

		# Add the cooldown
		hidden_effect = {
			add_character_flag = {
				flag = ck_battle_wipe_nightmare_cooldown
				years = 8
			}
		}
	}
}
