scripted_effect ck_contract_events_0001_create_peasant_effect = {
	if = {
		limit = {
			NOT = {
				exists = scope:destination_$NUM$_character
			}
			location = scope:ck_destination_$NUM$
		}
		create_character = {
			template = peasant_character
			location = scope:ck_destination_$NUM$
			dynasty = none
			culture = scope:ck_destination_$NUM$.culture
			faith = scope:ck_destination_$NUM$.faith
			gender_female_chance = 50
			after_creation = {
				set_to_lowborn = yes
			}
			save_scope_as = destination_$NUM$_character
		}
	}
}

scripted_effect ck_contract_events_0003_herd_costs_effect = { # Setting up costs
	# Randomize Numbers of Livestock at each stop
	scope:task_contract = {
		set_variable = {
			name = location_herd_number
			value = {
				value = { 1 3 }
				round = yes
			}
		}
		set_variable = {
			name = location_herd_cost
			value = {
				value = scope:ck_contract_events_settlement_county.development_level
				multiply = 1.5
			}
		}
	}
	save_scope_value_as = {
		name = ck_location_herd_num
		value = scope:task_contract.var:location_herd_number
	}
	if = {
		limit = {
			scope:ck_contract_events_settlement_county_holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:task_contract = {
			set_variable = {
				name = location_herd_nomad_cost # Cost for a nomad to give up their herd for you. Since it's divided the herd number, it's the 
				value = {
					value = scope:ck_contract_events_settlement_county_holder.domicile.herd
					divide = 4 # Not too high, not too low
					divide = scope:task_contract.var:location_herd_number
					round = yes
				}
			}
			debug_log = "NOMAD_LIVESTOCK_COST_SET"
		}
	}
}
scripted_effect ck_contract_events_0003_remove_herd_nomad = {
	if = {
		limit = {
			scope:ck_contract_events_settlement_county_holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:ck_contract_events_settlement_county_holder = {
			domicile ?= {
				change_herd = {
					value = scope:task_contract.var:location_herd_nomad_cost
					multiply = $HERD_NUM$
					multiply = -1
					max = scope:ck_contract_events_settlement_county_holder.domicile.herd
				}
			}
		}
	}
}

scripted_effect ck_contract_events_0003_employer_payment_effects = {
	# What will the employer get from the player's livestock
	# Both settled and nomad get county modifiers too
	save_scope_value_as = {
		name = ck_increased_dev
		value = {
			value = scope:ck_employer_capital.development_level
			multiply = 0.1
			round = yes
		}
	}
	if = { # If the employer was nomadic, give them herd based on num of livestock
		limit = {
			scope:employer ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:employer = {
			domicile = {
				change_herd = { # Pay them out the herd cost + the number of herds the player brings
					value = scope:task_contract.var:location_herd_nomad_cost
					multiply = scope:task_contract.var:ck_livestock_tally
				}
			}
			# Add herd county modifier
		}
		if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally > scope:task_contract.var:ck_livestock_desired
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_x2_modifier
							years = 5
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					scope:task_contract.var:ck_livestock_tally <= scope:task_contract.var:ck_livestock_desired
					scope:task_contract.var:ck_livestock_tally > 0
				}
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_modifier
							years = 5
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally = 0
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_bad_modifier
							years = 5
						}
					}
				}
			}
		}
	}
	else_if = { # Landed employer will NOT gain gold. They paid the player that and they may otherwise be frugal
		limit = {
			scope:employer ?= {
				is_landed = yes # Avoid giving an employer who somehow becomes unlanded to receive these rewards, the contract should've invalidated though
				exists = capital_county
			}
		}
		if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally > scope:task_contract.var:ck_livestock_desired
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_x2_modifier
						years = 5
					}
					change_development_level = scope:ck_increased_dev
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					scope:task_contract.var:ck_livestock_tally <= scope:task_contract.var:ck_livestock_desired
					scope:task_contract.var:ck_livestock_tally > 0
				}
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_modifier
						years = 5
					}
				}
			}
		}
		else_if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally = 0
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_bad_modifier
						years = 5
					}
				}
			}
		}
	}
}

scripted_trigger ck_contract_events_settlement_county_not_sell = {
	scope:ck_contract_events_settlement_county ?= {
		OR = {
			holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
				domicile = {
					herd < 100 # Will not sell herd as nomad if they're low
				}
			}
			any_county_province_epidemic = { # Will not sell if there is an epidemic. Need to conserve food.
				outbreak_intensity > minor
			}
		}
	}
}
