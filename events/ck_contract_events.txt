scripted_effect ck_contract_events_0001_create_peasant_effect = {
	if = {

		}

			template = peasant_character
			location = scope:ck_destination_$NUM$
			dynasty = none
			culture = scope:ck_destination_$NUM$.culture

			gender_female_chance = 50
			after_creation = {
				set_to_lowborn = yes
			}
			save_scope_as = destination_$NUM$_character
		}
	}
}

scripted_effect ck_contract_events_0003_herd_costs_effect = { # Setting up costs
	# Randomize Numbers of Livestock at each stop
	scope:task_contract = {
		set_variable = {
			name = location_herd_number
			value = {
				value = { 1 3 }
				round = yes
			}
		}
		set_variable = {
			name = location_herd_cost
			value = {
				value = scope:ck_contract_events_settlement_county.development_level
				multiply = 1.5
			}
		}
	}
	save_scope_value_as = {
		name = ck_location_herd_num
		value = scope:task_contract.var:location_herd_number
	}
	if = {
		limit = {
			scope:ck_contract_events_settlement_county_holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:task_contract = {
			set_variable = {
				name = location_herd_nomad_cost # Cost for a nomad to give up their herd for you. Since it's divided the herd number, it's the 
				value = {
					value = scope:ck_contract_events_settlement_county_holder.domicile.herd
					divide = 4 # Not too high, not too low
					divide = scope:task_contract.var:location_herd_number
					round = yes
				}
			}
			debug_log = "NOMAD_LIVESTOCK_COST_SET"
		}
	}
}
scripted_effect ck_contract_events_0003_remove_herd_nomad = {
	if = {
		limit = {
			scope:ck_contract_events_settlement_county_holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:ck_contract_events_settlement_county_holder = {
			domicile ?= {
				change_herd = {
					value = scope:task_contract.var:location_herd_nomad_cost
					multiply = $HERD_NUM$
					multiply = -1
					max = scope:ck_contract_events_settlement_county_holder.domicile.herd
				}
			}
		}
	}
}

scripted_effect ck_contract_events_0003_employer_payment_effects = {
	# What will the employer get from the player's livestock
	# Both settled and nomad get county modifiers too
	save_scope_value_as = {
		name = ck_increased_dev
		value = {
			value = scope:ck_employer_capital.development_level
			multiply = 0.1
			round = yes
		}
	}
	if = { # If the employer was nomadic, give them herd based on num of livestock
		limit = {
			scope:employer ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
			}
		}
		scope:employer = {
			domicile = {
				change_herd = { # Pay them out the herd cost + the number of herds the player brings
					value = scope:task_contract.var:location_herd_nomad_cost
					multiply = scope:task_contract.var:ck_livestock_tally
				}
			}
			# Add herd county modifier
		}
		if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally > scope:task_contract.var:ck_livestock_desired
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_x2_modifier
							years = 5
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					scope:task_contract.var:ck_livestock_tally <= scope:task_contract.var:ck_livestock_desired
					scope:task_contract.var:ck_livestock_tally > 0
				}
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_modifier
							years = 5
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally = 0
			}
			scope:employer = {
				domicile = {
					domicile_location.county = {
						add_county_modifier = {
							modifier = ck_deliver_herd_county_bad_modifier
							years = 5
						}
					}
				}
			}
		}
	}
	else_if = { # Landed employer will NOT gain gold. They paid the player that and they may otherwise be frugal
		limit = {
			scope:employer ?= {
				is_landed = yes # Avoid giving an employer who somehow becomes unlanded to receive these rewards, the contract should've invalidated though
				exists = capital_county
			}
		}
		if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally > scope:task_contract.var:ck_livestock_desired
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_x2_modifier
						years = 5
					}
					change_development_level = scope:ck_increased_dev
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					scope:task_contract.var:ck_livestock_tally <= scope:task_contract.var:ck_livestock_desired
					scope:task_contract.var:ck_livestock_tally > 0
				}
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_modifier
						years = 5
					}
				}
			}
		}
		else_if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally = 0
			}
			scope:employer = {
				capital_county = {
					add_county_modifier = {
						modifier = ck_deliver_herd_county_bad_modifier
						years = 5
					}
				}
			}
		}
	}
}

scripted_trigger ck_contract_events_settlement_county_not_sell = {
	scope:ck_contract_events_settlement_county ?= {
		OR = {
			holder ?= {
				government_has_flag = government_is_nomadic
				exists = domicile
				domicile = {
					herd < 100 # Will not sell herd as nomad if they're low
				}
			}
			any_county_province_epidemic = { # Will not sell if there is an epidemic. Need to conserve food.
				outbreak_intensity > minor
			}
		}
	}
}

					# Add Gallowsbait XP
					laamp_rewards_apply_criminal_xp_effect = {
						TRACK = thief
						XP_MAX = 30
						XP_MIN = 25
					}
					hidden_effect = {
						if = {
							limit = {
								scope:task_contract.var:location_herd_number = 1
							}
							ck_contract_events_0003_remove_herd_nomad = { HERD_NUM = 1 }
						}
						else_if = {
							limit = {
								scope:task_contract.var:location_herd_number = 2
							}
							ck_contract_events_0003_remove_herd_nomad = { HERD_NUM = 2 }
						}
						else_if = {
							limit = {
								scope:task_contract.var:location_herd_number = 3
							}
							ck_contract_events_0003_remove_herd_nomad = { HERD_NUM = 3 }
						}
						# Add Livestock Tally
						scope:task_contract = {
							increment_variable_effect = {
								VAR = ck_livestock_tally
								VAL = 1
							}
						}
					}
				}
			}
			70 = {
				desc = ck_contract_events.0003.d.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				send_interface_toast = {
					title = ck_contract_events.0003.d.failure
					type = event_toast_effect_bad
					left_icon = root
					right_icon = scope:destination_character
					reverse_add_opinion = {
						modifier = attempted_to_steal_herd_crime
						target = scope:ck_contract_events_settlement_county_holder
					}
					# Add Gallowsbait XP
					laamp_rewards_apply_criminal_xp_effect = {
						TRACK = thief
						XP_MAX = 5
						XP_MIN = 0
					}
				}
			}
		}
		
		ai_chance = {
			base = 20 # AI will always attempt to buy, unless they don't have the funds
			ai_value_modifier = {
				ai_greed = 1.25
				ai_honor = -1
			}
			modifier = {
				add = 50
				has_trait = gallowsbait
				has_trait_xp = {
					trait = gallowsbait
					track = thief
					value >= 60
				}
			}
			modifier = {
				add = 15
				intrigue >= 15
			}
		}
		# Stress Impacts
		stress_impact = {
			just = medium_stress_impact_gain
			deceitful = minor_stress_impact_loss
		}
	}
	## Move On (No livestock added)
	option = {
		name = ck_contracts_events.0003.e
		trigger = {
			location.county != scope:ck_employer_capital
		}
		custom_tooltip = ck_contracts_event.0003.e.tt
		custom_tooltip = ck_contract_events.0003.number_of_heads
		ck_contract_events_0003_success_tooltip = yes # Notify player if they have enough livestock

			lazy = minor_stress_impact_loss
		}
		ai_chance = {
			base = 60 # AI will always attempt to buy, unless they don't have the funds
			ai_value_modifier = {
				ai_honor = -0.5
			}
			modifier = {
				add = 30
				short_term_gold < scope:task_contract.var:location_herd_cost
			}
		}
	}
	## Taking the herd and run
	option = {
		name = ck_contracts_events.0003.f
		custom_tooltip = ck_contracts_events.0003.f.tt
		custom_tooltip = ck_contract_events.0003.number_of_heads
		trigger = {
			location.county != scope:ck_employer_capital
			scope:task_contract.var:ck_livestock_tally > 1 # Need something to steal
		}
		# Add Provisions from the livestock
		domicile ?= {
			change_provisions = {
				value = {
					value = scope:task_contract.var:ck_livestock_tally
					multiply = 100
				}
			}
		}
		# Remove the peasants
		hidden_effect = {
			scope:destination_1_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_2_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_3_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_4_character ?= {
				silent_disappearance_effect = yes
			}
		}
		# Fail the Contract - Gain Gallowsbait XP from the type of failure it was
		scope:task_contract = {
			complete_task_contract = failure_stole_herd
		}
		# Stress Impacts
		stress_impact = {
			arbitrary = minor_stress_impact_loss
			just = medium_stress_impact_gain
		}
		ai_chance = {
			base = 15 # AI will always attempt to buy, unless they don't have the funds
			ai_value_modifier = {
				ai_greed = 1.5
				ai_honor = -1
			}
			modifier = {
				add = 50
				has_trait = gallowsbait
				has_trait_xp = {
					trait = gallowsbait
					track = thief
					value >= 60
				}
			}
			modifier = {
				add = 15
				intrigue >= 15
			}
			modifier = {
				add = 30
				short_term_gold < scope:task_contract.var:location_herd_cost
			}
		}
	}
	## Finally meeting the herdsman after all of that traveling
	option = {
		trigger = {
			location = scope:ck_employer_province
		}
		name = { # Critical Success
			trigger = {
				scope:task_contract.var:ck_0003_success_type = flag:0003_critical_success
			}
			text = ck_contract_events.0003.g.cs
		}
		name = { # Success
			trigger = {
				scope:task_contract.var:ck_0003_success_type = flag:0003_success
			}
			text = ck_contract_events.0003.g.s
		}
		name = { # Sub-Success
			trigger = {
				scope:task_contract.var:ck_0003_success_type = flag:0003_sub_success
			}
			text = ck_contract_events.0003.g.ss
		}
		name = { # Failure
			trigger = {
				scope:task_contract.var:ck_0003_success_type = flag:0003_failure
			}
			text = ck_contract_events.0003.g.f
		}
		
		if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally > scope:task_contract.var:ck_livestock_desired
			}
			scope:task_contract = {
				complete_task_contract = success_critical
			}
		}
		else_if = {
			limit = {
				# AND = {
					scope:task_contract.var:ck_livestock_tally = scope:task_contract.var:ck_livestock_desired
				# }
			}
			scope:task_contract = {
				complete_task_contract = success_standard
			}
		}
		else_if = {
			limit = {
				AND = {
					scope:task_contract.var:ck_livestock_tally < scope:task_contract.var:ck_livestock_desired
					scope:task_contract.var:ck_livestock_tally > 0
				}
			}
			scope:task_contract = {
				complete_task_contract = success_substandard
			}
		}
		else_if = {
			limit = {
				scope:task_contract.var:ck_livestock_tally = 0
			}
			scope:task_contract = {
				complete_task_contract = failure_standard
			}
		}
		ck_contract_events_0003_employer_payment_effects = yes
		# Remove the peasants
		hidden_effect = {
			scope:destination_1_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_2_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_3_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_4_character ?= {
				silent_disappearance_effect = yes
			}
		}
	}
