ck_pilgrimage.0002 = { # Visiting al-Ifrit
	type = activity_event
	title = ck_pilgrimage.0002.t
	desc = { # Current phase of the volcano, otherwise have it be active but nothing happens
		first_valid = {
			triggered_desc = { # Active Flows
				trigger = {
					OR = {
						has_game_rule = volcano_situation_disabled # Do not see if the situation exists because an extinct volcano can end the situation
						situation:ck_volcano_situation ?= {
							situation_current_phase = phase_active_flows
						}
					}
				}
				desc = ck_pilgrimage.0002.desc.active
			}
			triggered_desc = { # Active Flows - Cynical
				trigger = {
					OR = {
						has_game_rule = volcano_situation_disabled # Do not see if the situation exists because an extinct volcano can end the situation
						situation:ck_volcano_situation ?= {
							situation_current_phase = phase_active_flows
						}
					}
					has_trait = cynical
				}
				desc = ck_pilgrimage.0002.desc.active_cynical
			}
			triggered_desc = { # Warning - WARNING!!!!
				trigger = {
					situation:ck_volcano_situation ?= {
						situation_current_phase = phase_volcano_warning
					}
				}
				desc = ck_pilgrimage.0002.desc.warning
			}
			triggered_desc = { # Eruption - Seek shelter
				trigger = {
					situation:ck_volcano_situation ?= {
						OR = {
							situation_current_phase = phase_minor_eruption
							situation_current_phase = phase_major_eruption
						}
					}
				}
				desc = ck_pilgrimage.0002.desc.eruption
			}
			triggered_desc = { # Ash clouds
				trigger = {
					situation:ck_volcano_situation ?= {
						OR = {
							situation_current_phase = phase_minor_ash_clouds
							situation_current_phase = phase_major_ash_clouds
						}
					}
				}
				desc = ck_pilgrimage.0002.desc.ash_clouds
			}
			triggered_desc = { # Inactive
				trigger = {
					situation:ck_volcano_situation ?= {
						situation_current_phase = phase_inactive
					}
				}
				desc = ck_pilgrimage.0002.desc.inactive
			}
			triggered_desc = { # Extinct :( ## CK_WEST_NAHASA CK_EASTWOOD - Make two West Nahasan special doctrines and apply one here. One faith views reincarnation through a journey while the other views it through the volcano
				trigger = {
					situation:ck_volcano_situation ?= {
						situation_current_phase = phase_extinct
					}
				}
				desc = ck_pilgrimage.0002.desc.extinct
			}
		}
	}
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { has_variable = pilgrimage_death_bury_home }
			animation = sadness
		}
		triggered_animation = {
			trigger = {
				situation:ck_volcano_situation ?= {
					OR = {
						situation_current_phase = phase_minor_eruption
						situation_current_phase = phase_major_eruption
						situation_current_phase = phase_volcano_warning
					}
				}
			}
			animation = fear
		}
		animation = ecstasy
	}
	theme = pilgrimage_destination
	trigger = { scope:activity.activity_location.county = title:c_ghilizi }
	immediate = {
		play_music_cue = mx_cue_pilgrimage_arrival
	}
	override_background = {
		reference = ep2_travel_settlement_mountains
	}

	# Incredible
	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { has_variable = pilgrimage_death_bury_home }
						desc = ck_pilgrimage.0002.a_dead
					}
					triggered_desc = {
						trigger = {
							situation:ck_volcano_situation ?= {
								OR = {
									situation_current_phase = phase_minor_eruption
									situation_current_phase = phase_major_eruption
									situation_current_phase = phase_volcano_warning
									situation_current_phase = phase_minor_ash_clouds
									situation_current_phase = phase_major_ash_clouds
								}
							}
						}
						desc = ck_pilgrimage.0002.a_dangerous
					}
					triggered_desc = {
						trigger = { has_trait = cynical }
						desc = ck_pilgrimage.0002.a_cynical
					}
					desc = ck_pilgrimage.0002.a
				}
			}
		}
		give_pilgrim_trait_or_xp_effect = yes
		pilgrimage_arrival_stress_loss_effect = yes

		# Stress Impact
		stress_impact = {
			zealous = minor_stress_impact_loss
		}
	}
}

	switch = {
		trigger = var:asked_0101_question

		flag:question_black_1 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_black_3 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_red_1 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_red_3 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_white_1 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_white_2 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_blue_1 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_blue_2 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_green_1 = {
			root = {
				save_scope_as = correct_answer
			}
		}
		flag:question_green_2 = {
			root = {
				save_scope_as = correct_answer
			}
		}
	}
	if = {
		limit = {
			NOT = {
				root = scope:correct_answer
			}
		}
		save_scope_as = wrong_answer
		scope:activity = {
			add_activity_log_entry = {
				key = mahdi_wrong_answer_log
				tags = { negative }
				score = -25
				character = root
				show_in_conclusion = yes
				#Effects
				hidden_effect = {
					root = {
						add_piety = miniscule_piety_loss
					}
				}
			}
			if = {
				limit = { exists = var:activity_special_type_progression }
				change_variable = {
					name = activity_special_type_progression
					add = -25 #
				}
			}
		}
	}
	else_if = {
		limit = {
			root = scope:correct_answer
		}
		increment_variable_effect = {
			VAR = 0101_correct_answers
			VAL = 1
		}
		scope:activity = {
			add_activity_log_entry = {
				key = mahdi_correct_answer_log
				tags = { positive }
				score = 25
				character = root
				show_in_conclusion = yes
				#Effects
				hidden_effect = {
					root = {
						add_piety = miniscule_piety_gain
					}
				}
			}
			if = {
				limit = { exists = var:activity_special_type_progression }
				change_variable = {
					name = activity_special_type_progression
					add = 25 #
				}
			}
		}
	}
}
