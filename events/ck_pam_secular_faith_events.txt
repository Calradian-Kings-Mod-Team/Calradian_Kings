scripted_effect pam_secular_faith_shabbat_rescue_failure_ck_effect = {
	scope:traveler_in_danger = {
		increase_wounds_no_death_effect = { REASON = fight }
		if = {
			limit = { exists = current_travel_plan }
			current_travel_plan = { add_travel_plan_modifier = pam_sf_waylaid_road_modifier }
		}
	}
}

pam_secular_faith_events.0002 = {

	theme = christian
	override_background = terrain_scope
	widgets = {
		widget = { gui = event_window_widget_vfx_heavy_smoke container = foreground_shader_vfx_container }

			}
		}
	}

	immediate = {
		cp:councillor_court_chaplain = { save_scope_as = chaplain }

		random_held_title = {
			limit = {
				tier = tier_county

					any_character_in_location = {
						is_ai = yes
						current_travel_plan.travel_plan_owner ?= this
						this != root
						is_physically_able_adult = yes
					}
				}
			}
			random_county_province = {
				limit = {
					any_character_in_location = {
						is_ai = yes
						current_travel_plan.travel_plan_owner ?= this
						this != root
						is_physically_able_adult = yes
					}
				}
				save_scope_as = background_terrain_scope
				random_character_in_location = {
					limit = {
						is_ai = yes
						current_travel_plan.travel_plan_owner ?= this
						this != root

				}
			}
		}
	}

	option = { # Ride out at once and save them
		name = pam_secular_faith_events.0002.a
		add_piety = major_piety_gain
		pam_secular_faith_shabbat_rescue_success_effect = yes
		random = {
			chance = {
				value = 30
				subtract = {
					add = root.prowess
					multiply = 2
				}
				min = 1
			}
			increase_wounds_no_death_effect = { REASON = fight }
		}
		stress_and_fulfillment_impact = {
			base = medium_stress_gain
			craven = massive_stress_impact_gain
			sadistic = minor_stress_impact_gain
			callous = minor_stress_impact_gain
			brave = massive_stress_impact_loss
			zealous = medium_stress_impact_loss
		}
		ai_chance = {
			base = 100
			modifier = {
				OR = {
					has_trait = craven
					has_trait = sadistic
					has_trait = callous
					prowess < 10
				}
				factor = 0
			}
		}
	}

	option = { # Learning challenge: the higher the learning, the likelier the rescue
		name = pam_secular_faith_events.0002.b
		duel = {
			skill = learning
			value = high_skill_rating

				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}

				add_piety = major_piety_gain

			callous = minor_stress_impact_gain
			humble = medium_stress_impact_loss

		}

			base = 100
			modifier = {
				OR = {
					has_trait = impatient
					has_trait = sadistic
					has_trait = callous
				}
				factor = 0
			}
		}
	}

	option = { # Do not save them
