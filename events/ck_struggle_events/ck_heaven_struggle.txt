# Setting AI Intent
ck_heaven_struggle.0001 = {
	hidden = yes
	scope = none
	scope = struggle

	trigger = { # Should not need this, but, redundancy check!
		exists = struggle:heaven_mandate_struggle
		struggle:heaven_mandate_struggle = { any_involved_ruler = { } } # For when the conflict is ongoing but nobody is left to fight it
	}

	immediate = {
		# prepare needed variables; they will be destroyed at the end of the immediate
		set_variable = {
			name = agents_for_escalation
			value = 0
		}
		set_variable = {
			name = agents_for_deescalation
			value = 0
		}

		# Go through the characters to distribute them between the potential transitions
		struggle:heaven_mandate_struggle = {
			every_involved_ruler = {
				limit = { is_alive = yes }

				#######################
				# Play cue track
				######################
				if = {
					limit = { is_ai = no }
					play_music_cue = "mx_Struggle_Opening"
				}

				#######################
				# Check existing flags
				######################
				if = {
					limit = { has_character_flag = agenda_towards_escalation }
					root = {
						change_variable = {
							name = agents_for_escalation
							add = 1
						}
					}
				}
				else_if = {
					limit = { has_character_flag = agenda_towards_deescalation }
					root = {
						change_variable = {
							name = agents_for_deescalation
							add = 1
						}
					}
				}
				else = { # Set a flag
					#############################
					# Current Phase: Tension
					#############################
					if = {
						limit = { root = { is_struggle_phase = struggle_heaven_phase_tension } }
						random_list = {
							# Working towards Hostility
							# It should be for characters looking for direct conflicts and domination, with the will and ability to expand their realms
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = chaos
									INTENT = escalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = escalation
								}
							}
							# Working towards Compromise
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = compromise
									INTENT = deescalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = deescalation
								}
							}
						}
					}

					#############################
					# Current Phase: Chaos
					#############################
					if = {
						limit = { root = { is_struggle_phase = struggle_heaven_phase_chaos } }
						random_list = {
							# Working towards Hostility: it means that the character wants to stay in
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = chaos
									INTENT = escalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = escalation
								}
							}
							# Working towards Armistice
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = armistice
									INTENT = deescalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = deescalation
								}
							}
						}
						# If the Calradian Empire exists, collapse it
						if = {
							limit = {
								exists = title:h_calradian_empire
								NOT = {
									exists = global_var:calradian_empire_restored_perm
								}
							}
							if = {
								limit = {
									exists = global_var:calradian_empire_collapse_cooldown
								}
								remove_global_variable = calradian_empire_collapse_cooldown
							}
							set_global_variable = {
								name = calradian_empire_collapse_cooldown
								value = {
									value = current_year
									add = 20
								}
							}
							title:h_calradian_empire.holder ?= {
								destroy_title = title:h_calradian_empire
							}
							# Destroy all Imperial Provinces
							title:h_calradian_empire ?= {
								every_in_de_jure_hierarchy = {
									limit = {
										has_variable = imperial_province
									}
									holder ?= {
										destroy_title = prev
									}
								}
							}
							title:h_eastern_calradian_empire ?= {
								every_in_de_jure_hierarchy = {
									limit = {
										has_variable = imperial_province
									}
									holder ?= {
										destroy_title = prev
									}
								}
							}
							# Remove the temporary global variable to ensure it can be reformed
							if = {
								limit = {
									exists = global_var:calradian_empire_restored_temp
								}
								remove_global_variable = calradian_empire_restored_temp
							}
						}
					}

					#############################
					# Current Phase: Compromise
					#############################
					if = {
						limit = { root = { is_struggle_phase = struggle_heaven_phase_compromise } }
						random_list = {

							# Working towards Tension
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = tension
									INTENT = escalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = escalation
								}
							}
							# Working towards Compromise: they want to stay in the same phase
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = compromise
									INTENT = deescalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = deescalation
								}
							}
						}
					}

					#############################
					# Current Phase: Armistice
					#############################
					if = {
						limit = { root = { is_struggle_phase = struggle_heaven_phase_armistice } }
						random_list = {
							# Working towards Compromise
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = compromise
									INTENT = escalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = escalation
								}
							}
							# Working towards Armistice: they want to stay in the same phase
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = armistice
									INTENT = deescalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = deescalation
								}
							}
						}
					}

					#############################
					# Current Phase: Supremacy
					#############################
					if = {
						limit = { root = { is_struggle_phase = struggle_heaven_phase_supremacy } }
						random_list = {
							# Working towards Tension
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = tension
									INTENT = escalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = escalation
								}
							}
							# Working towards Supremacy: they want to stay in the same phase
							100 = {
								# Odds modifiers
								ck_hm_struggle_set_ai_agenda_common_modifiers = {
									PHASE = supremacy
									INTENT = deescalation
								}

								# Apply effects
								ck_hm_struggle_apply_ai_agenda_effect = {
									INTENT = deescalation
								}
							}
						}
					}
				}
			}
		}

		# Clean variables
		remove_variable = agents_for_escalation
		remove_variable = agents_for_deescalation
	}
}

scripted_effect ck_hm_struggle_1000_create_patriarchate_effect = {
	# First, remove the variable from the Patriarchate to prevent reformation
	title:k_calradic = {
		remove_variable = ck_hm_disabled_patriarch
		if = {
			limit = {
				NOT = { has_variable = default_coa_var }
			}
			set_variable = default_coa_var
		}
	}
	# Pull the court chaplain from your court and use him as the new Patriarch
	# Otherwise pull a temple holder in your realm, oooootherwise, create one
	if = {
		limit = {
			exists = cp:councillor_court_chaplain
		}
		cp:councillor_court_chaplain = {
			save_scope_as = ck_hm_patriarch_candidate
		}
	}
	else_if = {
		limit = {
			any_sub_realm_barony = {
				exists = holder
				title_province = { has_holding_type = church_holding }
			}
		}
		holder = {
			save_scope_as = ck_hm_patriarch_candidate
		}
	}
	else = {
		create_character = {
			template = virtuous_priest_character_template
			location = root.capital_province
			gender = male # PATRIARCH, it's in the name
			dynasty = none
			save_scope_as = ck_hm_patriarch_candidate
		}
	}
}
scripted_effect ck_hm_1000_grant_patriarchate_effect = {
	if = {
		limit = {
			title:k_calradic ?= {
				NOR = {
					exists = holder
					holder = scope:ck_hm_patriarch_candidate
				}
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change2
			add_claim_on_loss = no
		}
		title:k_calradic = {
			change_title_holder = {
				holder = scope:ck_hm_patriarch_candidate
				change = scope:change2
			}
			scope:ck_hm_patriarch_candidate = {
				change_liege = {
					liege = root
					change = scope:change2
				}
			}
		}
		resolve_title_and_vassal_change = scope:change2
	}
	# Set global variable that gets removed if the title is destroyed
	hidden_effect = {
		set_global_variable = {
			name = restored_the_ck_patriarchy
			value = yes
		}
	}
	
	every_character_struggle = {
		limit = { phase_has_catalyst = catalyst_reformed_patriarchy }
		activate_struggle_catalyst = {
			catalyst = catalyst_reformed_patriarchy
			character = root
		}
	}
}
ck_heaven_struggle.1000 = { # Give Lands to the Patriarchate
	type = character_event
	title = ck_heaven_struggle.1000.t
	desc = ck_heaven_struggle.1000.desc
	theme = ck_imperial
	override_background = {
		reference = courtyard
	}
	left_portrait = {
		character = root
		animation = personality_zealous
		camera = camera_event_right_forward
	}
	trigger = {
		NOT = {
			exists = global_var:restored_the_ck_patriarchy
		}
		exists = scope:ck_chose_my_lands
		faith = {
			has_doctrine = special_doctrine_ecumenical_christian
		}
		OR = {
			NOT = { exists = faith.religious_head }
			title:k_calradic ?= {
				NOT = {
					has_variable = ck_hm_disabled_patriarch
				}
			}
		}
	}
	immediate = {
		ck_hm_struggle_1000_create_patriarchate_effect = yes
		# Pull the de jure duchies both without holders and 3 counties
		primary_title = {
			every_in_de_facto_hierarchy = {
				limit = {
					tier = tier_duchy
					OR = {
						NOT = { exists = holder } # Either doesn't exist
						title_capital_county = { # Or does but isn't the capital of the player
							AND = {
								root.capital_county != this
								holder = root # and the player holds it
							}
						}
					}
					any_de_jure_county = {
						count >= 3
						holder = root
						this != root.capital_county
					}
				}
				add_to_list = ck_hm_1000_duchies
			}
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_duchy
					NOT = { is_in_list = ck_hm_1000_duchies }
					OR = {
						NOT = { exists = holder } # Either doesn't exist
						title_capital_county = { # Or does but isn't the capital of the player
							AND = {
								root.capital_county != this
								holder = root # and the player holds it
							}
						}
					}
					any_de_jure_county = {
						count >= 3
						holder = root
						this != root.capital_county
					}
				}
				add_to_list = ck_hm_1000_duchies
			}
		}
		if = {
			limit = {
				any_in_list = {
					list = ck_hm_1000_duchies
					count > 1 # Check for more than one qualifying duchy
				}
			} # Scope the first one
			random_in_list = {
				list = ck_hm_1000_duchies
				save_scope_as = ck_hm_1000_option_A
			} # Scope the second one
			random_in_list = {
				list = ck_hm_1000_duchies
				limit = {
					this != scope:ck_hm_1000_option_A
				}
				save_scope_as = ck_hm_1000_option_B
			} # Scope a third one, if it even exists
			if = {
				limit = {
					any_in_list = {
						list = ck_hm_1000_duchies
						count >= 3
					}
				}
				random_in_list = {
					list = ck_hm_1000_duchies
					limit = {
						NOR = {
							this = scope:ck_hm_1000_option_A
							this = scope:ck_hm_1000_option_B
						}
					}
					save_scope_as = ck_hm_1000_option_C
				}
			}
		}
		else = { # Just scope the one otherwise
			random_in_list = {
				list = ck_hm_1000_duchies
				save_scope_as = ck_hm_1000_option_A
			}
		}
	}
	option = {
		name = ck_heaven_struggle.1000.a
		trigger = {
			exists = scope:ck_hm_1000_option_A
			piety >= ck_hm_piety_cost_mine
		}
		show_as_unavailable = {
			exists = scope:ck_hm_1000_option_A
			piety < ck_hm_piety_cost_mine
		}
		add_prestige = 1000
		add_legitimacy_effect = { LEGITIMACY = ck_hm_legitimacy_cost } # For some reason, it's not a perfect 100 when added in
