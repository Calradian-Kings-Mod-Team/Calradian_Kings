	}

	is_enabled = {

			# highest_held_title_tier >= tier_kingdom # recommendation, not required
			# government_has_flag = government_is_mandala
			capital_province = root #The province
		}
	}
	can_rebuild = {
		is_occupied = no
		is_raided = no
		scope:county.holder = {
			is_independent_ruler = yes
			# has_royal_court = yes
		}
	}

	show_disabled = yes

	rebuild_cost = {
		treasury_or_gold = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = { 
				value = {
					add = standard_rebuild_value
					multiply = 2
				}
			}
		}
		prestige = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = { 
				value = {
					add = standard_rebuild_value
					multiply = 2
				}
			}
		}
	}

	fallback = { # ruined modifiers
		character_government_modifier = {
			parameter = government_has_influence
			monthly_influence_mult = -0.15
			governor_xp_gain_mult = -0.15
		}
		county_modifier = {
			monthly_county_control_growth_add = -0.15
			county_opinion_add = -15
			build_speed = -0.15
		}
	}

	type = great_building
	great_project_type = pamphian_palaces

	flag = travel_point_of_interest_diplomatic
	flag = no_wall
	effect_desc = {
		triggered_desc = {
			trigger = { has_dlc_feature = tours_and_tournaments }
			desc = wedding_bonus_effect
		}
		# desc = wineyard_effect_desc
	}
