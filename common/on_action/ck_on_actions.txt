random_yearly_playable_pulse_ck = {
	random_on_action = {
		chance_of_no_event = {
			value = 0
			if = {
				limit = {
					is_ai = yes
					highest_held_title_tier < tier_duchy
				}
				add = 75 # AI counts aren't particularly important
			}
		}
		50 = ck_geroian_events
		200 = ck_rhodok_events
		20 = eastwood_flavor_events # Lower than the top two so they take priority
	}
}

ck_game_start_after_lobby = {
	effect = {
		# Destroy the titles of AI claimants, if game rule is enabled
		if = {
			limit = {
				has_game_rule = ck_claimants_no_camps
			}
			warband_claimant_destroy_title_effect = yes
			warband_claimant_location_effect = yes
		}
		else_if = {
			every_duchy = {
				limit = {
					is_landless_type_title = yes
					var:adventurer_creation_reason ?= flag:exile
					holder ?= { has_variable = wb_claimant }
				}
				holder = {
					if = {
						limit = {
							has_game_rule = ck_claimants_off
							is_ai = yes # Need a chance for the player to not just instantly game over
						}
						var:wb_claimant = {
							holder ?= {
								save_temporary_scope_as = killer
							}
						}
						if = {
							limit = {
								exists = scope:killer
							}
							death = {
								killer = scope:killer
								death_reason = death_coup_successful
							}
						}
						else = {
							silent_disappearance_effect = yes
						}
					}
					else_if = {
						limit = {
							has_game_rule = ck_claimants_invert
						}
						if = {
							limit = {
								primary_title.var:can_keep_realm_law ?= flag:equal_law
							}
							var:wb_claimant = {
								set_variable = { name = can_keep_realm_law value = flag:equal_law }
							}
						}
						warband_claimants_retake_effect = yes
					}
					else = { # populate contracts without any visits
						trigger_event = ep3_laamps.9982
					}
				}
			}
		}
		
		if = {
			limit = {
				game_start_date >= 1257.3.23
			}
			steppe_great_horde_effects = yes
			# Gamerule: Canonical Families - If enabled, rulers like Harlaus will start with a pre-generated family
			if = {
				limit = {
					NOT = {
						has_game_rule = ck_canon_single
					}
				}
				# this effect was moved down so the player can be checked
				ck_warband_canonical_family_effect = yes
			}
			# else = {
 			# 	# disappear the fake family if true?
			# }
		}
	}
}
