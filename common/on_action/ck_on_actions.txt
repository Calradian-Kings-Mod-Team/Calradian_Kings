random_yearly_playable_pulse_ck = {
	random_on_action = {
		chance_of_no_event = {
			value = 0
			if = {
				limit = {
					is_ai = yes
					highest_held_title_tier < tier_duchy
				}
				add = 75 # AI counts aren't particularly important
			}
		}
		50 = ck_geroian_events
		200 = ck_rhodok_events
	}
}

	effect = {
		### CALRADIC KINGS - GAMERULES ###
		if = {
		 	limit = {
		 		has_game_rule = ck_full_shatter_calradia
		 	}

			shatter_calradian_empires_effect = yes
		}
		if = {
		 	limit = {
		 		has_game_rule = ck_shatter_calradian_empires
		 	}
		 	shatter_calradian_empires_effect = yes
		}
		if = {
		 	limit = {
		 		has_game_rule = ck_shatter_all_but_calradians
		 	}
		 	shatter_kingdoms_and_empires_effect = yes
		}
		#### === WARBAND RELATED === ####
		### TITLE FIXES - TEMPORARY FIX UNTIL WE CAN RESOLVE WHY THE RHODOK DUKES ARE INDEPENDENT AT START
		if = {
			limit = {
				game_start_date = 1257.3.23
			}
			#

				type = swear_fealty
				save_scope_as = warband_change
				add_claim_on_loss = no
			}
			# Rhodoks
			character:rhodok_maras_3 = {
				change_liege = {
					liege = title:k_rhodok.holder
					change = scope:warband_change
				}
			}
			character:rhodok_maras_6 = {
				change_liege = {
					liege = title:k_rhodok.holder
					change = scope:warband_change
				}
			}
			character:rhodok_jamiche_5 = {
				change_liege = {
					liege = title:k_rhodok.holder
					change = scope:warband_change
				}
			}
			character:rhodok_jamiche_2 = {
				change_liege = {
					liege = title:k_rhodok.holder
					change = scope:warband_change
				}
			}
			# Vaegirs
			character:Sturgia_VH_13 = {
				change_liege = {
					liege = title:e_sturgia.holder
					change = scope:warband_change
				}
			}
			# Swadians
			character:swadian_derchios_4 = {
				change_liege = {
					liege = title:k_swadia.holder
					change = scope:warband_change

			character:wb_corenia_3 = {
				change_liege = {
					liege = title:k_ilkhalsea.holder
					change = scope:warband_change
				}
			}
			resolve_title_and_vassal_change = scope:warband_change
		}
		### Warband Setup
		if = {
			limit = {
				game_start_date >= 1257.3.23
			}
			# MAKE NORDIC & STURGIC FAITHS REFORMED 
			faith:jumne = {
				set_variable = { name = has_been_reformed }

			faith:northern_gods = {
				set_variable = { name = has_been_reformed }

			# Characters from these cultures will know both languages
			culture:palaic_myzead = {
				set_language_from = culture:lascoicau
			}
			culture:eleftheroi = {
				set_language_from = culture:lascoicau
			}
			# Spawn in companions into random cities
			warband_companions_location_effect = yes
			# Gamerule: Canonical Families - If enabled, rulers like Harlaus will start with a pre-generated family
			if = {
				limit = {
					has_game_rule = ck_non_canon
				}
				ck_warband_canonical_family_effect = yes
			}
			## Change Khergit name list to name_list_khergit for proper Noyan titles
			culture:khergit = {
				set_name_list = name_list_khergit
			}
		}
		
	}

				warband_claimant_location_effect = yes
			}
		}
	}
