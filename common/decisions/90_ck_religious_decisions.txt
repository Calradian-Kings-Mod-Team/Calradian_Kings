select_personal_deity_northern_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp1_decisons_hold_bolt.dds"
	}

	desc = select_personal_deity_northern_decision_desc
	selection_tooltip = select_personal_deity_northern_decision_tooltip

	is_shown = {
		OR = {
			AND = {
				rite ?= { rite_has_parameter = select_personal_god_active }
				# Following a valid religion & faith.
				religion ?= religion:germanic_religion
			}
			AND = {
				has_personal_tenet_flag = select_personal_god_active_personal
				trigger_if = {
					limit = {
						religion != religion:germanic_religion
					}
					rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
					knows_tenet = tenet_unreformed_syncretism # at least you learn something about them.
				}
			}
		}
		NOT = { government_has_flag = government_is_mandala }
		# DLC check.
		has_fp1_dlc_trigger = yes
		# Following a valid religion & faith.
		religion = religion:jumne_religion
		faith = { has_doctrine_parameter = select_personal_god_active }
	}

	is_valid = {
	}

	is_valid_showing_failures_only = {
		# Have to at _least_ not be in piety debt.
		piety >= 0
	}

	cooldown = { years = 2 }

	effect = {
		# Show the possible options.
		show_as_tooltip = {
			random_list = {
				desc =  select_personal_deity_northern_decision_tt
				100 = {
					show_chance = no
					desc = select_personal_deity_northern_decision.select_odin
					add_character_modifier = bhakti_northern_generic_wodan
				}
				100 = {
					show_chance = no
					desc = select_personal_deity_northern_decision.select_nordr
					add_character_modifier = bhakti_northern_generic_nordr
				}
				100 = {
					show_chance = no
					desc = select_personal_deity_northern_decision.select_tyras
					add_character_modifier = bhakti_northern_generic_tyras
				}
				100 = {
					show_chance = no
					desc = select_personal_deity_northern_decision.select_bjorn
					add_character_modifier = bhakti_northern_generic_bjorn
				}
				100 = {
					show_chance = no
					desc = select_personal_deity_northern_decision.select_freyja
					add_character_modifier = bhakti_northern_generic_freyja
				}
			}
		}
		# And the actual effect.
		trigger_event = religious_decision.0102
	}

	ai_potential = {
		piety >= minor_piety_value
		NOR = {
			has_character_modifier = bhakti_northern_generic_wodan
			has_character_modifier = bhakti_northern_generic_nordr
			has_character_modifier = bhakti_northern_generic_tyras
			has_character_modifier = bhakti_northern_generic_bjorn
			has_character_modifier = bhakti_northern_generic_freyja
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 96
		duchy = 96
		kingdom = 96
		empire = 96
		hegemony = 96
	}
}
