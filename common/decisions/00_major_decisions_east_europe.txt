	decision_group_type = major
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}
	desc = defenders_of_highgod_decision_desc

	is_shown = {
		OR = {
			faith ?= faith:northern_gods
			faith ?= faith:vakken_faith
			#faith = faith:anburian_faith # CK_NORTHEAST CK_JUMNE - Add an Anburian religion in the future
		}
		#Only once per faith
		NOR = {
			# AND = {
			# 	faith = faith:anburian_faith
			# 	is_target_in_global_variable_list = {
			# 		name = unavailable_unique_decisions
			# 		target = flag:defenders_of_highgod_baltic_decision
			# 	}
			# }
			# AND = {
			# 	is_target_in_global_variable_list = {
			# 		name = unavailable_unique_decisions
			# 		target = flag:defenders_of_highgod_baltic_decision
			# 	}
			# 	faith ?= faith:baltic_pagan
			# }
			AND = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:defenders_of_highgod_slavic_decision
				}
				faith ?= faith:northern_gods
			}
			AND = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:defenders_of_highgod_finnish_decision
				}
				faith ?= faith:vakken_faith
			}
		}
	}

	widget = {
		controller = create_holy_order
		title_valid = {
			trigger_if = {
				limit = { exists = this }
				county.holder = scope:ruler
				title_province = {
					has_holding = yes
					has_special_building = no
				}
				is_holy_site_of = county.holder.faith
				is_leased_out = no
			}
		}
	}

	widget = { #Unop: The goal is to prevent players to waste this decision that IMHO is hard to setup
		controller = create_holy_order #revoke_holy_order_lease #create_holy_order
		barony_valid = {
			trigger_if = {
				limit = { exists = this }
				title_province = { has_special_building_slot = no }
				holder = scope:ruler
				is_leased_out = no
			}
		}
	}

	is_valid = {
		#Holds a holy site
		faith = {
			any_holy_site_barony = {
				county.holder = root
				title_province = {
					has_holding = yes
					has_special_building = no
				}
			}
		}
		piety_level >= 3
		trigger_if = {
			limit = { any_vassal = { is_powerful_vassal = yes } }
			any_vassal = {
				filter = {
					is_powerful_vassal = yes
				}
				count = all
				opinion = {
					target = root
					value >= high_positive_opinion
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		top_liege = this
		is_landed = yes
	}


	effect = {
		scope:barony = { save_scope_as = holy_site }
		save_scope_as = defender_of_highgod

		show_as_tooltip = { defenders_of_highgod_decision_effects = yes } #Actually applied in east_europe.0010 - prestige, laws, title, innovation effects

		#Events
		trigger_event = east_europe.0016
		#Event to every player (east_europe.0017) is actually send through east_europe.0016 to make sure the right holy site scope is set

		#Can only happen once per faith
		# if = {

		# 	add_to_global_variable_list = {
		# 		name = unavailable_unique_decisions
		# 		target = flag:defenders_of_highgod_baltic_decision
		# 	}
		# }
		if = {
			limit = { faith = faith:northern_gods }
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:defenders_of_highgod_slavic_decision
			}
		}
		else_if = {
			limit = { faith = faith:vakken_faith }
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:defenders_of_highgod_finnish_decision
			}
		}
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = 100
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = 100
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
