recruit_local_troops_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}
	desc = recruit_local_troops_decision_desc
	decision_group_type = adventurer

	is_shown = { # nomad probably needs to convert domicile culture first
		government_allows = conditional_maa_refill
		is_available_quick = {
			ai = no
			landed = no
			adult = yes
			travel = no
			# at_war = no
			imprisoned = no
			incapable = no
		}
		domicile ?= {
			domicile_uses_culture_and_faith = no
			domicile_location ?= {
				has_holding = yes
				county ?= {
					exists = culture
				}
			}
		}
	}
	cooldown = {
		weeks = 12
	}
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { # common/lifestyle_perks/00_learning_2_scholarship_tree_perks.txt
				government_has_flag = government_is_landless_adventurer
			}
			OR = {
				has_realm_law = camp_purpose_mercenaries
				# has_trait_with_flag = unlock_voluntary_laampdom_trait
				has_perk = planned_cultivation_perk
			}
			# check as a blocker if they're culture_head as well?
			custom_tooltip = {
				text = can_move_domicile_camp_exile.tt
				is_laamp_exiled_from_holder_or_liege_trigger = {
					PROVINCE = domicile.domicile_location
					LAAMP = root
				}
			}
			has_domicile_temperament_low = no
		}
	}
	is_valid = {
		domicile ?= {
			trigger_if = {
				limit = {
					domicile_uses_provisions = yes
				}
				provisions >= {
					value = 100
					multiply = root.maa_regiments_count
				}
			}
			domicile_location.county ?= {
				ck_unique_county_troops_trigger = yes
			}
		}
		trigger_if = {
			limit = {
				has_variable = recruits_local_troops
			}
			
			custom_tooltip = {
				text = recruits_local_troops_tt_desc
				var:recruits_local_troops != domicile.domicile_location.county.culture
			}
		}
	}
	cost = {
		gold = {
			value = 50
			multiply = maa_regiments_count
		}
		herd = {
			value = 100
			if = { 
				limit = {
					NOT = { government_has_flag = government_has_herd }
				}
				value = 0
			}
			else = {
				multiply = maa_regiments_count
			}
		}
	}

	effect = {
		set_variable = {
			name = recruits_local_troops
			value = domicile.domicile_location.county.culture
		}
		# enumerate the available options via custom loc? this needs to coerced values in that case
		var:recruits_local_troops ?= {
			custom_tooltip = will_recruit_variable
			if = {
				limit = {
					ck_single_heritage_trigger = yes
				}
				switch = {
					trigger = has_cultural_pillar
					heritage_calradian = { custom_tooltip = calradian_tooltip }
					heritage_battanian = { custom_tooltip = battanian_tooltip }
					heritage_geroian = { custom_tooltip = geroian_tooltip }
					heritage_byalic = { custom_tooltip = nord_tooltip }
				}
			}
			else = {
				switch = {
					trigger = var:troops
					culture:swadian = { custom_tooltip = swadian_tooltip }
					culture:vaegir = { custom_tooltip = vaegir_tooltip }
					culture:khergit = { custom_tooltip = khergit_tooltip }
					culture:nord = { custom_tooltip = nord_tooltip }
					culture:rhodok = { custom_tooltip = rhodok_tooltip }
					culture:sarranid = { custom_tooltip = sarranid_tooltip }
					culture:strugic = { custom_tooltip = sturgic_tooltip }
					culture:nahasi = { custom_tooltip = nahasi_tooltip }
					culture:khuzic = { custom_tooltip = khuzic_tooltip }
					culture:vlandian = { custom_tooltip = vlandian_tooltip }
					culture:togerd = { custom_tooltip = togerd_tooltip }
					culture:sudican = { custom_tooltip = sudican_tooltip }
					culture:kassen = { custom_tooltip = kassen_tooltip }
					culture:rhoyd = { custom_tooltip = rhoyd_tooltip }
					fallback = { custom_tooltip = NOT_ck_unique_troops_global }
				}
				# put the other ones
			}
		}

		# custom_tooltip = will_recruit_variable
		every_knight = {
			if = {
				limit = {
					culture = domicile.domicile_location.county.culture
				}
				# non-stacking default is 25
				add_opinion = {
					target = root
					modifier = loyalty_opinion
					opinion = 15
				}
			}
			else_if = {
				limit = {
					culture = {
						has_same_culture_heritage = root.culture
					}
				}
				# stacking
				add_opinion = {
					target = root
					modifier = unimpressed_opinion
					opinion = -10
				}
			}
		}
		if = {
			limit = {
				NOT = {
					has_character_modifier = recruits_local_troops
				}
			}
			add_character_modifier = {
				modifier = recruits_local_troops
				desc = will_recruit_variable
				weeks = 12
			}
		}
		domicile ?= {
			if = {
				limit = {
					 domicile_uses_provisions = yes
				}
				change_provisions = {
					value = 100
					multiply = prev.maa_regiments_count
				}
			}
		}
		open_view = {
			view = military
			player = root
		}
		# custom_tooltip = will_receive_cooldown
	}
	ai_check_interval = 0
}
