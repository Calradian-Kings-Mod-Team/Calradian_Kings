recruit_local_troops_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}
	desc = recruit_local_troops_decision_desc
	decision_group_type = adventurer

	is_shown = { # nomad probably needs to convert domicile culture first
		government_allows = conditional_maa_refill
		is_available_quick = {
			ai = no
			landed = no
			adult = yes
			travel = no
			# at_war = no
			imprisoned = no
			incapable = no
		}
		domicile ?= {
			domicile_uses_culture_and_faith = no
			domicile_location ?= {
				has_holding = yes
				county ?= {
					ck_unique_troops_trigger = yes
				}
			}
		}
	}
	cooldown = {
		weeks = 12
	}
	is_valid_showing_failures_only = {
		# common/lifestyle_perks/00_learning_2_scholarship_tree_perks.txt
		trigger_if = {
			limit = {
				NOT = { has_realm_law = camp_purpose_mercenaries }
			}
			has_perk = planned_cultivation_perk
		}
		trigger_if = {
			limit = {
				NOT = { has_perk = planned_cultivation_perk }
			}
			has_realm_law = camp_purpose_mercenaries
		}
		#no bungler allowed
		domicile.domicile_location ?= {
			is_raided = no
			is_occupied = no
			province_owner ?= {
				NOT = {
					has_opinion_modifier = {
						modifier = failed_task_contract_opinion
						target = root
					}
				}
				custom_tooltip = {
					text = domicile_location_patronage
					OR = {
						is_contact_of = root
						any_liege_or_above = {
							is_contact_of = root
						}
					}
				}
			}
		}
		# check as a blocker if they're culture_head as well?
		custom_tooltip = {
			text = can_move_domicile_camp_exile.tt
			NOT = {
				is_laamp_exiled_from_holder_or_liege_trigger = {
					LAAMP = root
				}
			}
		}
		has_domicile_temperament_low = no
		NOT = {
			ck_can_recruit_scoped_troops = { SCOPE = root.domicile.domicile_location.county.culture.var:troops_culture }
		}
	}
	is_valid = {
		domicile ?= {
			trigger_if = {
				limit = {
					domicile_uses_provisions = yes
					root.maa_regiments_count > 0
				}
				provisions >= {
					value = 100
					multiply = root.maa_regiments_count
				}
			}
		}
	}
	cost = {
		gold = {
			value = 50
			multiply = maa_regiments_count
		}
		herd = {
			value = 100
			if = { 
				limit = {
					NOT = { government_has_flag = government_has_herd }
				}
				value = 0
			}
			else = {
				multiply = maa_regiments_count
			}
		}
	}

	effect = {
		set_variable = {
			name = recruits_local_troops
			value = domicile.domicile_location.county.culture.var:troops_culture
		}
		# custom_tooltip = will_recruit_variable
		domicile.domicile_location.county.culture ?= {
			# var:troops_culture = {
			custom_description_no_bullet = {
				text = local_recruit_tree
				object = var:troops_culture
				subject = this
			}
			# }
			root.culture = { 
				change_cultural_acceptance = {
					target = prev
					desc = recruits_local_troops
					value = root.prestige_level
				}
			}
		}
		# add_contact = domicile.domicile_location.province_owner
		save_temporary_scope_as = liege
		every_knight = {
			custom = every_knight_troops_approval
			# add_piety = age
			trigger_event = {
				id = culture_emergence_events.1085
				delayed = no 
			}
		}
		if = {
			limit = {
				NOT = {
					has_character_modifier = recruits_local_troops
				}
			}
			add_character_modifier = {
				modifier = recruits_local_troops
				desc = will_recruit_variable
				weeks = 12
			}
		}
		domicile ?= {
			if = {
				limit = {
					 domicile_uses_provisions = yes
					 prev.maa_regiments_count > 0
				}
				change_provisions = {
					value = -100
					multiply = prev.maa_regiments_count
				}
			}
		}
		open_view = {
			view = military
			player = root
		}
		play_sound_effect = "event:/SFX/UI/Unit/Army/sfx_ui_unit_army_new_army"
		# custom_tooltip = will_receive_cooldown
	}
	ai_check_interval = 0
}

			text = recruits_local_cooldown_desc

						culture ?= {
							var:troops_culture ?= root.culture.var:troops_culture
						}
					}
				}
			}
		}
	}

		if = {
			limit = {
				culture = { 
					NOT = { has_variable = troops_culture }
				}
			}
			if = {
				limit = {
					has_character_modifier = recruits_local_troops
				}
				remove_character_modifier = recruits_local_troops
			}
			else_if = {
				limit = {
					has_character_modifier = recruits_local_troops_landed
				}
				remove_character_modifier = recruits_local_troops_landed
			}
			custom_tooltip = NOT_ck_unique_troops_first
			custom_tooltip = generic_maa_tooltip
			if = {
				limit = {
					is_landless_adventurer = yes
				}
				every_knight = {
					remove_variable = recruits_local_troops
				}
			}
		}
		else = {
			send_interface_toast = {
				title = recruit_cultural_troops_message
				left_icon = root
				right_icon = primary_title
				# remove_character_modifier = recruits_local_troops

					limit = {
						has_character_modifier = recruits_local_troops_landed
					}
					remove_character_modifier = recruits_local_troops_landed
				}
			}
			custom_tooltip = returns_to_cultural_troops_desc
			if = {
				limit = {
					is_landless_adventurer = yes
				}
				every_knight = {
					set_variable = {
						name = recruits_local_troops
						value = root.culture.var:troops_culture
					}
				}
			}
			root.culture = {
				custom_description_no_bullet = {
					text = local_recruit_tree
					object = var:troops_culture
					subject = var:troops_culture
				}
			}
		}
		
		remove_variable = recruits_local_troops
		open_view = {
			view = military
			player = root
		}
		play_sound_effect = "event:/SFX/UI/War/sfx_ui_window_war_hide"
	}
	
	ai_check_interval = 0
}
