
diarch_coup_valid_conspiracy_culmination_vassals_trigger = {
	OR = {
		is_powerful_vassal_of = $LIEGE$
		house ?= {
			OR = { #Unop: Include dominant families
				is_powerful_family = yes
				is_dominant_family = yes
			}
		}
	}
	is_imprisoned = no
}

is_powerful_vassal_or_house_head_trigger = {
	custom_tooltip = {
		text = custom_description.tt.is_power_vassal_or_family
		OR = {
			AND = {
				house ?= {
					OR = { #Unop: Include dominant families
						is_powerful_family = yes
						is_dominant_family = yes
					}
				}
				is_house_head = yes
			}
			is_powerful_vassal = yes
		}
	}
}
