
is_suitable_gg_leader_trigger = {

		adult = yes
		incapable = no
		imprisoned = no
	}
	pam_holds_church_seat_trigger = no

is_ghibelline_with_advantage_trigger = {
	is_ghibelline_trigger = yes
	situation:guelph_ghibelline ?= { situation_current_phase = ghibelline_advantage }
}

is_ghibelline_leader_with_advantage_trigger = {
	is_ghibelline_with_advantage_trigger = yes
	is_ghibelline_leader_trigger = yes
}

is_guelph_with_advantage_trigger = {
	is_guelph_trigger = yes
	situation:guelph_ghibelline ?= { situation_current_phase = guelph_advantage }
}

is_guelph_leader_with_advantage_trigger = {
	is_guelph_with_advantage_trigger = yes
	is_guelph_leader_trigger = yes
}

is_guelph_leader_trigger = {
	top_participant_group:guelph_ghibelline ?= { participant_group_type = guelph_leader }
}

is_guelph_trigger = {
	top_participant_group:guelph_ghibelline ?= { 
		OR = { 
			participant_group_type = guelph_leader
			participant_group_type = guelph_member
		}
	}
}

is_ghibelline_leader_trigger = {
	top_participant_group:guelph_ghibelline ?= { participant_group_type = ghibelline_leader }
}

is_ghibelline_trigger = {
	top_participant_group:guelph_ghibelline ?= { 
		OR = { 
			participant_group_type = ghibelline_leader
			participant_group_type = ghibelline_member
		}
	}
}

is_ghibelline_or_close_family_of_ghibelline_leader_trigger = {
	OR = {
		is_ghibelline_trigger = yes
		is_family_of_ghibelline_leader_trigger = yes
	}
}

is_family_of_ghibelline_leader_trigger = {
	any_close_family_member = { is_ghibelline_leader_trigger = yes }
}

is_guelph_or_ghibelline_trigger = {
	top_participant_group:guelph_ghibelline ?= { 
		OR = { 
			participant_group_type = guelph_leader
			participant_group_type = guelph_member
			participant_group_type = ghibelline_leader
			participant_group_type = ghibelline_member
		}
	}
}

is_gg_undecided_trigger = {
	top_participant_group:guelph_ghibelline ?= { participant_group_type = undecided }
}

gg_first_and_second_are_opposite_sides_trigger = {
	OR = {
		AND = {
			$FIRST$ = { is_guelph_trigger = yes }
			$SECOND$ = { is_ghibelline_trigger = yes }
		}
		AND = {
			$FIRST$ = { is_ghibelline_trigger = yes }
			$SECOND$ = { is_guelph_trigger = yes }
		}
	}
}

gg_target_house_head_is_opposite_side_trigger = {
	exists = $TARGET$.house.house_head
	gg_first_and_second_are_opposite_sides_trigger = {
		FIRST = $ACTOR$
		SECOND = $TARGET$.house.house_head
	}
}

gg_first_and_second_are_same_side_trigger = {
	OR = {
		AND = {
			$FIRST$ = { is_guelph_trigger = yes }
			$SECOND$ = { is_guelph_trigger = yes }
		}
		AND = {
			$FIRST$ = { is_ghibelline_trigger = yes }
			$SECOND$ = { is_ghibelline_trigger = yes }
		}
	}
}

gg_hre_causes_additional_tyranny_trigger = {
	save_temporary_scope_as = gg_tyranny_hre
	situation:guelph_ghibelline ?= { 
		NOT = { situation_current_phase = ghibelline_advantage }
	}
	has_variable = ghibelline_leader
	any_held_title = { this = title:e_calradian_republic }
	$TARGET$ = {
		is_ghibelline_trigger = no
		OR = {
			is_vassal_of = scope:gg_tyranny_hre
			liege ?= { is_vassal_of = scope:gg_tyranny_hre }
		}
		primary_title = {
			title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
		}
	}
}

	$TITLE$.holder ?= { 
		OR = {
			primary_title.tier = $TITLE$.tier
			primary_title = $TITLE$
		}
	}
}

holder_would_join_lombard_league_via_decision_trigger = {
	OR = {
		opinion = {
			target = scope:league_creator
			value >= 50
		}
		has_relation_friend = scope:league_creator
		is_guelph_trigger = yes
		trigger_if = {
			limit = { exists = title:e_calradian_republic.holder }
			has_relation_rival = title:e_calradian_republic.holder
		}
	}
}
