	# DLCs which enable The Silk Road situation
	OR = {
		has_tgp_dlc_trigger = yes
		has_gdp_dlc_trigger = yes
	}
	OR = {
		# Chinese
		has_cultural_pillar = heritage_chinese_ck
		# Discovered via Silk Road
		any_in_list = {
			variable = silk_road_unlocked_innovations
			this = culture_innovation:innovation_$INNOVATION$
		}
		# AND = {
			# OR = {

				limit = { has_tgp_dlc_trigger = yes }
				has_cultural_parameter = may_adopt_silk_road_innovations

				OR = {
					has_cultural_pillar = heritage_khuzic

					has_cultural_pillar = heritage_viet_ck
