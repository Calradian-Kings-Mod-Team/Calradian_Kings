burgher_heritage_germanic_trigger = {
	culture = {
		OR = {
			has_cultural_pillar = heritage_balian
			has_cultural_pillar = heritage_esbalian
			has_cultural_pillar = heritage_gottic
		}
	}
}

burgher_heritage_latin_trigger = {
	culture = {
		OR = {
			has_cultural_pillar = heritage_esvulgeste
			has_cultural_pillar = heritage_valanscau
			has_cultural_pillar = heritage_calradian
		}
	}
}

is_hanseatic_league_member_trigger = {
	is_confederation_member = yes
	confederation ?= {
		has_variable = is_hanseatic_league
	}
}

burgher_league_route_peer_market_trigger = {
	trade_market_owner ?= {
		holder ?= {
			this != scope:league_route_owner
			is_member_of_confederation = scope:league_route_league
		}
	}
}
