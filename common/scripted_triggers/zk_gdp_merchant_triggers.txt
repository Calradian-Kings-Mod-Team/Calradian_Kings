### Merchant company domicile building asset variants (scope with domicile_location, e.g. domicile owner)
company_domicile_western_asset_trigger = {
	domicile_location.culture = {
		OR = {
			has_graphical_western_culture_group_trigger = yes
			has_graphical_norse_culture_group_trigger = yes
			has_graphical_mediterranean_culture_group_trigger = yes
			AND = {
				has_graphical_iberian_culture_group_trigger = yes
				NOT = { has_cultural_pillar = language_nahasi }
			}
		}
	}
}

company_domicile_asian_asset_trigger = {
	domicile_location ?= {
		OR = {
			culture = { has_graphical_east_asia_or_southeast_asia_group_trigger = yes }
			ck_original_nomad_philosophy = yes # as long as you don't check trait on a province
		}
	}
}

culture_uses_camel_trigger = {
	OR = {
		has_cultural_tradition = tradition_caravaneers
		has_cultural_tradition = tradition_desert_nomads
		has_cultural_tradition = tradition_nahasan_nomads # no longer saharan
		has_innovation = innovation_war_camels # for fun
		has_cultural_pillar = heritage_nahasi
		has_cultural_pillar = heritage_darshi
	}
}
