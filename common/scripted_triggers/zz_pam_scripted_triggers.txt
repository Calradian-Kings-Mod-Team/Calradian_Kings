pam_valid_excommunication_target_trigger = {
	age >= 12
	save_temporary_scope_as = target_char
	$EXCOMMUNICATOR$ = { save_temporary_scope_as = excommunicator }
	trigger_if = {
		limit = { has_title = title:k_darusan }
		custom_tooltip = {
			text = cannot_excommunicate_pope_without_patriarch_or_antipope_tt
			liege ?= $EXCOMMUNICATOR$
			$EXCOMMUNICATOR$ = {
				OR = {
					pam_holds_christian_patriarch_title_trigger = yes
					pam_is_antipope_trigger = yes
				}
			}
		}
	}
	#always valid if Hof
	#This trigger is only relevant if excommunicator is not
	trigger_if = {
		limit = {
			scope:target_char = { need_hof_for_clergy_interaction_trigger = yes }
			scope:target_char.faith = {
				has_doctrine = doctrine_spiritual_head
				exists = scope:target_char.religious_head_or_challenger
			}
		}
		custom_tooltip = {
			text = excommunicator_must_be_hof_tt
			scope:excommunicator = scope:target_char.religious_head_or_challenger
		}
	}
	trigger_else_if = {
		limit = {
			scope:target_char.religious_head_or_challenger = scope:excommunicator
		}
	}
	trigger_else = {
		# If there is no HoF, a clergy HoR can take it
		trigger_if = {
			limit = {
				NOT = { exists = scope:target_char.faith.religious_head }
				scope:target_char.rite.head_of_rite ?= $EXCOMMUNICATOR$
			}
		}
		#Wanderers and courtiers
		trigger_else_if = {
			limit = {
				is_ruler = no
			}
			#Courtiers
			trigger_if = {
				limit = {
					exists = liege
				}
				custom_tooltip = {
					text = bishop_excommunication_target_courtier_tt
					liege = {
						OR = {
							target_is_same_character_or_above = $EXCOMMUNICATOR$
							AND = {
								is_landed = yes
								any_character_to_title_neighboring_county = {
									holder ?= {
										target_is_same_character_or_above = $EXCOMMUNICATOR$
									}
								}
							}
							capital_province.county = {
								clerical_region_title ?= {
									holder = $EXCOMMUNICATOR$
								}
							}
							cp:councillor_court_chaplain ?= $EXCOMMUNICATOR$
							any_liege_or_above = {
								cp:councillor_court_chaplain ?= $EXCOMMUNICATOR$
							}
						}
					}
				}
			}
			#Wanderers
			trigger_else = {
				custom_tooltip = {
					text = bishop_excommunication_target_wanderer_tt
					location.county = {
						OR = {
							holder ?= {
								target_is_same_character_or_above = $EXCOMMUNICATOR$
							}
							any_neighboring_county = {
								holder ?= {
									target_is_same_character_or_above = $EXCOMMUNICATOR$
								}
							}
							clerical_region_title ?= {
								holder = $EXCOMMUNICATOR$
							}
							holder.cp:councillor_court_chaplain ?= $EXCOMMUNICATOR$
							holder = {
								any_liege_or_above = {
									cp:councillor_court_chaplain ?= $EXCOMMUNICATOR$
								}
							}
						}
					}
				}
			}
		}
		#Theocrats/ecclesiasticals
		trigger_else_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			custom_tooltip = {
				text = target_char_must_be_clerical_subordinate_tt
				OR = {
					$EXCOMMUNICATOR$ = {
						any_subordinate = {
							OR = {
								this = scope:target_char
								any_subordinate = {
									OR = {
										this = scope:target_char
										any_subordinate = {
											OR = {
												this = scope:target_char
												any_subordinate = {
													this = scope:target_char
												}
											}
										}
									}
								}
							}
						}
					}
					target_is_liege_or_above = $EXCOMMUNICATOR$
				}
				
			}
		}
		#Secular rulers
		trigger_else = {
			custom_tooltip = {
				text = bishop_excommunication_target_secular_tt
				OR = {
					$EXCOMMUNICATOR$ = {
						exists = scope:target_char.cp:councillor_court_chaplain
						OR = {
							AND = {
								this = scope:target_char.cp:councillor_court_chaplain
								NOT = {
									exists = superior
								}
							}
							any_subordinate = {
								OR = {
									this = scope:target_char.cp:councillor_court_chaplain
									any_subordinate = {
										OR = {
											this = scope:target_char.cp:councillor_court_chaplain
											any_subordinate = {
												OR = {
													this = scope:target_char.cp:councillor_court_chaplain
													any_subordinate = {
														this = scope:target_char.cp:councillor_court_chaplain
													}
												}
											}
										}
									}
								}
							}
						}
					}
					target_is_liege_or_above = $EXCOMMUNICATOR$
					AND = {
						is_landed = yes
						any_character_to_title_neighboring_county = {
							holder ?= {
								target_is_same_character_or_above = $EXCOMMUNICATOR$
							}
						}
					}
					capital_county ?= {
						clerical_region_title ?= {
							holder = $EXCOMMUNICATOR$
						}
					}
					any_liege_or_above = {
						cp:councillor_court_chaplain ?= $EXCOMMUNICATOR$
					}
				}
			}
		}
	}
}
