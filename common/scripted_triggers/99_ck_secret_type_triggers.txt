secret_undermining_rule_is_valid_trigger = {
	# Both of y'all must still have treasuries
	$OWNER$ = {
		has_variable = schemed_against_employer
		var:schemed_against_employer ?= $TARGET$
	}
	$TARGET$ = {
		NOT = {
			has_opinion_modifier = {
				modifier = schemed_against_me
				target = $OWNER$
			}
		}
	}
}
secret_undermining_rule_is_criminal_trigger = {
	OR = {
		$TARGET$ = {
			OR = {
				is_liege_or_above_of = $OWNER$
				AND = {
					exists = title:h_china
					has_title = title:h_china
				}
			}
		}	
		$OWNER$ = {
			employer = $TARGET$
		}
	}
}
