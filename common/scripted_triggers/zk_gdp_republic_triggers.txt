# Can the scoped character use the Free City obligation on republic vassals?
gdp_can_have_free_cities_trigger = {
	# Must be an independent emperor or hegemon
	this = top_liege
	highest_held_title_tier >= tier_empire
	# But is not a republic
	NOT = { government_has_flag = government_is_republic }
	# Must have access to free cities somehow
	OR = {
		# Always true for (not) the HRE
		has_title = title:e_calradian_republic
		# Cultural traditions
		culture ?= { has_cultural_parameter = more_willing_to_create_republics }
		# check legacy track?
		has_trait = ck_oligarch
	}
}


# Character scope - check if a ruler's capital historically was maritime (or should be in terms of gameplay)
gdp_is_historical_maritime_republic = {
	capital_county ?= {
		OR = {
			this = title:c_charas
		}
	}
}
