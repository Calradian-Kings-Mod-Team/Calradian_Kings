## TRIGGERS FOR CHARACTERS ##
ck_does_not_start_with_family = { ## Mostly for a few hand selected characters to avoid the generation of a wife/husband and children
	OR = {
		AND = {
			exists = character:Simp_P_1
			this = character:Simp_P_1 #Rhagaea, her husband hit the can before game start
		}
		AND = {
			exists = character:vlandian_family_6_X
			this = character:vlandian_family_6_X # Do not allow this leper to have a wife
		}
		AND = {
			exists = character:harlaus_wb
			this = character:harlaus_wb # Does not any family other than Isolla, daughter of his cousin the former king, Esterich 
		}
		AND = {
			exists = character:ragnar_wb
			this = character:ragnar_wb # less sweet-talking more raiding
		}
		AND = {
			exists = character:patreon_jay_1
			this = character:patreon_jay_1 # Do not allow this leper to have a wife
		}
		AND = {
			exists = character:qurgazi_sanala_2
			this = character:qurgazi_sanala_2 # Still young enough to have a choice as a player
		}
		AND = {
			exists = character:wb_doroghalam_4
			this = character:wb_doroghalam_4 # Still young enough to have a choice as a player
		}
		AND = {
			exists = character:warrider_laamp_1
			this = character:warrider_laamp_1 # Still young enough to have a choice as a player
		}
		AND = {
			exists = character:ndakaaru_13
			this = character:ndakaaru_13 # Still young enough to have a choice as a player
		}
		AND = {
			exists = character:wb_nurazi_3
			this = character:wb_nurazi_3 # Recent peasant ruler, needs a choice among his vassals... or others
		}
		AND = {
			exists = character:yangifi_calradic_10
			this = character:yangifi_calradic_10 # Still young enough to have a choice as a player... maybe with Miqoul???
		}
	}
}

#used in animation.modifierspack
portrait_ck_swadian_german_trigger = {
	primary_title = title:e_west_calradia
	culture ?= {
		OR = {
			has_cultural_pillar = heritage_balian
			has_cultural_pillar = language_esbalian
		}
	}
}
portrait_ck_rhodok_italian_trigger = {
	primary_title ?= title:e_west_calradia
	is_culture_or_descended_from_trigger = {
		CULTURE = culture:rhodok
	}
}
portrait_ck_the_northern_lords_trigger = {
	has_dlc_feature = the_northern_lords
	OR = {
		culture = culture:nord 
		is_culture_or_descended_from_trigger = {
			CULTURE = culture:sturgic
		}
	}
}
portrait_ck_north_african_attire_trigger = {
	has_dlc_feature = north_african_attire
	OR = {
		culture = culture:nahrawi
		culture = culture:kuqaean
		culture = culture:tamari

	}
