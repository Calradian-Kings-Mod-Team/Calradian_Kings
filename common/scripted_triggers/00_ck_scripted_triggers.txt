# validate if the heritage matches culture vars
ck_unique_culture_troop_heritage_match = {
	# single-tree heritages
	trigger_if = {
		limit = {
			OR = {
				has_cultural_pillar = heritage_calradian
				has_cultural_pillar = heritage_battanian
				has_cultural_pillar = heritage_byalic
				has_cultural_pillar = heritage_geroian
			}
		}
		ck_unique_culture_troop_vars_trigger = no
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_balian
		}
		OR = {
			has_variable = swadian_troops_var
			has_variable = rhodok_troops_var
			has_variable = vlandian_troops_var
			has_variable = togerd_troops_var
			has_variable = sudican_troops_var
			has_variable = kassen_troops_var
			has_variable = rhoyd_troops_var
			has_variable = heimarkel_troops_var
			has_variable = eilander_troops_var
			# has_variable = balian_troops_var
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_esbalian
		}
		OR = {
			has_variable = swadian_troops_var
			has_variable = rhodok_troops_var
			# has_variable = esbalian_troops_var
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_sturgic
		}
		OR = {
			has_variable = sturgic_troops_var
			has_variable = vaegir_troops_var
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_khuzic
		}
		OR = {
			has_variable = khuzic_troops_var
			has_variable = khergit_troops_var
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_nahasi
		}
		OR = {
			has_variable = nahasi_troops_var
			has_variable = sarranid_troops_var
		}
	}
	trigger_else = {
		always = no
	}
}
