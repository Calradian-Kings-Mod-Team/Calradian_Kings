		government_has_flag = government_is_tribal
		government_has_flag = government_is_theocracy

		top_liege ?= { government_has_flag = government_is_theocracy }
	}

		character_is_calradian = yes

			OR = {
				# capital_county ?= title:c_lycaron

		}
	}
