
is_suitable_gg_leader_trigger = {
	is_available_quick = {
		adult = yes
		incapable = no
		imprisoned = no
	}
	NOT = { 
		any_held_title = { this = title:k_darusan }
	}
}

gg_hre_causes_additional_tyranny_trigger = {
	save_temporary_scope_as = gg_tyranny_hre
	situation:guelph_ghibelline ?= { 
		NOT = { situation_current_phase = ghibelline_advantage }
	}
	has_variable = ghibelline_leader
	any_held_title = { this = title:e_west_calradia }
	$TARGET$ = {
		is_ghibelline_trigger = no
		OR = {
			is_vassal_of = scope:gg_tyranny_hre
			liege ?= { is_vassal_of = scope:gg_tyranny_hre }
		}
		primary_title = {
			title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
		}
	}
}

gg_can_create_northern_italian_titles_trigger = {
	trigger_if = {

		custom_tooltip = {
			text = gg_hre_may_not_create_north_italian_titles_tt
			NOT = { exists = situation:guelph_ghibelline }
		}	
	}
	trigger_else = { always = yes }
}

title_province_is_part_of_gg_in_northern_italy_trigger = {
	any_character_situation = { this ?= situation:guelph_ghibelline }
	primary_title = {
		title_province = { geographical_region = gdp_guelph_ghibelline_northern_italy_region }
	}
	is_independent_ruler = no
	title:e_west_calradia.holder ?= root.liege
}

is_hre_vassal_or_independent_trigger = {
	OR = {
		is_independent_ruler = yes
		trigger_if = {
			limit = { 
				exists = title:e_west_calradia.holder
				exists = liege
			}
			top_liege = title:e_west_calradia.holder
		}
	}
}

holder_would_join_lombard_league_via_decision_trigger = {
	OR = {
		opinion = {
			target = scope:league_creator
			value >= 50
		}
		has_relation_friend = scope:league_creator
		is_guelph_trigger = yes
		trigger_if = {
			limit = { exists = title:e_west_calradia.holder }
			has_relation_rival = title:e_west_calradia.holder
		}
	}
}
