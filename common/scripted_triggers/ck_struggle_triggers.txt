## Heaven's Mandate
ck_hm_character_any_involvement_heaven_mandate_struggle_trigger = {
	OR = {
		ck_hm_character_involved_in_struggle_trigger = yes
		ck_hm_character_interloper_in_struggle_trigger = yes
	}
}

ck_hm_character_involved_in_struggle_trigger = {
	any_character_struggle = {
		involvement = involved
		is_struggle_type = heaven_mandate_struggle
	}
}

ck_hm_character_interloper_in_struggle_trigger = {
	any_character_struggle = {
		involvement = interloper
		is_struggle_type = heaven_mandate_struggle
	}
}

ck_hm_character_uninvolved_in_struggle_trigger = {
	exists = struggle:heaven_mandate_struggle
	ck_hm_character_interloper_in_struggle_trigger = no
	ck_hm_character_involved_in_struggle_trigger = no
}
