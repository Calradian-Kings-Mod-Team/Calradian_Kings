might_cheat_on_every_partner_trigger = {
	save_temporary_scope_as = potential_cheater
	# Technically (and, historically, frequently) eunuchs can still have affairs, they just aren't able to impregnate.
	# is_eunuch_trigger = no
	NOR = {
		has_trait = celibate
		has_trait = widow
	}
	trigger_if = {
		limit = { is_ai = yes }
		trigger_if = {
			limit = {
				# I have neither people to cheat
				NOR = {
					exists = concubinist
					any_spouse = {}
					any_relation = { type = lover }
					exists = betrothed
				}
			}
			OR = {
				# nor religious principle to cheat..
				accepts_adultery_without_penalty_trigger = yes
				NOT = {
					trait_is_shunned_or_criminal_in_rite_trigger = {
						TRAIT = trait:adulterer
						RITE = scope:potential_cheater.rite
						GENDER_CHARACTER = scope:potential_cheater
					}
				}
				# so I need lustful enough to break the rule
				has_lustful_adjacent_trait_trigger = yes
				has_focus = intrigue_temptation_focus
				ai_honor <= high_negative_ai_value
			}
		}
		# you don't need be honest to your concubine, but concubinist!
		trigger_if = {
			limit = { exists = concubinist }
			concubinist = {
				save_temporary_scope_as = partner1
			}
			scope:potential_cheater = { might_cheat_on_partner_trigger = { PARTNER = scope:partner1 } }
		}
		any_spouse = {
			count = all
			save_temporary_scope_as = partner2
			scope:potential_cheater = { might_cheat_on_partner_trigger = { PARTNER = scope:partner2 } }
		}
		any_relation = {
			type = lover
			count = all
			save_temporary_scope_as = partner3
			scope:potential_cheater = { might_cheat_on_partner_trigger = { PARTNER = scope:partner3 } }
		}
		trigger_if = {
			limit = { exists = betrothed }
			betrothed = {
				save_temporary_scope_as = partner4
			}
			scope:potential_cheater = { might_cheat_on_partner_trigger = { PARTNER = scope:partner4 } }
		}
	}
}
