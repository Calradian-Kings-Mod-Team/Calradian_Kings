# check if characters can hire from the same trees
ck_shares_unique_troops = {
	trigger_if = {
		limit = {
			$CHARACTER$ = {
				has_variable = recruits_local_troops
			}
		}
		ck_can_recruit_scoped_troops = {
			SCOPE = $CHARACTER$.var:recruits_local_troops
		}
	}
	trigger_else_if = {
		limit = {
			$CHARACTER$.culture = {
				has_variable = troops_culture 
			}

		}
		ck_can_recruit_scoped_troops = {
			SCOPE = $CHARACTER$.culture.var:troops_culture 
		}
	}
	trigger_else = { # handle generic == generic?
		always = no
	}
}
# validate if the heritage matches culture vars
ck_unique_culture_troop_heritage_match = {
	# single-tree heritages
	trigger_if = {
		limit = {
			ck_single_heritage_trigger = yes
			has_variable = troops_culture
		}
		switch = {
			trigger = var:troops_culture
			culture:calradian = { has_cultural_pillar = heritage_calradian }
			culture:battanian = { has_cultural_pillar = heritage_battanian }
			culture:nord = { has_cultural_pillar = heritage_byalic }
			culture:geroian = { has_cultural_pillar = heritage_geroian }
			# culture:kohinik = { has_cultural_pillar = heritage_kohinik }
		}
		# ck_unique_heritage_troop_vars_trigger = yes
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_balian
		}
		OR = {
			var:troops_culture ?= culture:swadian
			var:troops_culture ?= culture:rhodok
			var:troops_culture ?= culture:vlandian
			var:troops_culture ?= culture:togerd
			var:troops_culture ?= culture:sudican
			var:troops_culture ?= culture:kassen
			var:troops_culture ?= culture:rhoyd
			var:troops_culture ?= culture:heimarkel
			var:troops_culture ?= culture:eilander
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_esbalian
		}
		OR = {
			var:troops_culture ?= culture:swadian
			var:troops_culture ?= culture:rhodok
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_sturgic
		}
		OR = {
			var:troops_culture ?= culture:sturgic
			var:troops_culture ?= culture:vaegir
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_khuzic
		}
		OR = {
			var:troops_culture ?= culture:khzuait
			var:troops_culture ?= culture:khergit
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_nahasi
		}
		OR = {
			var:troops_culture ?= culture:aserai
			var:troops_culture ?= culture:sarranid
		}
	}
	trigger_else_if = {
		limit = {
			has_cultural_pillar = heritage_kohinik
		}
		OR = {
			var:troops_culture ?= culture:kartinakh
			var:troops_culture ?= culture:kohayan
		}
	}
	trigger_else = {
		always = no
	}
}

# no more base heritage blockers
ck_can_recruit_culture_heritage_troops = {
	trigger_if = {
		limit = { # do not check heritage here
			has_variable = recruits_local_troops
		}
		# exists = culture:$CULTURE$
		var:recruits_local_troops = {
			ck_can_recruit_culture_troop_vars = { CULTURE = $CULTURE$ }
		}
	}
	trigger_else = {
		culture ?= {
			ck_can_recruit_culture_troop_vars = { CULTURE = $CULTURE$ }
		}
	}
}
# when not passing in culture:X
ck_can_recruit_scoped_troops = {
	custom_description = {
		text = ck_can_recruit_culture_troop_vars
		object = $SCOPE$
		trigger_if = {
			limit = { # do not check heritage here
				has_variable = recruits_local_troops
			}
			var:recruits_local_troops = {
				var:troops_culture ?= $SCOPE$
			}
		}
		trigger_else = {
			culture.var:troops_culture ?= $SCOPE$
		}
	}
}

ck_local_culture_shown = {
	trigger_if = {
		limit = {
			has_game_rule = local_troops_vassals
		}
		any_powerful_vassal = {
			ck_happy_vassal_valid = yes
			culture = {
				var:troops_culture ?= culture:$CULTURE$
			}
		}
	}
	trigger_else = {
		any_sub_realm_county = {
			culture = {
				var:troops_culture ?= culture:$CULTURE$
			}
		}
	}
}

ck_local_county_valid = {
	# != doesn't tooltip well
	custom_tooltip = {
		text = recruits_local_troops_valid_desc
		ck_unique_troops_trigger = yes
		culture ?= {
			trigger_if = {
				limit = {
					root = {
						ck_unique_troops_trigger = yes
					}
				}
				root.culture.var:troops_culture != var:troops_culture
			}
			trigger_else = {
				this != root.culture
			}
		}
	}
}
ck_happy_vassal_valid = {
	is_at_war_with_liege = no
	OR = {
		is_councillor_of = liege
		is_allied_to = liege
		opinion = {
			target = liege
			value >= 50
		}
	}
	ck_local_county_valid = yes
	culture ?= { # explicitly must be different heritage?
		NOT = { has_same_culture_heritage = prev.liege.culture }
		# }
	}
}
ck_local_culture_valid = {
	trigger_if = {
		limit = {
			culture.var:troops_culture ?= culture:$CULTURE$
		}
		is_decision_on_cooldown = recruit_cultural_troops_decision
	}
	NOT = { ck_can_recruit_culture_heritage_troops = { CULTURE = $CULTURE$ } }
	save_temporary_scope_as = liege
	switch = {
		trigger = has_game_rule
		local_troops_third = {
			custom_description = {
				text = ck_local_culture_invalid
				value = culture:$CULTURE$.culture_player_holdings
				subject = culture:$CULTURE$
				any_sub_realm_barony = {
					culture ?= { var:troops_culture ?= culture:$CULTURE$ }
					count > sub_realm_holdings_three_fiddy
				}
			}
		}
		local_troops_plurality = {
			custom_description = {
				text = ck_local_culture_invalid
				value = culture:$CULTURE$.culture_player_holdings
				subject = culture:$CULTURE$
				any_sub_realm_barony = {
					culture ?= { var:troops_culture ?= culture:$CULTURE$ }
					count >= sub_realm_holdings_plurality
				}
			}
		}
		local_troops_majority = {
			any_sub_realm_barony = {
				culture ?= { var:troops_culture ?= culture:$CULTURE$ }
				count > sub_realm_holdings_supermajority
			}
		}
		local_troops_vassals = {
			custom_tooltip = {
				text = happy_vassal_invalid_decision
				any_powerful_vassal = {
					count > 2
					ck_happy_vassal_valid = yes
					ck_can_recruit_culture_heritage_troops = { CULTURE = $CULTURE$ }
				}
			}
		}
		fallback = {
			always = yes
		}
	}
}

# works inside maa that have access_through_subject defined
ck_obedient_maa_access = {
	government_allows = obedience # else check alliance/has_dread_level_towards?
	any_subject = {
		trigger_if = {
			limit = {
				culture = { has_cultural_parameter = harder_access_through_subject }
			}
			is_obedient_to = prev
		}
		has_access_to_maa = $MAA$
		# the following are normally not blockers but they really shouldn't have "access"
		highest_held_title_tier > tier_barony
		NOT = { government_has_flag = government_is_herder }
	}
}
