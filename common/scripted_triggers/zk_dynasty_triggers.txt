		has_cultural_pillar = heritage_chinese_ck
		has_name_list = name_list_han_ck
		# Korea
		has_cultural_pillar = heritage_korean_ck
		has_name_list = name_list_korean_ck
		# Buyeo
		has_cultural_pillar = heritage_buyeo
		has_name_list = name_list_balhae
		## Vlandians, Swadians, Rhodoks, and Gottics
		has_cultural_pillar = heritage_balian
		has_name_list = name_list_vlandian
		has_cultural_pillar = heritage_esbalian
		has_name_list = name_list_swadian
		has_name_list = name_list_rhodok
		has_cultural_pillar = heritage_gottic
		has_name_list = name_list_gottic
	}
	NOT = { faith.religion = religion:aserai_religion }
}
