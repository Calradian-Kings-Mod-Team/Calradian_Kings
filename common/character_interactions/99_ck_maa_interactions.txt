						add_piety_experience = medium_gold_value
					}
				}
				else = {
					scope:actor = {
						add_prestige_experience = medium_gold_loss
					}
				}
			}
			scope:target ?= {
				change_maa_regiment_size = {
					size = -1
				}

				limit = {
					scope:actor = { has_variable = recruits_local_troops }

				limit = {
					NOT = {
						has_character_modifier = recruits_local_troops
					}
				}
				add_character_modifier = {
					modifier = recruits_local_troops
					# desc = will_recruit_variable
					weeks = 12
				}
			}
		}
		# worded the other way around because it doesn't show x-y acceptance
		scope:actor.culture = {
			change_cultural_acceptance = {
				target = scope:recipient.culture
				desc = request_laamp_troops_used
				value = 1
			}
		}
	}

	ignores_pending_interaction_block = yes

		scope:recipient = {
			is_ruler = yes
			any_maa_regiment = {}
			# this != scope:actor
		}
	}
	ai_frequency = 0
	options_heading = debug_yoink_troops_heading
	auto_accept = yes
	send_options_exclusive = no
	send_option = {
		flag = realm
		is_shown = {
			scope:recipient = {
				is_landed = yes
				any_vassal = { number_of_maa_regiments > 0 }
			}
		}
		starts_enabled = {
			scope:recipient = {
				is_independent_ruler = yes
			}
		}
		localization = take_realm_interaction
		current_description = yoink_realm
	}
	send_option = {
		flag = title_only
		is_shown = {
			scope:recipient = {
				highest_held_title_tier >= min_title_maa_tier
			}
		}
		localization = take_title_interaction
		current_description = yoink_title
	}
	send_option = {
		flag = culture
		is_shown = {
			scope:recipient = {
				# OR = {
				# 	number_of_maa_regiments > 0
				# 	any_vassal = { number_of_maa_regiments > 0 }
				# }
				ck_unique_troops_trigger = yes

		}
		starts_enabled = {
			scope:recipient = {
				culture ?= { culture_head = prev }
			}
		}
		localization = COUNTY_OPINION_SAME_CULTURE
		current_description = yoink_culture
	}
	send_option = {
		flag = destroy
		is_shown = {
			scope:actor = {
				number_of_maa_regiments > 0
			}
		}
		starts_enabled = {
			scope:actor = {
				number_of_maa_regiments > 9
			}
		}
		localization = DISBAND_ALL
		current_description = yoink_clear
	}
	on_accept = {
		if = {
			limit = {
				scope:destroy = yes
			}
			scope:actor = {  # spawn_army needs an actual key, can't use type_of and regiment.maa_type isn't a thing?
				# destroy_all_event_troops = yes
				hidden_effect = {
					every_maa_regiment = {
						destroy_maa_regiment = yes
					}
				}
			# every_owned_title_maa_regiment = { 
			# 	destroy_maa_regiment = yes
			# }
			}
		}
		scope:recipient = {
			# capital_province ?= {
			# 	save_scope_as = origin
			# }
			culture ?= {
				var:troops_culture ?= {
					save_scope_as = liege_culture
				}
			}
			save_scope_as = secondary_recipient
			# add_gold = "number_maa_regiments_of_base_type(siege_weapon)"
			custom_tooltip = {
				text = debug_yoink_recipient
				if = {
					limit = {
						this != scope:actor
					}
					every_maa_regiment = {
						save_scope_as = reg
						# regiment_station ?= {
						# 	save_scope_as = origin
						# }
						scope:actor = {
							create_maa_sized = { SIZE = scope:reg.maa_size TYPE = type_of OF = scope:reg }
							# # note that your max != their max
						}
					}
				}
			}
			if = {
				limit = {
					scope:realm = yes
				}
				every_vassal_or_below = {
					limit = {
						any_maa_regiment = { }
						trigger_if = {
							limit = {
								scope:culture = yes
								exists = scope:liege_culture
							}
							ck_can_recruit_scoped_troops = { SCOPE = scope:liege_culture }
						}
					}
					save_scope_as = secondary_recipient
					# capital_province ?= {
					# 	save_scope_as = origin
					# }
					custom_tooltip = {
						text = debug_yoink_recipient
						every_maa_regiment = {
							save_scope_as = reg
							scope:actor = {
								create_maa_sized = { SIZE = scope:reg.maa_size TYPE = type_of OF = scope:reg }
							}
						}
					}
				}
			}
			if = {
				limit = {
					scope:title_only = yes
				}
				every_sub_realm_title = {
					every_title_maa_regiment = {
						save_scope_as = reg
						scope:actor = {
							create_maa_sized = { SIZE = scope:reg.maa_size TYPE = type_of OF = scope:reg }
						}
					}
				}
			}
		}
		scope:actor = {
			open_view = {
				view = military
				data = scope:actor
				player = scope:actor
			}
			open_view_data = {
				view = character
				data = scope:actor
				player = scope:actor
			}
		}
	}
