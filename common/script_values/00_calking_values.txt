struggle_hm_ending_compromise_decision_war_counter_value = {
	struggle:heaven_mandate_struggle = {
		every_involved_ruler = {
			limit = {
				top_liege = this
				primary_title = { is_mercenary_company = no }
				primary_title = { is_holy_order = no }
				any_primary_war_enemy = {
					top_liege = this
					any_character_struggle = {
						involvement = involved
						is_struggle_type = heaven_mandate_struggle
					}
					primary_title = { is_holy_order = no }
					primary_title = { is_mercenary_company = no }
				}
			}
			add =  {
				value = 0
				if = {
					limit = {
					 	any_character_war = {
							OR = {
								AND = {
									primary_attacker = prev
									primary_defender = {
										any_character_struggle = {
											involvement = involved
											is_struggle_type = heaven_mandate_struggle
										}
										top_liege = this
										primary_title = { is_mercenary_company = no }
									}
								}
								AND = {
									primary_defender = prev
									primary_attacker = {
										any_character_struggle = {
											involvement = involved
											is_struggle_type = heaven_mandate_struggle
										}
										top_liege = this
										primary_title = { is_mercenary_company = no }
									}
								}
							}
						}
					 }
					add = 0.5 # they are both primary participant so they will add 0.5 each
				}
				else = {
					add = 1 # only one of them is primary participant, so we need to add 1
				}
			}
		}
	}
}

dead_courtiers_and_guests_from_eruption = {
	value = {
		value = 0
		if = {
			limit = {
				situation:ck_volcano_situation ?= {
					situation_current_phase = phase_minor_eruption
				}
			}
			value = total_number_of_courtiers_and_guests
			multiply = 0.3
			round = yes
		}
		else_if = {
			limit = {
				situation:ck_volcano_situation ?= {
					situation_current_phase = phase_major_eruption
				}
			}
			value = total_number_of_courtiers_and_guests
			multiply = 0.6
			round = yes
		}
	}
}
