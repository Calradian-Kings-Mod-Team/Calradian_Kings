feudalize_holding_interaction_cost_ck = {
	value = 0
	if = {
		limit = {
			scope:target.county ?= {
				has_variable = was_razed
			}
		}
		add = {
			value = 50
			desc = BASE
		}
	}
	else_if = {
		limit = {
			scope:target.county.culture ?= culture
		}
		add = {
			value = 200
			desc = BASE
		}
	}
	else = {
		add = {
			value = 500
			desc = BASE
		}
	}
	if = {
		limit = {
			is_ai = yes
			capital_province = { has_holding_type = tribal_holding }
		}
		multiply = 0.5
	}
	if = {
		limit = {
			scope:target.county ?= {
				OR = {
					has_county_modifier = mpo_siberian_permafrost_modifier
					has_county_modifier = mpo_siberian_permafrost_modifier_bad
				} 
			}
		}
		add = {
			value = 2000
			desc = mpo_siberian_permafrost_modifier
		}
	}
	if = {
		limit = {
			scope:target.county ?= {
				OR = {
					has_county_modifier = pam_moving_faithfuls_to_frontiers
					has_county_modifier = pam_building_new_shrines
				}
			}
		}
		multiply = {
			value = 0.5
			desc = pam_mass_rite_conversion_feudalize_discount
		}
	}
	if = {
		limit = {
			government_has_mechanic = administrative
		}
		if = {
			limit = { realm_law_authority_level_3 = yes }
		}
		else_if = {
			limit = { realm_law_authority_level_2 = yes }
			add = {
				multiply = 0.15
				desc = has_admin_government
			}
		}
		else = {
			add = {
				multiply = 0.25
				desc = has_admin_government
			}
		}
	}
	if = { # Cheaper for Swadians expanding east after Rhodok Rebellion, against Battanian tribal holdings
		limit = {
			culture = {
				has_innovation = innovation_swadian_settling
			}
			scope:target.county = {
				culture = {
					has_cultural_pillar = heritage_battanian
				}
			}
		}
		subtract = {
			value = feudalize_holding_interaction_cost
			multiply = 0.25
			desc = swadian_conquest_tt
		}
	}
}

### PAM-RELATED VALUES
pam_heresy_pulse_no_event_value = {
	if = {
		limit = { fervor > 30 }
		value = 99
	}
	else_if = {
		limit = { fervor > 20 }
		value = 32
	}
	else_if = {
		limit = { fervor > 10 }
		value = 11
	}
	else = { value = 2 }
	# In the more turbulent phases we lower this, so outbreaks happen more often
	if = {
		limit = {
			religion = religion:calradian_religion
			situation:the_christian_church ?= {
				OR = {
					situation_current_phase = saeculum_obscurum
					situation_current_phase = investiture_crisis
				}
			}
		}
		divide = 1.5
	}
	else_if = {
		limit = {
			religion = religion:calradian_religion
			situation:the_christian_church ?= { situation_current_phase = concord }
		}
		multiply = 2
	}
	else_if = {
		limit = {
			religion = religion:calradian_religion
			situation:the_christian_church ?= {
				OR = {
					situation_current_phase = conversion_phase
					situation_current_phase = crusade_phase
				}
			}
		}
		multiply = 1.4
	}
	# For a small faith we make outbreaks much rarer
	if = {
		limit = { NOT = { pam_heresy_main_rite_substantial_trigger = yes } }
		multiply = 4
	}
	if = {
		limit = { has_game_rule = much_less_heresy_outbreak_frequency }
		multiply = 4
	}
	else_if = {
		limit = { has_game_rule = less_heresy_outbreak_frequency }
		multiply = 2
	}
	else_if = {
		limit = { has_game_rule = more_heresy_outbreak_frequency }
		multiply = 0.5
	}
	else_if = {
		limit = { has_game_rule = much_more_heresy_outbreak_frequency }
		multiply = 0.25
	}
	min = 1
}

#	root: the evaluating character
#	ec_desire_target: the doctrine being evaluated
ecumenical_council_doctrine_desire_core_value = { ### CK_DARUSOS CK_DARUSAN CK_CHURCH CK_WARBAND CK_TODO CK_120 - Maybe look into Ecumenical Councils on Mandatist stances? THEO A IRLFIT RUK EASTWOOD
	value = 0
	save_temporary_scope_as = voter
	# generally
	# good to me-> +100
	# very good to me-> +125
	# alternative option but still good to me -> +75
	# bad to opposite of me -> +50
	# bad to me -> -100
	# I don't like but not harmful to me -> -50

	switch = {
		trigger = scope:ec_desire_target
		doctrine:doctrine_monogamy = {
			# A chaste ruler favours monogamy over concubinage.
			if = {
				limit = {
					has_trait = chaste
					faith = {
						NOT = { has_doctrine = doctrine_monogamy }
					}
				}
				add = 50
			}
			else_if = {
				limit = {
					OR = {
						has_trait = lustful
						any_personal_tenet = { tenet_has_virtue = lustful }
						rite = { trait_is_virtue_rite = lustful }
						any_consort = { count > 1 }
					}
				}
				add = -100
			}
		}
		doctrine:doctrine_polygamy = {
			if = {
				limit = {
					the_christian_church_radical_change_allowed_trigger = yes
					OR = {
						has_trait = lustful
						any_personal_tenet = { tenet_has_virtue = lustful }
						rite = { trait_is_virtue_rite = lustful }
					}
					# don't switch between polygamy and concubines - it make no sense.
					faith = {
						has_doctrine = doctrine_monogamy
					}
				}
				add = 75
			}
			else_if = {
				limit = {
					OR = {
						has_trait = chaste
						any_consort = { count <= 1 }
					}
				}
				add = -50
			}
		}

				}

clerical_hesitation_claim_interaction_value = {
	if = {
		limit = {
			OR = {
				character_considered_wicked_trigger = { RITE = scope:cleric.rite }
				scope:cleric = {
					has_relation_rival = scope:actor
				}
				tyranny > medium_tyranny
				has_trait = excommunicated
			}
		}
		add = 4
	}
	else_if = {
		limit = {
			OR = {
				character_considered_morally_dubious_trigger = { RITE = scope:cleric.rite }
				scope:cleric = {
					is_ai = yes
					opinion = {
						target = scope:actor
						value <= -50
					}
				}
				tyranny >= low_tyranny
			}
		}
		add = 2
	}
	if = {
		limit = {
			OR = {
				piety_level >= 5
				has_trait = saint
				has_trait = herald
				has_trait = saoshyant_descendant
				has_trait = order_member
				has_trait = faith_warrior
				has_trait = saoshyant
				has_trait = paragon
				has_trait = savior
				has_trait = crusader_king
			}
		}
		add = -4
	}
	else_if = {
		limit = {
			OR = {
				piety_level >= 4
				has_trait = zealous
				has_trait = theologian
				has_trait = divine_blood
				has_trait = consecrated_blood
			}
		}
		add = -2
	}
	else_if = {
		limit = {
			piety_level >= 3
		}
		add = -1
	}
	#Your rite divergence
	if = {
		limit = {
			rite = {
				this != scope:cleric.rite
				"divergence(scope:cleric.rite)" >= monumental_rite_divergence_value
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			rite = {
				this != scope:cleric.rite
				"divergence(scope:cleric.rite)" >= massive_rite_divergence_value
			}
		}
		add = 1
	}
	#Scope saved in interactions that do not need these values
	if = {
		limit = {
			NOT = { exists = scope:only_consideration }
		}
		add = clerical_hesitation_from_target_holder_value
		add = clerical_hesitation_from_potential_defender_value
	}
}


clerical_hesitation_from_target_holder_value = {
	value = 0
	#Target holder exists
	if = {
		limit = {
			exists = scope:owner
		}
		#Target holder is wicked
		if = {
			limit = {
				scope:owner = {
					OR = {
						character_considered_wicked_trigger = { RITE = scope:cleric.rite }
						has_relation_rival = scope:cleric
						tyranny > medium_tyranny
						has_trait = excommunicated
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value >= faith_evil_level
							}
						}
					}
				}
			}
			add = -4
		}
		#Target holder is morally dubious
		else_if = {
			limit = {
				scope:owner = {
					OR = {
						character_considered_morally_dubious_trigger = { RITE = scope:cleric.rite }
						scope:cleric = {
							is_ai = yes
							opinion = {
								target = scope:owner
								value <= -50
							}
						}
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value >= faith_hostile_level
							}
						}
					}
				}
			}
			add = -2
		}
		#Target holder is same-faith, very pious
		if = {
			limit = {
				scope:owner = {
					faith = scope:cleric.faith
					OR = {
						piety_level >= 4
						has_trait = saint
						has_trait = herald
						has_trait = saoshyant_descendant
						has_trait = order_member
						has_trait = faith_warrior
						has_trait = saoshyant
						has_trait = paragon
						has_trait = savior
						has_trait = crusader_king
						has_trait = zealous
						has_trait = theologian
						has_trait = divine_blood
						has_trait = consecrated_blood
					}
					OR = {
						rite = scope:cleric.rite
						rite = {
							"divergence(scope:cleric.rite)" < massive_rite_divergence_value
						}
					}
				}
			}
			add = 4
		}
		#Target holder is same faith
		else_if = {
			limit = {
				scope:owner = {
					piety_level >= 1
					NOT = { has_trait = excommunicated }
					faith = scope:cleric.faith
					OR = {
						rite = scope:cleric.rite
						rite = {
							"divergence(scope:cleric.rite)" < monumental_rite_divergence_value
						}
					}
				}
			}
			add = 2
		}
		#Target holder is a friendly faith
		else_if = {
			limit = {
				scope:owner = {
					piety_level >= 1
					NOT = { has_trait = excommunicated }
					OR = {
						AND = {
							faith = scope:cleric.faith
							NOT = { rite = scope:cleric.rite }
						}
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value < faith_astray_level
							}
						}
					}
				}
			}
			add = 1
		}
	}
}

					faith_hostility_level = {
						target = scope:cleric.faith
						value >= faith_evil_level
					}
				}
			}
		}
		add = -2
	}
	#potential claim defender is morally dubious
	else_if = {
		limit = {
			OR = {
				character_considered_morally_dubious_trigger = { RITE = scope:cleric.rite }
				trigger_if = {
					limit = {
						scope:cleric = {
							is_ai = yes
						}
					}
					is_ai = yes
					reverse_opinion = {
						target = scope:cleric
						value <= -50
					}
				}
				faith = {
					faith_hostility_level = {
						target = scope:cleric.faith
						value >= faith_hostile_level
					}
				}
			}
		}
		add = -1
	}
	#potential claim defender is same-faith, very pious
	if = {
		limit = {
			faith = scope:cleric.faith
			OR = {
				piety_level >= 4
				has_trait = saint
				has_trait = herald
				has_trait = saoshyant_descendant
				has_trait = order_member
				has_trait = faith_warrior
				has_trait = saoshyant
				has_trait = paragon
				has_trait = savior
				has_trait = crusader_king
				has_trait = zealous
				has_trait = theologian
				has_trait = lifestyle_scholar
				has_trait = divine_blood
				has_trait = consecrated_blood
			}
			OR = {
				rite = scope:cleric.rite
				rite = {
					"divergence(scope:cleric.rite)" < massive_rite_divergence_value
				}
			}
		}
		add = 2
	}
	#potential claim defender is same faith
	else_if = {
		limit = {
			piety_level >= 1
			NOT = { has_trait = excommunicated }
			faith = scope:cleric.faith
			OR = {
				rite = scope:cleric.rite
				rite = {
					"divergence(scope:cleric.rite)" < monumental_rite_divergence_value
				}
			}
		}
		add = 1
	}
}
