### CLAIMANT-RELATED EFFECTS (WARBAND)
## applying when a province is found
apply_random_location_effect = {
	# set_location = {
	# 	location = province:$PROV$
	# 	stick_to_location = yes
	# }
	if = {
		limit = {
			location != province:$PROV$
		}
		move_to_pool_at = province:$PROV$
	}
	province:$PROV$ = {
		set_variable = { name = $FLAG$ value = yes }
	}
	if = {
		limit = {
			exists = location.county.holder
		}
		# if = { # this is currently bugged as random_character_in_location ignores the pool
		# 	limit = { is_lowborn = yes }
		# 	visit_court_of = location.county.holder
		# }
		# else = {
		if = {
			limit = {
				NOT = { exists = employer }
			}
			set_employer = location.county.holder
		}
		# }
	}
	else = { # log name + location
		debug_log_stack_trace = "unemployment rates are going up"
	}
}
