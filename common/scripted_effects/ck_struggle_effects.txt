### Struggle for Heaven's Mandate
ck_hm_struggle_apply_ai_agenda_effect = {
	# Apply effects
	add_character_flag = {
		flag = agenda_towards_$INTENT$
	}
	root = {
		change_variable = {
			name = agents_for_$INTENT$
			add = 1
		}
	}
}

hm_struggle_apply_catalyst_interloper_uninvolved_gain_struggle_titles = {
	if = {
		limit = {
			scope:war = {
				var:struggle_scope_defender ?= struggle:heaven_mandate_struggle
				var:struggle_involvement_defender ?= flag:involved
			}
			scope:attacker = {
				OR = {
					ck_uninvolved_in_struggle = { STRUGGLE = heaven_mandate_struggle }
					ck_interloper_in_struggle = { STRUGGLE = heaven_mandate_struggle }
				}
			}
			struggle:heaven_mandate_struggle = {
				phase_has_catalyst = catalyst_interloper_uninvolved_gain_struggle_titles
			}
		}
		struggle:heaven_mandate_struggle = {
			activate_struggle_catalyst = {
				catalyst = catalyst_interloper_uninvolved_gain_struggle_titles
				character = scope:attacker
			}
		}
	}
}
hm_remove_vassal_contract_cooldown_for_tension_effect = {
	if = {
		limit = {
			OR = {
				is_struggle_phase = struggle_heaven_phase_armistice
				is_struggle_phase = struggle_heaven_phase_compromise
			}
		}
		every_involved_ruler = {
			remove_interaction_cooldown = liege_modify_vassal_contract_interaction
			remove_interaction_cooldown = vassal_modify_vassal_contract_interaction
			remove_interaction_cooldown = ai_only_liege_modify_vassal_contract_interaction
			remove_interaction_cooldown = ai_only_vassal_modify_vassal_contract_interaction
		}
	}
}

### Restoring the Calradian Empire - WB
create_calradian_empire_wb_scripted_effect = {
	primary_title = { save_scope_as = old_primary_title }
	add_trait = augustus
	every_character_struggle = {
		limit = { phase_has_catalyst = catalyst_reformed_patriarchy }
		activate_struggle_catalyst = {
			catalyst = catalyst_restored_calradian_empire
			character = root
		}
	}
	if = {
		limit = { has_dlc_feature = legends }
		create_legend_seed = {
			type = legitimizing
			quality = illustrious
			chronicle = new_title
			properties = {
				title = title:h_calradian_empire
				founder = root
			}
		}
	}

	title:h_calradian_empire = {
		set_capital_county = root.capital_county
	}


	hidden_effect = {
		# Make sure the COA cannot change even if set in the effect previously
		if = {
			limit = {
				title:h_calradian_empire = {
					NOT = { has_variable = default_coa_var }
				}
			}
			title:h_calradian_empire = {
				set_variable = default_coa_var
			}
		}
		every_held_title = { #Should shift all dejure of all Empires owned at the time.
			title_tier = empire

			set_de_jure_liege_title = title:h_calradian_empire
		}
		every_empire = {
			limit = {
				NOT = { exists = hegemony }
				OR = {
					any_de_jure_county = {
						percent >= 0.6
						holder = {
							OR = {
								this = root
								any_liege_or_above = { this = root }
							}
						}
					}
					this = title:e_west_calradia
					this = title:e_khergits
					this = title:e_antipari
					this = title:e_myzea
				}
			}
			set_de_jure_liege_title = title:h_calradian_empire
		}
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:h_calradian_empire = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
		copy_title_history = title:e_calradian_empire
	}
	resolve_title_and_vassal_change = scope:change

	if = { # Move your Title MaAs to your new title if you have any
		limit = {
			government_allows = administrative
			primary_title = {
				any_title_maa_regiment = { }
			}
		}
		hidden_effect = {
			primary_title = { transfer_title_maa_ownership = title:h_calradian_empire }
		}
	}
	if = { # If you have founded the Varangian Guard, you get to keep the benefits
		limit = {
			scope:old_primary_title = { has_variable = founded_varangian_guard }
		}
		hidden_effect = {
			title:h_calradian_empire = { set_variable = founded_varangian_guard }
		}
	}
	hidden_effect = {
		every_player = {
			limit = {
				has_trait = claimant_emperor
			}
			remove_trait = claimant_emperor
		}
		set_primary_title_to = title:h_calradian_empire
		set_variable = { # Make it harder to vassal duchies and above, drive the player to expand manually by war and intrigue 
			name = ck_harder_vassalization
			value = yes
			years = 10
		}
	}
	every_held_title = { #Should destroy all other Empires owned at the time.
		title_tier = empire
		set_variable = {
			name = ck_block_catalyst
			value = yes
			days = 2 # To avoid catalyst firing for this particular thing
		}
		root = { destroy_title = prev }
	}
}

### Supremacy

ck_hm_struggle_ending_culture_list_effect = {
	every_culture_global = {		
		limit = {
			save_temporary_scope_as = this_culture
			struggle:heaven_mandate_struggle = { is_culture_involved_in_struggle = scope:this_culture }
			culture_number_of_counties > 0
		}
		save_scope_as = culture_scope
		add_to_global_variable_list = {
			name = ck_hm_struggle_ending_culture_list
			target = scope:culture_scope
		}
	}
}
ck_hm_struggle_ending_involved_list_effect = {
	save_scope_as = struggle_ender
	struggle:heaven_mandate_struggle = {
		every_involved_ruler = {
			limit = {
				top_liege = this
				primary_title = { is_mercenary_company = no }
			}
			add_to_list = struggle_involvees
		}
	}
	# Add all involved cultures with counties to a list, for war checks
	ck_hm_struggle_ending_culture_list_effect = yes
	# Add all involved faiths with counties to a list, for marriage and holy war checks
	ck_hm_struggle_ending_faith_list_effect = yes
}
# Add all Involved faiths with counties to a list
ck_hm_struggle_ending_faith_list_effect = {
	every_religion_global = {
		limit = {
			any_faith = {
				save_temporary_scope_as = this_faith
				struggle:heaven_mandate_struggle = { is_faith_involved_in_struggle = scope:this_faith }
				num_county_followers > 0
			}
		}
		every_faith = {
			limit = {
				struggle:heaven_mandate_struggle = { is_faith_involved_in_struggle = prev }
				num_county_followers > 0
			}
			save_scope_as = faith_scope
			add_to_global_variable_list = {
				name = ck_hm_struggle_ending_faith_list
				target = scope:faith_scope
			}
		}
	}
}
# Set cultural acceptance between all Involved cultures
ck_struggle_supremacy_cultural_acceptance_effect = {
	hidden_effect = {
		while = {
			limit = {
				any_in_global_list = {
					variable = ck_hm_struggle_ending_culture_list
					NOT = { has_variable = ck_struggle_supremacy_flag }
				}
			}
			ordered_in_global_list = {
				variable = ck_hm_struggle_ending_culture_list
				limit = {
					NOT = { has_variable = ck_struggle_supremacy_flag }
				}
				order_by = culture_number_of_counties
				save_scope_as = culture_scope
				set_variable = ck_struggle_supremacy_flag # variable ensures acceptance is only lost once per couple
			}
			every_in_global_list = {
				variable = ck_hm_struggle_ending_culture_list
				limit = {
					NOR = {
						has_variable = ck_struggle_supremacy_flag
						this = scope:culture_scope
					}
				}
				change_cultural_acceptance = {
					target = scope:culture_scope
					value = fp2_struggle_hostility_culture_acceptance_value
					desc = fp2_struggle_hostility_acceptance_tt
				}
			}
		}
		every_in_global_list = {
			variable = ck_hm_struggle_ending_culture_list
			remove_variable = ck_struggle_supremacy_flag
		}
	}
}
