	title:k_vlandia = {
		holder = {
			every_vassal_or_below = {
				limit = {
					NOT = {
						has_character_flag = rhodok_instigator_flag
					}
					age >= 16
					is_landed = yes
					NOT = { has_culture = culture:swadian }
					capital_province = { geographical_region = region_rhodokia }
				}
				trigger_event = {
					id = ck_global_culture.0019
					days = 1
				}
			}
		}
	}

# Remove Jizya tax - Realm Law
remove_jizya_tax_law = {
	if = {
		limit = {
			faith = {
				NOR = {
					has_doctrine_parameter = unlock_jizya_contract
					has_doctrine = tenet_tax_nonbelievers
				}
			}
			any_vassal = {
				vassal_contract_obligation_level_can_be_decreased = jizya_special_rights
			}
		}
		every_vassal = {
			limit = {
				vassal_contract_obligation_level_can_be_decreased = jizya_special_rights
			}
			vassal_contract_set_obligation_level = { type = jizya_special_rights level = 0 }
		}
	}
}

ck_wb_move_duchies_effect = {
	# Move duchies with capitals in other regions to a nearby kingdom instead
	if = {
		limit = {
			$WRONG_KINGDOM$ = {
				any_de_jure_county = {
					duchy = {
						title_capital_county.title_province = {
							ck_provinces_in_$GEO_REGION$ = yes
						}
					}
				}
			}
		}
		$WRONG_KINGDOM$ = {
			while = {
				limit = {
					any_de_jure_county = {
						duchy = {
							title_capital_county.title_province = {
								ck_provinces_in_$GEO_REGION$ = yes
							}
						}
					}
				}
				random_de_jure_county = {
					limit = {
						duchy = {
							title_capital_county.title_province = {
								ck_provinces_in_$GEO_REGION$ = yes
							}
							any_title_to_title_neighboring_duchy = {
								title_capital_county.title_province = {
									ck_provinces_in_$GEO_REGION$ = yes
								}
							}
						}
					}
					duchy = {
						random_title_to_title_neighboring_duchy = {
							limit = {
								title_capital_county.title_province = {
									ck_provinces_in_$GEO_REGION$ = yes
								}
							}
							kingdom = {
								save_temporary_scope_as = new_kingdom
							}
						}
						set_de_jure_liege_title = scope:new_kingdom
					}
				}
			}
		}
	}
}
ck_wb_destroy_old_empires_effect = {
	every_county_in_region = {
		region = wb_province_$PROVINCE_NAME$
		limit = {
			empire = {
				NOT = {
					has_variable = imperial_province
				}
			}
		}
		empire = {
			if = {
				limit = {
					exists = holder
					NOT = {
						has_variable = imperial_province
					}
					title_capital_county.title_province = { # Only if the empire's de jure capital is inside the province, otherwise, give them a fighting chance
						geographical_region = wb_province_$PROVINCE_NAME$
					}
				}
				holder = { destroy_title = prev }
			}
		}
	}
}

# takes $HERITAGE$, branching off at $CULTURE$
assign_culture_troop_vars = {
	custom_description = {
		text = set_martial_custom_from
		object = $CULTURE$
		set_variable = {
			name = troops_culture
			value = $CULTURE$
		}
	}
}
# takes $HERITAGE$, branching off at $CULTURE$
ck_reassign_culture_troop_vars = {
	every_culture_global = {
		limit = {
			has_cultural_pillar = heritage_$HERITAGE$
			ck_unique_culture_troop_vars_trigger = no # otherwise next loop will replace it
		}
		if = { # this could've been a $CULTURE$_unit_gfx
			limit = {
				exists = culture:$CULTURE$
				culture_is_or_has_ancestor_trigger = { CULTURE = culture:$CULTURE$ }
			}
			assign_culture_troop_vars = { CULTURE = culture:$CULTURE$ }
		}
		else = {
			assign_culture_troop_vars = { CULTURE = culture:$BASE$ }
		}
	}
}
