notify_senate_faction_effect = {
	scope:emperor_notification ?= {
		if = {
			limit = {
				has_trait = ck_$TRAIT$
			}
			send_interface_toast = {
				type = msg_senate_faction_$TRAIT$
				title = faction_$TRAIT$.notification
				desc = faction_$TRAIT$.notification_same
				left_icon = scope:root_icon
			}
		}
		else = {
			send_interface_toast = {
				type = msg_senate_faction_trait
				title = faction_$TRAIT$.notification
				desc = faction_$TRAIT$.notification_desc
				left_icon = scope:root_icon
				# show_as_tooltip = {
				# 	scope:root_icon = { add_trait_force_tooltip = ck_$TRAIT$ }
				# }
			}
		}
	}
}
set_senate_faction_effect = {
	## Senate Factions
	if = {
		limit = {
			culture = {
				has_cultural_pillar = heritage_calradian
			}
		}
		if = {	
			limit = {
				has_trait = ck_oligarch
			}
			remove_trait_force_tooltip = ck_oligarch
			add_prestige = medium_prestige_loss
		}
		else_if = {	
			limit = {
				has_trait = ck_militarist
			}
			remove_trait_force_tooltip = ck_militarist
			add_prestige = medium_prestige_loss
		}
		else_if = {	
			limit = {
				has_trait = ck_populist
			}
			remove_trait_force_tooltip = ck_populist
			add_prestige = medium_prestige_loss
		}
		else_if = {	
			limit = {
				has_trait = ck_royalist
			}
			remove_trait_force_tooltip = ck_royalist
			add_prestige = medium_prestige_loss
		}
	}
	else = {
		# If no current bhakti, cost is reduced.
		add_prestige = minor_prestige_loss
	}

	add_trait_force_tooltip = $IMPFACTION$
}
