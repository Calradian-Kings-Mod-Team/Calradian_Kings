			trigger = {
				any_courtier_or_guest = {
					save_temporary_scope_as = courtier_to_check

						faith_hostility_level = {
							target = scope:courtier_to_check.faith
							value >= faith_hostile_level
						}
					}
				}
			}
			random_courtier_or_guest = {
				limit = {
					save_temporary_scope_as = courtier_to_check

				top_liege ?= {
