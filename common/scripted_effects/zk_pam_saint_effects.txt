
# Canonizer is root, and this expects scope:saint_candidate and consecration_province already saved
pam_commit_canonization_effect = {
	if = {
		limit = { pam_needs_hof_approval_for_canonization_trigger = yes }
		faith.religious_head = { save_scope_as = petition_recipient }
		# Pick the wording here so saint_candidate / consecration_province / petition_recipient stay reachable in the loc key (SelectLocalization would drop them).
		if = {
			limit = {
				OR = {
					religion = religion:calradian_religion 
					rite ?= { rite_has_tenet = tenet_mystical_birthright }
				}
			}
			custom_tooltip = pam_canonize_saint_petition_sent_tt_saint
		}
		else = {
			custom_tooltip = pam_canonize_saint_petition_sent_tt_va
		}
		# Players acknowledge via event 0001, whose option starts the HoF's clock (0002 -> 0003/0004). The AI never sees that event, so schedule the HoF decision directly for it.
		if = {
			limit = { is_ai = no }
			trigger_event = pam_saint_events.0001
		}
		else = {
			hidden_effect = {
				scope:petition_recipient = {
					trigger_event = {
						id = pam_saint_events.0002
						days = 30
					}
				}
			}
		}
	}
	else = {
		custom_tooltip = pam_canonize_saint_decision_canonize_tt
		pam_canonize_saint_effect = {
			CANONIZER = root
			CANDIDATE = scope:saint_candidate
			CONSECRATION_PROVINCE = scope:consecration_province
			PRESET = no
		}
		# No HoF (or self-HoF): canonization is done, so show the finalization event now (players only).
		if = {
			limit = { is_ai = no }
			trigger_event = pam_saint_events.0003
		}
	}
}
