# Starting Holy Site Relics #
#
# Helpers used by historical_artifacts.0050 to seed enshrined relics at
# holy sites. The event runs two passes:
#   Pass 1: for every faith, at each of its eminent/regular holy sites, if the
#           county controller shares that faith, create one relic.
#   Pass 2: for every faith with at least 3 county followers, at each of
#           its eminent holy sites that still has no artifact, if the
#           controller's faith rates the diaspora faith astray-or-better,
#           create one relic anyway.
#
# Expected scopes on entry to any create_starting_relic_*_effect:
#   scope:relic_faith = the faith the relic is for (may not be the
#                       controller's faith in pass 2)
#   scope:relic_site  = the holy site scope
#   scope:relic_owner = the temporary artifact-creation-scope owner
#                       (the finalizer's set_holy_site_owner transfers
#                       ownership to the holy site itself)
#
# Dispatchers route on (relic_site, relic_faith.religion) to either a
# curated named-relic block or a per-religion generic block.

# Per-religion helpers vary names so sites do not clone the same Branch of the Tree of Life
create_starting_generic_relic_effect = {
	if = {
		limit = { scope:relic_faith.religion = religion:calradian_religion }
		create_starting_generic_christian_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:aserai_religion }
		create_starting_generic_islamic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:shantaya_religion }
		create_starting_generic_buddhist_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:ghavati_religion}
		create_starting_generic_hindu_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:darshi_religion }
		create_starting_generic_zoroastr_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:jumne_religion }
		create_starting_generic_germanic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:northern_gods_religion }
		create_starting_generic_slavic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		# Only Roog passes branch_boog_template is_holy_relic
		limit = { scope:relic_faith.religion = religion:gazima_religion }
		create_starting_generic_west_african_relic_effect = { RARITY = $RARITY$ }
	}
	else = {
		# No dedicated template: misc helper sets var:relic_religion on the artifact
		create_starting_generic_misc_relic_effect = { RARITY = $RARITY$ }
	}
}
