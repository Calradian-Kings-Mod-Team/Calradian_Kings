				}

	}

pam_cardinal_vacancy_on_death_effect = {
	if = {
		limit = {
			has_pam_dlc_trigger = yes
			OR = {
				exists = clerical_elector_title
				has_character_flag = cardinal_flag
			}
			OR = {
				exists = faith.religious_head
				exists = title:k_darusan.holder
			}
			faith = {
				has_doctrine_parameter = has_clerical_electors
				any_clerical_elector_title = { count < 10 }
			}
		}
		if = {
			limit = { exists = faith.religious_head }
			faith.religious_head = { save_scope_as = head_of_faith }
		}
		else = {
			title:k_darusan.holder = { save_scope_as = head_of_faith }
		}
		if = {
			limit = {
				scope:head_of_faith = { is_ai = yes }
			}
			faith = {
				ordered_clerical_region_title = {
					limit = {
						holder ?= {
							is_alive = yes
							is_adult = yes
							is_incapable = no
							NOT = { has_trait = excommunicated }
							NOT = { exists = clerical_elector_title }
							this != scope:head_of_faith
							pam_can_hold_cardinalate_trigger = yes
							pam_pope_is_head_of_faith_or_rite_trigger = yes
						}
					}
					order_by = holder.piety_level
					max = 1
					check_range_bounds = no
					holder = {
						save_scope_as = cardinal_candidate
						trigger_event = {
							id = pam_clergy_events.0040
							days = 3
						}
					}
				}
			}
		}
		else = {
			faith = {
				ordered_clerical_region_title = {
					limit = {
						holder ?= {
							is_ai = yes
							is_alive = yes
							is_adult = yes
							is_incapable = no
							NOT = { has_trait = excommunicated }
							NOT = { exists = clerical_elector_title }
							this != scope:head_of_faith
							pam_can_hold_cardinalate_trigger = yes
							pam_pope_is_head_of_faith_or_rite_trigger = yes
						}
					}
					order_by = holder.piety_level
					max = 1
					check_range_bounds = no
					holder = {
						save_scope_as = cardinal_candidate
					}
				}
			}
			if = {
				limit = {
					exists = scope:cardinal_candidate
				}
				scope:head_of_faith = {
					trigger_event = {
						id = pam_clergy_events.0041
						days = 3
					}
				}
			}
		}
	}
}

pam_update_schism_rite_variables_effect = {
	if = {
		limit = {
			#stop updating it if the schism is going to happen
			situation:the_christian_church ?= {
				NOT = { has_variable = pam_great_schism_status }
			}
		}
		if = {
			limit = { faith:calradic.main_rite ?= rite:darusan }
			set_variable = { name = schism_mainline_rite value = rite:darusan }
			set_variable = { name = schism_competitor_rite value = rite:calradic }
		}
		else_if = {
			limit = { faith:calradic.main_rite ?= rite:calradic }
			set_variable = { name = schism_mainline_rite value = rite:calradic }
			set_variable = { name = schism_competitor_rite value = rite:darusan }
		}
		else_if = {
			limit = { NOT = { has_variable = schism_mainline_rite } }
			set_variable = { name = schism_mainline_rite value = rite:darusan }
			set_variable = { name = schism_competitor_rite value = rite:calradic }
		}
	}
}
pam_apply_great_schism_creation_effect = {
	if = {
		limit = { NOT = { has_global_variable = pam_great_schism_happened } }
		set_global_variable = {
			name = pam_great_schism_happened
			value = current_year
		}
	}
	situation:the_christian_church.var:schism_mainline_rite = { save_scope_as = schism_mainline_rite }
	if = {
		limit = { scope:schism_mainline_rite = rite:darusan }
		faith:darusan = { save_scope_as = schism_mainline_faith }
		faith:calradic = { save_scope_as = schism_competitor_faith }
	}
	else = {
		faith:calradic = { save_scope_as = schism_mainline_faith }
		faith:darusan = { save_scope_as = schism_competitor_faith }
	}
	custom_tooltip = {
		text = pam_create_great_schism_effect_tt
		faith:calradic = {
			save_scope_as = old_rite
			every_faith_rite = {
				limit = { rite_has_doctrine = doctrine_no_head }
				change_rite_doctrine = doctrine_spiritual_head
			}
		}
		faith:darusan = { 
			set_religious_head_title = title:k_darusan
		}
		faith:calradic = { 
			set_religious_head_title = title:k_calradic
		}
		rite:darusan = {
			save_scope_as = roman_rite
			set_can_convert_to_rite = yes
			set_parent_faith = {
				target = faith:darusan
				main = yes
			}
		}
		rite:western_calradic = {
			set_can_convert_to_rite = no
			set_rite_enabled = no
			set_parent_faith = {
				target = faith:darusan
				main = no
			}
		}
		rite:ember_flame = {
			set_can_convert_to_rite = no
			set_rite_enabled = no
			set_parent_faith = {
				target = faith:darusan
				main = no
			}
		}
		rite:calradic = {
			save_scope_as = byzantine_rite
			set_can_convert_to_rite = yes
			set_parent_faith = {
				target = faith:calradic
				main = yes
			}
		}
		rite:hetjur = {
			# set_can_convert_to_rite = no
			# set_rite_enabled = no
			set_parent_faith = {
				target = faith:darusan
				main = no
			}
		}
		# struggle:iberian_struggle ?= {
		# 	set_faith_as_involved = rite:mozarabic_church.faith
		# 	set_faith_as_uninvolved = faith:christian_faith
		# }
	}
	show_as_tooltip = {
		faith:darusan = { set_religious_head_title = title:k_darusan }
	}
	hidden_effect = {
		situation:the_christian_church ?= {
			every_situation_participant = {
				limit = {
					OR = {
						var:pam_schism_stance ?= flag:mainline
						var:pam_schism_stance ?= flag:competitor
					}
					OR = {
						faith = faith:calradic
						faith = scope:schism_mainline_faith
						faith = scope:schism_competitor_faith
					}
					NOR = {
						any_held_title = { is_head_of_faith = yes }
						primary_title ?= { is_holy_order = yes }
					}
				}
				if = {
					limit = {
						var:pam_schism_stance ?= flag:mainline
						NOT = { faith = scope:schism_mainline_faith }
					}
					set_character_faith = scope:schism_mainline_faith
				}
				else_if = {
					limit = {
						var:pam_schism_stance ?= flag:competitor
						NOT = { faith = scope:schism_competitor_faith }
					}
					set_character_faith = scope:schism_competitor_faith
				}
			}
		}
		faith:calradic = {
			every_clerical_elector_title = {
				add_to_list = pam_schism_cardinalates
			}
		}
		every_in_list = {
			list = pam_schism_cardinalates
			save_temporary_scope_as = pam_schism_cardinalate
			if = {
				limit = {
					holder ?= {
						OR = {
							portrait_religious_catholic_trigger = yes
							AND = {
								faith = faith:calradic
								OR = {
									rite = rite:darusan
									rite = rite:hetjur
									rite = rite:ember_flame
									rite = rite:western_calradic
								}
							}
						}
					}
				}
				holder = {
					if = {
						limit = { NOT = { faith = faith:darusan } }
						set_character_faith = faith:darusan
					}
				}
				add_clerical_elector = faith:darusan
			}
			else_if = {
				limit = { exists = holder }
				holder = {
					pam_destroy_cardinal_title_effect = { CARDINALATE = scope:pam_schism_cardinalate }
				}
			}
		}
		faith:calradic = { remove_doctrine = special_doctrine_has_clerical_electors }
		# Catholicism starts with a College so AI clergy do not spam the Pope to fill vacancies
		faith:darusan = { pam_seed_cardinals_effect = yes }
		if = {
			limit = { exists = situation:the_christian_church }
			the_christian_church_set_main_power_faith_effect = yes
			the_christian_church_recalculate_participant_groups_effect = yes
		}
	}
}

pam_the_christian_church_county_conversion_catalysts_effect = {

			faith.religion = religion:calradian_religion

			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_county_converted_to_faith
		}
	}
	if = { # PAM - Catalyst when Christian county is converted away from Christianity
		limit = {
			faith != scope:old_faith
			scope:old_faith = {
				religion = religion:calradian_religion
			}
			faith.religion != religion:calradian_religion
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_county_converted_from_faith
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_county_converted_from_faith
		}
	}
	if = { # PAM - Jerusalem county is no longer Christian
		limit = {
			this = title:c_charas
			scope:old_faith = {
				religion = religion:calradian_religion
			}
			faith.religion != religion:calradian_religion
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_lose_jerusalem
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_lose_jerusalem
		}
	}
	if = { # PAM - Lycaron county is no longer Christian
		limit = {
			this = title:c_lycaron
			scope:old_faith = {
				religion = religion:calradian_religion
			}
			faith.religion != religion:calradian_religion
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_lose_jerusalem
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_lose_lycaron
		}
	}
	if = { # PAM - Other Christian holy site county is no longer Christian
		limit = {
			NOR = {
				this = title:c_charas
				this = title:c_lycaron
			}
			scope:old_faith = {
				religion = religion:calradian_religion
			}
			faith.religion != religion:calradian_religion
			any_county_province = {
				barony = { is_holy_site_of = scope:old_faith }
			}
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_lose_holy_site
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_lose_holy_site
		}
	}
}

# Shared on_accept for the three Sway to Rite interactions
# Option flags exist only on outreach, so the ?= checks do nothing elsewhere
pam_sway_to_rite_on_accept_effect = {
	if = {
		limit = { scope:is_puppet_action ?= yes }
		scope:puppet_or_actor = {
			if = {
				limit = { pam_sway_ruler_to_my_rite_cost_value > 0 }
				add_piety = {
					value = {
						add = pam_sway_ruler_to_my_rite_cost_value
						max = piety
						multiply = -1
					}
				}
			}
		}
	}
	scope:puppet_or_actor = {
		send_interface_toast = {
			type = event_toast_effect_good
			title = sway_ruler_to_my_rite_interaction_toast
			left_icon = scope:puppet_or_actor
			right_icon = scope:recipient
			scope:recipient = {
				set_character_rite = scope:puppet_or_actor.rite
				if = {
					limit = { is_ai = yes }
					add_opinion = {
						target = scope:puppet_or_actor
						modifier = pious_opinion
						opinion = 10
					}
				}
				if = { # Accepting the founding wave of a freshly created rite pulls the whole domain and court along, but only if they are actually tied to us
					limit = {
						is_ruler = yes
						is_landed = yes
						scope:puppet_or_actor.rite ?= {
							has_variable = pam_rite_founding_window
						}
						pam_rite_sway_connected_recipient_trigger = { ACTOR = scope:puppet_or_actor }
					}
					pam_convert_domain_counties_to_rite_effect = { RITE = scope:puppet_or_actor.rite }
					pam_convert_court_to_rite_effect = { RITE = scope:puppet_or_actor.rite }
				}
				else_if = { # We also convert their capital county
					limit = {
						is_ruler = yes
						is_landed = yes
						capital_county = {
							this.faith = scope:puppet_or_actor.faith # set_county_rite would drag a foreign-faith capital across faiths
							this.rite != scope:puppet_or_actor.rite
							any_neighboring_county = { # only if a neighbor shares the same liege or IS the liege, aka I'm not surrounded by complete strangers
								OR = {
									this.holder.liege = prev.holder.liege
									this.holder = prev.holder.liege
								}
							}
						}
					}
					capital_county = { set_county_rite = scope:puppet_or_actor.rite }
				}
				if = {
					limit = { scope:pam_sway_pledge_favor ?= yes }
					add_hook = {
						target = scope:puppet_or_actor
						type = favor_hook
					}
				}
			}
			if = {
				limit = { scope:pam_sway_endowment ?= yes }
				scope:puppet_or_actor = {
					pay_treasury_or_gold = {
						target = scope:recipient
						value = pam_sway_to_rite_endowment_cost_value
					}
				}
			}
		}
		# Catalysts
		if = {
			limit = {
				scope:puppet_or_actor.religion = religion:calradian_religion
				situation:the_christian_church ?= { situation_top_has_catalyst = catalyst_the_christian_church_swayed_to_rite }
			}
			situation:the_christian_church = { trigger_situation_catalyst = catalyst_the_christian_church_swayed_to_rite }
		}
	}
	if = {
		limit = {
			scope:is_puppet_action ?= yes
			scope:actor != scope:recipient
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = sway_ruler_to_my_rite_interaction_toast_puppeteer
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
			}
		}
	}
	scope:puppet_or_actor = {
		pam_tax_nonbelievers_conversion_fulfillment_effect = yes
	}
}

# PAM Great Schism: shows $TT$, and $VALUE$ goes up when positive (zero still shows the tooltip)
# Progress only goes up, and we do nothing once the schism has happened or been prevented
pam_change_great_schism_progress_effect = {
	custom_tooltip = $TT$
	hidden_effect = {
		if = {
			limit = { pam_great_schism_progress_can_change_trigger = yes }
			if = {
				limit = { $VALUE$ > 0 }
				situation:the_christian_church = {
					if = {
						limit = { NOT = { has_variable = pam_progress_towards_great_schism } }
						set_variable = { name = pam_progress_towards_great_schism value = 0 }
					}
					change_variable = {
						name = pam_progress_towards_great_schism
						add = $VALUE$
					}
				}
				every_player = {
					limit = {
						faith.religion = religion:calradian_religion
						exists = top_participant_group:the_christian_church
					}
					send_interface_message = {
						type = msg_pam_great_schism_progress
						desc = $TT$
					}
				}
			}
			if = {
				limit = {
					situation:the_christian_church = {
						NOT = { has_variable = pam_great_schism_status }
						pam_great_schism_progress_value >= pam_great_schism_progress_threshold_value
					}
				}
				situation:the_christian_church = {
					pam_update_schism_rite_variables_effect = yes
					var:schism_mainline_rite ?= { save_scope_as = schism_mainline_rite }
					var:schism_competitor_rite ?= { save_scope_as = schism_competitor_rite }
					set_variable = {
						name = pam_great_schism_status
						value = flag:threshold_reached
					}
				}
				scope:schism_mainline_rite.head_of_rite ?= { save_scope_as = schism_mainline_head }
				scope:schism_competitor_rite.head_of_rite ?= { save_scope_as = schism_competitor_head }
				every_player = {
					limit = {
						faith.religion = religion:calradian_religion
						any_character_situation = { this = situation:the_christian_church }
					}
					trigger_event = pam_the_christian_church.0064
				}
			}
		}
	}
}

		limit = {
			has_pam_dlc_trigger = yes
			current_date_is_start_date_trigger = no

		}
		situation:the_christian_church ?= {
			if = {
				limit = {
					exists = scope:pam_new_papal_dynasty
					has_variable = pam_papal_dynasty
					var:pam_papal_dynasty = scope:pam_new_papal_dynasty
				}
				change_variable = {
					name = pam_papal_dynasty_reigns
					add = 1
				}
			}
			else_if = {
				limit = { exists = scope:pam_new_papal_dynasty }
				set_variable = {
					name = pam_papal_dynasty
					value = scope:pam_new_papal_dynasty
				}
				set_variable = {
					name = pam_papal_dynasty_reigns
					value = 1
				}
				remove_variable = pam_papal_dynasty_capture_catalyst_fired
			}
			else = {
				remove_variable = pam_papal_dynasty
				remove_variable = pam_papal_dynasty_reigns
				remove_variable = pam_papal_dynasty_capture_catalyst_fired
			}
			if = {
				limit = {
					has_variable = pam_papal_dynasty_reigns
					var:pam_papal_dynasty_reigns >= @pam_papal_dynasty_capture_reigns
					NOT = { has_variable = pam_papal_dynasty_capture_catalyst_fired }
					situation_top_has_catalyst = catalyst_the_christian_church_papal_dynasty_capture
				}
				set_variable = pam_papal_dynasty_capture_catalyst_fired
				trigger_situation_catalyst = catalyst_the_christian_church_papal_dynasty_capture
			}
		}
	}

consecrate_kingdom_effect = {
	#If you don't have sway over the pope... they get sway over you
    if = {
    	limit = {
    		NOT = {
    			this = scope:religious_authority
    		}
    	}
    	if = {
    		limit = {
    			NOR = {
    				has_hook = scope:religious_authority
    				scope:king_aspirant.puppet:theological_agent_puppet ?= scope:religious_authority
    				scope:king_aspirant.puppet:external_theological_agent_puppet ?= scope:religious_authority
    			}
    		}
    		scope:religious_authority = {
    			add_hook = {
    				type = strong_obligation_hook
    				target = scope:king_aspirant
    			}
    		}
    	}
    	else = {
    		scope:religious_authority = {
    			add_hook = {
    				type = obligation_hook
    				target = scope:king_aspirant
    			}
    		}
    	}
    }
    if = {
    	limit = {
    		scope:religious_authority = {
    			this = scope:king_aspirant
    		}
    	}
    	every_held_title = {
    		limit = {
    			tier = tier_county
    		}
    		empire = {
    			every_de_jure_county = {
    				limit = {
    					holder ?= {
    						faith = scope:king_aspirant.faith
    						NOT = {
    							government_has_flag = government_is_theocracy
    						}
    					}	
    				}
    				holder = {
    					add_to_list = angry_seculars
    					every_liege_or_above = {
    						limit = {
    							faith = scope:king_aspirant.faith
    							NOT = {
    								government_has_flag = government_is_theocracy
    							}
    						}
    						add_to_list = angry_seculars
    					}
    				}
    			}
    		}
    	}
    	every_in_list = {
    		custom = custom.every_empire_secular_ruler
    		list = angry_seculars

    		add_opinion = {
    			modifier = power_hungry_priest_opinion
    			target = scope:king_aspirant
    			opinion = -30
    		}
    	}
    }
    create_custom_kingdom_effect = yes
    if = {
    	limit = {
    		NOT = {
    			this = scope:religious_authority
    		}
    		top_participant_group:the_christian_church ?= {
				OR = {
					participant_group_type = christian_clerical_main_power_rulers
					participant_group_type = christian_regular_main_power_rulers
				}
			}
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_consecrated_kingdom_for_ruler
			}
			NOT = {
				scope:king_aspirant = { government_has_flag = government_is_theocracy }
			}
    	}
    	situation:the_christian_church = {
    		trigger_situation_catalyst = catalyst_consecrated_kingdom_for_ruler
    	}
    }
    else_if = {
    	limit = {
    		this = scope:religious_authority
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_prelate_empowered_self_with_kingdom
			}
    	}
    	situation:the_christian_church = {
    		trigger_situation_catalyst = catalyst_prelate_empowered_self_with_kingdom
    	}
    }
    else_if = {
    	limit = {
    		top_participant_group:the_christian_church ?= {
				OR = {
					participant_group_type = christian_clerical_main_power_rulers
					participant_group_type = christian_regular_main_power_rulers
				}
			}
			situation:the_christian_church ?= {
				situation_top_has_catalyst = catalyst_consecrated_kingdom_for_theocrat
			}
			government_has_flag = government_is_theocracy
    	}
    	situation:the_christian_church = {
    		trigger_situation_catalyst = catalyst_consecrated_kingdom_for_theocrat
    	}
    }
    if = {
        limit = {
            is_ai = no
            NOT = { exists = global_var:found_kingdom_decision  }
        }
        set_global_variable = {
            name = found_kingdom_decision
            value = this #Unop Root is not always the new king, e.g. in pam_interactions_events
        }
        set_global_variable = {
            name = found_kingdom_decision_kingdom
            value = primary_title #Unop Root is not always the new king
        }
    }
}

update_syncretism_personal_tenet_effect = {
	if = {
		limit = {
			is_available_quick = { advanced_ruler = yes }
			scope:old_faith.religion != faith.religion
		}
		scope:old_faith = {
			if = {
				limit = { has_doctrine = sinitic_hostility_doctrine }
				prev = { add_known_tenet_if_not = { TENET = tenet_sinitic_syncretism } }
			}
			else_if = {
				limit = { has_doctrine = eastern_hostility_doctrine }
				prev = { add_known_tenet_if_not = { TENET = tenet_eastern_syncretism } }
			}
			else_if = {
				limit = { has_doctrine = pagan_hostility_doctrine }
				prev = { add_known_tenet_if_not = { TENET = tenet_unreformed_syncretism } }
			}
			else_if = {
				limit = { religion_tag = calradian_religion }
				prev = { add_known_tenet_if_not = { TENET = tenet_christian_syncretism } }
			}
			else_if = {
				limit = { religion_tag = aserai_religion }
				prev = { add_known_tenet_if_not = { TENET = tenet_islamic_syncretism } }
			}
			else_if = {
				limit = { religion_tag = surrani_religion }
				prev = { add_known_tenet_if_not = { TENET = tenet_jewish_syncretism } }
			}
		}
		faith = {
			if = {
				limit = { has_doctrine = sinitic_hostility_doctrine }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_sinitic_syncretism } }
			}
			else_if = {
				limit = { has_doctrine = eastern_hostility_doctrine }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_eastern_syncretism } }
			}
			else_if = {
				limit = { has_doctrine = pagan_hostility_doctrine }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_unreformed_syncretism } }
			}
			else_if = {
				limit = { religion_tag = calradian_religion }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_christian_syncretism } }
			}
			else_if = {
				limit = { religion_tag = aserai_religion }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_islamic_syncretism } }
			}
			else_if = {
				limit = { religion_tag = surrani_religion }
				prev = { remove_personal_tenet_if_has = { TENET = tenet_jewish_syncretism } }
			}
		}
	}
}
