## Mahdi Grove - Lost Effect
# Leave early with a chance of being lost
leave_grove_early_loss_effect = {
	if = {
		limit = {
			has_variable = grove_lost_chance_gui
		}
		random_list = {
			10 = { # Lost, scales by the lost variable
				trigger = {
					var:grove_lost_chance_gui > 0 
				}
				min = 0
				modifier = { # Scaled by the gui
					add = {
						value = 1
						multiply = var:grove_lost_chance_gui
					}
				}
				desc = ck_pilgrimage.0003.a.missing_in_grove_tt
				death = {
					death_reason = death_missing_in_grove
				}
				remove_variable = grove_lost_chance_gui
			}
			90 = {
				min = 0
				desc = ck_pilgrimage_leave_grove
				custom_tooltip = ck_pilgrimage_leave_grove_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = ck_pilgrimage.0050.b.succeed
				}
				remove_variable = grove_lost_chance_gui
			}
		}
	}
	ck_remove_character_flags = { FLAG = ck_0003_seeking_grove }
	ck_remove_character_flags = { FLAG = in_the_mahdi_grove }
	ck_remove_character_flags = { FLAG = in_west_nahasa_event_chain }
	ck_remove_character_flags = { FLAG = mahdi_grove_event_chain }
	scope:activity ?= { progress_activity_phase_after = { days = 1 } }
	stress_impact = {
		zealous = major_stress_impact_gain
	}
}
