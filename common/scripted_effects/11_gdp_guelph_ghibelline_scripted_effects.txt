try_set_up_new_gg_leader_effect = {
	save_temporary_scope_as = gg_departing_leader
	#Special handling to make sure Pope stays Guelph leader, and HRE inheritor stays Ghibelline leader
	if = {
		limit = { 
			gg_phase_is_pre_gg_trigger = yes
			exists = var:guelph_leader
		}
		if = {
			limit = {
				exists = title:k_darusan
				this != title:k_darusan.holder
				title:k_darusan.holder = {
					NOT = { exists = var:guelph_leader }
				}
			}
			title:k_darusan.holder = {
				set_variable = guelph_leader
				recalculate_participant_group = situation:guelph_ghibelline
				if = {
					limit = { NOT = { exists = var:gg_involvement } }
					set_variable = {
						name = gg_involvement
						value = gg_involvement_base_value
					}
				}
			}
			remove_variable = guelph_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}		
	}
	else_if = {
		limit = { 
			gg_phase_is_pre_gg_trigger = yes
			exists = var:ghibelline_leader
		}
		if = {
			limit = { 
				exists = title:e_west_calradia
				this != title:e_west_calradia.holder
				title:e_west_calradia.holder = {
					NOT = { exists = var:ghibelline_leader }
				} 
			}
			title:e_west_calradia.holder = {
				set_variable = ghibelline_leader
				recalculate_participant_group = situation:guelph_ghibelline
				if = {
					limit = { NOT = { exists = var:gg_involvement } }
					set_variable = {
						name = gg_involvement
						value = gg_involvement_base_value
					}
				}
			}
			remove_variable = ghibelline_leader
			recalculate_participant_group = situation:guelph_ghibelline
		}
	}
	else = {
		if = {
			limit = { has_variable = guelph_leader }
			if = {
				limit = {
					primary_heir ?= {
						this != scope:gg_departing_leader # Republics can re-elect, so the primary heir may be this dead ruler
						save_temporary_scope_as = heir_check
						situation:guelph_ghibelline ?= { 
							any_situation_participant = { this = scope:heir_check }
						}
						NOR = { 
							has_variable = ghibelline_leader
							any_held_title = { this = title:k_darusan }
						}
						top_liege = scope:gg_departing_leader
					}
				}
				primary_heir = {
					set_variable = guelph_leader
					save_scope_as = new_gg_leader
				}
			}
			else_if = {
				limit = {
					situation:guelph_ghibelline = {
						situation_participant_group:guelph_member = {
							any_situation_group_participant = { 
								count >= 1
								NOT = { 
									any_held_title = { this = title:k_darusan }
								}
							}
						}
					}
				}
				situation:guelph_ghibelline = {
					situation_participant_group:guelph_member = {
						ordered_situation_group_participant = {
							limit = { 
								NOT = { 
									any_held_title = { this = title:k_darusan }
								}
							}
							order_by = primary_title.tier
							set_variable = guelph_leader
							save_scope_as = new_gg_leader
						}
					}
				}
			}
			else = {
				save_scope_value_as = {
					name = missing_leader
					value = flag:guelph
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
				situation:guelph_ghibelline = { end_situation = yes }
			}
			remove_variable = guelph_leader
		}
		if = {
			limit = { has_variable = ghibelline_leader }
			if = {
				limit = {
					primary_heir ?= {
						this != scope:gg_departing_leader
						save_temporary_scope_as = heir_check
						situation:guelph_ghibelline ?= { 
							any_situation_participant = { this = scope:heir_check }
						}
						NOT = { has_variable = guelph_leader }
						top_liege = scope:gg_departing_leader
					}
				}
				primary_heir = { 
					set_variable = ghibelline_leader
					save_scope_as = new_gg_leader
				}
			}
			else_if = {
				limit = {
					situation:guelph_ghibelline = {
						situation_participant_group:ghibelline_member = {
							any_situation_group_participant = { count >= 1 }
						}
					}
				}
				situation:guelph_ghibelline = {
					situation_participant_group:ghibelline_member = {
						ordered_situation_group_participant = {
							order_by = primary_title.tier
							set_variable = ghibelline_leader
							save_scope_as = new_gg_leader
						}
					}
				}
			}
			else = {
				save_scope_value_as = {
					name = missing_leader
					value = flag:ghibelline
				}
				gg_invalidation_missing_leader_message_effect = { MISSING_LEADER = scope:missing_leader }
				situation:guelph_ghibelline = { end_situation = yes }
			}
			remove_variable = ghibelline_leader
		}
		if = {
			limit = { exists = situation:guelph_ghibelline }
			custom_tooltip = {
				text = new_gg_leader_tt
				scope:new_gg_leader = { recalculate_participant_group = situation:guelph_ghibelline }
				recalculate_participant_group = situation:guelph_ghibelline
			}
		}
	}
}

gg_clean_up_pope_modifiers_effect = {
	#Remove Pope-specific GG modifiers
	if = {
		limit = {
			faith = {
				OR = {
					religion != religion:calradian_religion
					has_doctrine_parameter = no_head_of_faith
				}
			}				
		}
		if = {
			limit = { has_character_modifier = favored_by_pope_modifier }
			remove_character_modifier = favored_by_pope_modifier
		}
		if = {
			limit = { has_character_modifier = scorned_pope_modifier }
			remove_character_modifier = scorned_pope_modifier
		}
	}
}

gg_situation_compromise_display_effect = {

					add = prev.gg_leader_compromise_renown

				add_dynasty_prestige = {

		}
		else = {
			custom_description_no_bullet = { text = gg_increased_by_independent_ghibelline_supporters }

			modifier = compromising_leader_modifier

			}
		}
		#Save the scope
		scope:title_to_check.holder ?= { create_lombard_league_effect = yes }
	}
}

create_lombard_league_effect = {
	#Ditch those shackles, man
	if = {
		limit = { is_independent_ruler = no }
		custom_description_no_bullet = {
			text = become_independent_effect
			gg_become_independent_effect = { TITLE = scope:league_creator.primary_title }
		}
	}
	#Create the Lombard League
	custom_description_no_bullet = {
		text = create_lombard_league_effect
		create_confederation = {
			name = lombard_league_name
			type = confederation_type:lombard_league
		}
	}
	if = {
		limit = { is_ai = yes }
		add_character_modifier = lombard_founder_modifier
	}
	#See who's available to join
	scope:new_confederation ?= {
		set_coa = lombard_league
	}
	#Notify the relevant people
	situation:guelph_ghibelline = {
		random_situation_participant = {
			limit = { has_variable = ghibelline_leader }
			save_scope_as = ghibelline_leader
		}
		every_situation_participant = {
			limit = { 
				is_ai = no
				save_temporary_scope_as = participant_check
				scope:new_confederation ?= {
					NOT = { 
						any_confederation_member = { this = scope:participant_check }
					}
				}
			}
			if = {
				limit = { scope:became_independent ?= scope:league_creator }
				send_interface_message = {
					type = msg_guelph_ghibelline_neutral_text
					title = gg_league_created.t
					left_icon = scope:league_creator
					desc = gg_league_created_and_independence.desc
				}
			}
			else = {
				send_interface_message = {
					type = msg_guelph_ghibelline_neutral_text
					title = gg_league_created.t
					left_icon = scope:league_creator
					desc = gg_league_created.desc
				}
			}
		}
		situation:guelph_ghibelline = {
			if = {
				limit = { 
					NOT = { situation_current_phase = agitation }
				}
				situation_top_sub_region = { change_phase = agitation }
			}		
		}
	}
	#... and make the HRE angry
	scope:ghibelline_leader = {
		if = {
			limit = { is_ai = no }
			custom_tooltip = hre_is_a_player_and_may_start_war_against_you.tt
			trigger_event = gdp_guelph_ghibelline.7010
		}
		else = {
			if = {
				limit = { is_at_war = no }
				custom_description_no_bullet = {
					text = hre_starts_war_ll_effect
					start_war = {
						casus_belli = "dissolve_lombard_league_cb"
						target = scope:league_creator
					}
				}
			}
		}
	}
	set_global_variable = {
		name = lombard_league_created
		value = scope:league_creator
	}
	situation:guelph_ghibelline = {
		record_situation_special_event = {
			key = gg_history_key_precursor_league_is_created
			variables = {
				league_creator = scope:league_creator
			}
		}
	}
}
