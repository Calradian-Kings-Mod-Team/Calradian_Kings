
# Raises $ANTIPOPE$ as an independent rival Head of Faith, sponsored by $SPONSOR$
pam_establish_antipope_effect = {
	$ANTIPOPE$ = { save_scope_as = antipope }
	$SPONSOR$ = { save_scope_as = sponsor }
	# A sponsored antipope takes up their sponsor's rite
	if = {
		limit = {
			scope:antipope != scope:sponsor
			scope:antipope = {
				NOT = { rite = scope:sponsor.rite }
			}
		}
		scope:antipope = {
			set_character_rite = scope:sponsor.rite
		}
	}
	if = {
		limit = { exists = scope:antipope.capital_county }
		scope:antipope.capital_county = { save_scope_as = capital_county }
	}
	else = {
		scope:sponsor.capital_county = { save_scope_as = capital_county }
	}
	scope:antipope = {
		faith = { save_scope_as = pam_antipope_faith }
		if = {
			limit = { exists = capital_province }
			capital_province = { save_scope_as = pam_antipope_capital }
		}
		hidden_effect_new_object = {
			if = {
				limit = {
					faith.religious_head_title.tier = tier_kingdom
				}
				create_dynamic_title = {
					tier = kingdom
					name = {
						first_valid = {
							triggered_desc = {
								trigger = { exists = scope:pam_antipope_capital }
								desc = PAM_ANTIPOPE_TITLE_NAME
							}
							desc = PAM_ANTIPOPE_TITLE_NAME_NO_CAPITAL
						}
					}
				}
			}
			else = {
				create_dynamic_title = {
					tier = duchy
					name = {
						first_valid = {
							triggered_desc = {
								trigger = { exists = scope:pam_antipope_capital }
								desc = PAM_ANTIPOPE_TITLE_NAME
							}
							desc = PAM_ANTIPOPE_TITLE_NAME_NO_CAPITAL
						}
					}
				}
			}
			scope:new_title = {
				save_scope_as = pam_antipope_title
				# Faith challenger list is dropped by code if the rite detaches into its own faith
				set_variable = { name = pam_antipope_office }
				set_destroy_if_invalid_heir = yes
				set_no_automatic_claims = yes
				set_definitive_form = yes
				set_always_follows_primary_heir = yes
				set_can_be_named_after_dynasty = no
				set_capital_county = scope:capital_county
				set_landless_title = yes
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = pam_antipope_title_change
				add_claim_on_loss = no
			}
			scope:pam_antipope_title = {
				change_title_holder = {
					holder = scope:antipope
					change = scope:pam_antipope_title_change
				}
			}
			resolve_title_and_vassal_change = scope:pam_antipope_title_change
			# Needs a holder before it can derive a coat of arms
			scope:pam_antipope_title = {
				generate_coa = religious_title
			}
			set_primary_title_to = scope:pam_antipope_title
		}
		# GOVERNMENT
		if = {
			limit = {
				NOT = { government_has_flag = government_is_theocracy }
				scope:sponsor != this
				theocratic_lay_clergy_trigger = yes
				trigger_if = {
					limit = {
						rite = { rite_has_doctrine = doctrine_theocracy_lay_clergy }
					}
					is_landed = no
				}
			}
			if = {
				limit = { religion = religion:calradian_religion }
				change_government = ecclesiastical_government
			}
			else = { change_government = theocracy_government }
		}
		# A rival Head of Faith answers to no secular liege
		if = {
			limit = { NOT = { this = top_liege } }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = pam_antipope_independence_change
				add_claim_on_loss = no
			}
			becomes_independent = { change = scope:pam_antipope_independence_change }
			resolve_title_and_vassal_change = scope:pam_antipope_independence_change
		}
	}
	scope:antipope = {
		if = {
			limit = { exists = clerical_elector_title }
			pam_destroy_cardinal_title_effect = { CARDINALATE = clerical_elector_title }
		}
	}
	if = {
		limit = { scope:antipope = scope:sponsor }
		hidden_effect_new_object = {
			scope:pam_antipope_title = {
				sponsor_new_religious_head_challenger = scope:pam_antipope_title
			}
		}
		pam_antipope_declared_catalyst_effect = yes
	}
	else = {
		hidden_effect_new_object = {
			scope:sponsor.primary_title ?= {
				sponsor_new_religious_head_challenger = scope:pam_antipope_title
			}
		}
		if = {
			limit = { has_pam_dlc_trigger = yes }
			pam_make_antipope_clerical_puppet_effect = {
				ANTIPOPE = scope:antipope
				SPONSOR = scope:sponsor
			}
		}
		# The sponsor gains a favor hook over the antipope they installed
		scope:sponsor = {
			add_hook_if_possible_default_length_effect = {
				TARGET = scope:antipope
				TYPE = favor_hook
			}
		}
		scope:antipope = {
			add_opinion = {
				modifier = grateful_opinion
				target = scope:sponsor
				opinion = 25
			}
		}
		pam_antipope_sponsored_catalyst_effect = yes
	}
	scope:antipope = {
		if = {
			limit = { exists = clerical_elector_title }
			pam_destroy_cardinal_title_effect = { CARDINALATE = clerical_elector_title }
			situation:the_christian_church ?= {
				add_manual_participant = prev
			}
		}
	}
	# Flavourization flag
	scope:antipope = {
		add_character_flag = antipope_flag
		hidden_effect_new_object = {
			pam_family_becomes_prelate_effect = {
				TITLE = scope:pam_antipope_title
				KIND = flag:antipope
			}
		}
	}
	# CE3 achievement: Guido's Guiding Light
}
pam_antipope_become_hof_effect = {
	$WINNER$ = { save_scope_as = new_hof }
	scope:new_hof.faith = { save_scope_as = antipope_faith }
	scope:antipope_faith = {
		if = {
			limit = { exists = religious_head_title }
			religious_head_title = { save_scope_as = hof_title }
		}
	}
	scope:antipope_faith.religious_head ?= { save_scope_as = old_hof }
	if = {
		limit = { exists = scope:old_hof }
		if = {
			limit = {
				scope:antipope_faith = { has_doctrine_parameter = spiritual_head_of_faith }
				scope:old_hof = { exists = capital_county }
			}
			scope:old_hof = {
				capital_county = { save_scope_as = old_hof_capital }
			}
		}
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = hof_titles_change
			add_claim_on_loss = no
		}
		scope:old_hof = {
			every_held_title = {
				limit = {
					tier >= tier_county
				}
				change_title_holder = {
					holder = scope:new_hof
					change = scope:hof_titles_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hof_titles_change
	}
	if = {
		limit = {
			scope:new_hof = {
				has_title = title:k_darusan
				faith = { has_doctrine_parameter = has_clerical_electors }
				NOT = { has_realm_law = theocratic_elective_succession_law }
			}
		}
		scope:new_hof = {
			add_realm_law_skip_effects = theocratic_elective_succession_law
		}
	}
	if = {
		limit = { exists = scope:hof_title }
		scope:antipope_faith = {
			set_religious_head_title = scope:hof_title
		}
	}
	if = {
		limit = {
			scope:antipope_faith = { has_doctrine_parameter = spiritual_head_of_faith }
			exists = scope:old_hof_capital
			scope:new_hof = {
				is_landed = yes
				NOT = { capital_county = scope:old_hof_capital }
			}
		}
		scope:new_hof = {
			set_realm_capital = scope:old_hof_capital
		}
	}
	scope:new_hof = {
		pam_set_pope_rite_mainline_effect = yes
	}
	pam_remove_antipope_effect = yes	
}
