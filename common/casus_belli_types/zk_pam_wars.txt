
##########
# Antiking — remove a rival's Antipope
##########

pam_antiking_cb = {
	icon = pam_antiking_cb
	group = religious
	ai_only_against_neighbors = yes
	allow_hostages = no
	is_holy_war = yes

	allowed_for_character = {
		is_landed = yes
		highest_held_title_tier >= tier_kingdom
		has_religion = religion:calradian_religion
		pam_is_antipope_sponsor_trigger = no
	}

	allowed_against_character = {
		pam_can_target_antiking_cb_trigger = yes
	}

	should_invalidate = {
		scope:defender = {
			NOT = { pam_is_antipope_sponsor_trigger = yes }
		}
	}

	on_invalidated_desc = msg_pam_antiking_cb_invalidated_message

	cost = {
		piety = {
			value = medium_piety_value
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						faith = {
							exists = religious_head
							religious_head != scope:attacker
						}
					}
				}
				desc = pam_antiking_cb_victory_desc_attacker
			}
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = pam_antiking_cb_victory_desc_attacker_no_gratitude
			}
			desc = pam_antiking_cb_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		pam_antiking_victory_effect = yes
		add_legitimacy_attacker_victory_effect = yes
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = pam_antiking_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = pam_antiking_cb_defeat_desc_attacker
			}
			desc = pam_antiking_cb_defeat_desc
		}
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { add_piety = medium_piety_loss }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "PAM_ANTIKING_WAR_CB_NAME"
	cb_name = "PAM_ANTIKING_WAR_CB_NAME"
	interface_priority = 77
	attacker_wargoal_percentage = 0.8

	ai_score = {
		value = 20
		if = {
			limit = {
				scope:attacker = {
					has_trait = excommunicated
				}
			}
			multiply = 0
		}
	}
}
