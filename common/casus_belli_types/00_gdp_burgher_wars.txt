burgher_war = {
	icon = peasant_war
	group = civil_war
	ai_only_against_liege = yes
	target_titles = none

	allowed_for_character = {
		scope:attacker = {
			is_leading_faction_type = burgher_autonomy_faction
		}
	}

	allowed_against_character = {
		scope:attacker = {
			joined_faction = {
				faction_leader = scope:attacker
				any_faction_county_member = {
					holder = {
						OR = {
							is_vassal_or_below_of = scope:defender
							this = scope:defender
						}
					}
				}
			}
		}
	}

	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no

	should_invalidate = {
		NOT = {
			scope:attacker = {
				has_variable = burgher_leader_of
				joined_faction ?= var:burgher_leader_of
			}
		}
	}

	on_invalidated_desc = {
		desc = burgher_war_ended_invalid.desc
	}

	on_invalidated = {
		scope:attacker = { burgher_dissolve_leader_effect = yes }
	}

	on_declaration = {
		on_declared_war = yes

		scope:defender = {
			if = {
				limit = { is_valid_for_legitimacy_change = yes }
				add_legitimacy = medium_legitimacy_loss
			}
		}

		scope:attacker = {
			joined_faction ?= {
				burgher_notify_sponsor_effect = {
					FACTION = this
					TOAST_TYPE = event_toast_effect_neutral
					TITLE = burgher_sponsor_war_declared_toast
					EFFECT = "custom_tooltip = burgher_sponsor_war_declared_toast_desc"
				}
			}
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { gdp_can_have_free_cities_trigger = yes }
				}
				desc = burgher_war_victory_free_city_desc
			}
			desc = burgher_war_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		scope:attacker = {
			joined_faction ?= {
				save_scope_as = burgher_faction

				burgher_resolve_county_effect = { FACTION = this }

				if = {
					limit = { exists = scope:burgher_county }
					scope:burgher_county = { save_scope_as = self_rule_county }
				}
				every_faction_county_member = {
					limit = {
						this != scope:self_rule_county
					}
					add_to_list = burgher_freed_towns
				}

				burgher_notify_sponsor_effect = {
					FACTION = this
					TOAST_TYPE = event_toast_effect_good
					TITLE = burgher_sponsor_self_rule_toast
					EFFECT = "custom_tooltip = burgher_sponsor_self_rule_toast_desc"
				}
			}
		}

		if = {
			limit = { exists = scope:self_rule_county }
			burgher_win_self_rule_effect = {
				LEADER = scope:attacker
				COUNTY = scope:self_rule_county
			}
			scope:attacker = { add_to_list = burgher_new_communes }
		}

		every_in_list = {
			list = burgher_freed_towns
			burgher_free_member_county_effect = { COUNTY = this }
		}

		if = {
			limit = { exists = scope:burgher_faction }
			every_in_list = {
				list = burgher_new_communes
				burgher_sponsor_republic_type_effect = {
					FACTION = scope:burgher_faction
					RULER = this
				}
			}
		}

		burgher_form_civic_league_effect = { FOUNDER = scope:attacker }

		if = {
			limit = { exists = scope:burgher_faction }

			if = {
				limit = { exists = scope:self_rule_county }
				scope:attacker = { save_scope_as = burgher_ruler }
				scope:self_rule_county = { save_scope_as = freed_county }
			}

			burgher_sponsor_spoils_effect = { FACTION = scope:burgher_faction }
		}

		scope:defender = {
			faction_war_end_defeat_legitimacy_effect = yes
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}

		scope:burgher_faction ?= { destroy_faction = yes }

		scope:attacker = { burgher_dissolve_leader_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		desc = burgher_war_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			stress_and_fulfillment_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
			joined_faction ?= {
				burgher_notify_sponsor_effect = {
					FACTION = this
					TOAST_TYPE = event_toast_effect_neutral
					TITLE = burgher_sponsor_white_peace_toast
					EFFECT = "custom_tooltip = burgher_sponsor_white_peace_toast_desc"
				}
				set_variable = burgher_outcome_reported
				if = {
					limit = {
						exists = var:burgher_county
					}
					burgher_grant_lesser_privilege_effect = { COUNTY = var:burgher_county }
				}

				every_faction_county_member = {
					set_variable = {
						name = burgher_faction_cooldown
						years = burgher_white_peace_cooldown_years_value
					}
				}

				destroy_faction = yes
			}
		}

		scope:attacker = { burgher_dissolve_leader_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		desc = burgher_war_defeat_desc
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			add_prestige = medium_prestige_gain
		}

		scope:attacker = {
			joined_faction ?= {
				burgher_notify_sponsor_effect = {
					FACTION = this
					TOAST_TYPE = event_toast_effect_bad
					TITLE = burgher_sponsor_defeat_toast
					EFFECT = "custom_tooltip = burgher_sponsor_defeat_toast_desc"
				}
				set_variable = burgher_outcome_reported
				every_faction_county_member = {
					save_scope_value_as = {
						name = crushed_liberties_hit
						value = burgher_revoke_liberties_opinion_value
					}

					burgher_revoke_privileges_effect = { COUNTY = this }
					if = {
						limit = { scope:crushed_liberties_hit < 0 }
						add_popular_opinion = scope:crushed_liberties_hit
					}

					add_county_modifier = {
						modifier = burgher_grievance_county_modifier
						years = burgher_grievance_years_value
					}

					set_variable = {
						name = burgher_faction_cooldown
						years = burgher_defeat_cooldown_years_value
					}
				}

				destroy_faction = yes
			}
		}

		scope:attacker = { burgher_dissolve_leader_effect = yes }

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "BURGHER_WAR_NAME"
	cb_name = "BURGHER_WAR_CB_NAME"
	interface_priority = 70

	ticking_war_score_targets_entire_realm = yes
	attacker_ticking_warscore = 0.5
	attacker_score_from_occupation_scale = 100
	defender_score_from_occupation_scale = 100
	attacker_score_from_battles_scale = 300
	defender_score_from_battles_scale = 300

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100

	ai_score = {
		value = 100
	}
}

dissolve_the_hanse_cb = {
	icon = lombard_league_cb
	group = subjugation
	combine_into_one = yes
	ai_only_against_neighbors = yes
	allow_hostages = yes
	should_show_war_goal_subview = no

	allowed_for_character = {
		always = no
	}

	allowed_against_character = {
		always = no
	}

	cost = {
		piety = {
			value = 0
			add = common_cb_impious_piety_cost
		}
		prestige = {
			value = medium_prestige_value
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	should_invalidate = {
		OR = {
			scope:defender = {
				NOT = {
					confederation ?= { has_variable = is_hanseatic_league }
				}
			}
			scope:attacker = {
				is_member_of_confederation = scope:defender.confederation
			}
		}
	}

	on_invalidated_desc = {
		desc = dissolve_the_hanse_cb_ended_invalid.desc
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = dissolve_the_hanse_cb_victory_desc
	}

	on_victory = {
		add_legitimacy_attacker_victory_effect = yes

		scope:attacker = {
			show_pow_release_message_effect = yes
			accolade_attacker_war_end_glory_gain_low_effect = yes

			switch = {
				trigger = scope:defender.primary_title.tier
				tier_county = { add_prestige = medium_prestige_gain }
				tier_duchy = { add_prestige = major_prestige_gain }
				tier_kingdom = { add_prestige = major_prestige_gain }
				tier_empire = { add_prestige = massive_prestige_gain }
				tier_hegemony = { add_prestige = monumental_prestige_gain }
			}
		}

		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:defender = {
			confederation ?= {
				disband_confederation = yes
			}
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_victory_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_white_peace_desc = {
		desc = dissolve_the_hanse_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
			stress_and_fulfillment_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		scope:defender = {
			accolade_defender_war_end_glory_gain_low_effect = yes
			confederation ?= {
				change_cohesion = dissolve_hanse_white_peace_cohesion_value
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_white_peace_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		desc = dissolve_the_hanse_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_legitimacy_attacker_defeat_effect = yes
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
			add_prestige = {
				value = major_prestige_value
				multiply = -1.0
			}
		}
		scope:defender = {
			mandala_peacemaker_perk_serenity_effect = yes
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = major_prestige_value
			}
			confederation ?= {
				change_cohesion = dissolve_hanse_defeat_cohesion_value
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		add_truce_attacker_defeat_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "DISSOLVE_THE_HANSE_WAR_NAME"
	war_name_base = "DISSOLVE_THE_HANSE_WAR_NAME_BASE"
	cb_name = "DISSOLVE_THE_HANSE_CB_NAME"
	interface_priority = 80

	ticking_war_score_targets_entire_realm = yes
	attacker_ticking_warscore = 0.01
	attacker_wargoal_percentage = 0.01
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 250
	defender_score_from_battles_scale = 250

	max_attacker_score_from_battles = 200
	max_defender_score_from_battles = 200

	max_defender_score_from_occupation = 50
	max_attacker_score_from_occupation = 50

	max_ai_diplo_distance_to_title = 500

	ai_score = {
		value = 1
	}
}
