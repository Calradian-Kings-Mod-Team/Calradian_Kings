tenet_djinns = {
	icon = core_tenet_djinnism

	piety_cost = {
		value = faith_tenet_cost_low

		# Multiplier for keeping same tenet
		if = {
			limit = {
				rite = { rite_has_tenet = tenet_djinns }
			}
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	is_shown = {
		religion = { is_in_family = rf_nahasan }
	}
	can_pick = {
		custom_description = {
			text = INCOMPATIBLE_TENET_ADORCISM_TRIGGER
			NOT = { tenet:tenet_adorcism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = INCOMPATIBLE_TENET_BAD_DJINN
			NOT = { tenet:tenet_djinns_positive = { is_in_list = selected_doctrines } }
		}
	}

	character_modifier = {
		monthly_county_control_growth_factor = 0.1
		desert_attrition_mult = -0.25
		desert_mountains_attrition_mult = -0.25
		desert_travel_danger = desert_low_danger_reduction
		drylands_travel_danger = drylands_low_danger_reduction
		desert_mountains_travel_danger = desert_mountains_low_danger_reduction
	}

	parameters = {
		spirit_possession_crime
	}

	traits = {
		sins = {  
			possessed_1 = { weight = 10 scale = 1}
			possessed_genetic = { weight = 10 scale = 1}
		}
	}
}

## Vlandian / Balionic Tenets (Calradic Religion) ##
# Vlandian Honor System #
tenet_vlandian_honor = {
	icon = core_tenet_vlandian
	piety_cost = {
		value = faith_tenet_cost_mid

		# Multiplier for keeping same tenet
		if = {
			limit = { rite = { rite_has_tenet = tenet_vlandian_honor } }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	is_shown = {
		rite = { rite_has_tenet = tenet_vlandian_honor }
	}

	can_pick = {
		custom_description = {
			text = incompatible_tenet_pacifism_trigger
			NOT = { tenet:tenet_pacifism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = incompatible_tenet_dharmic_pacifism_trigger
			NOT = { tenet:tenet_dharmic_pacifism = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		#warmonger = yes
		invasion_cb_enabled
		clergy_can_fight
	}
	
	character_modifier = {
		ai_war_chance = 0.5
		ai_war_cooldown = -0.2
		accolade_glory_gain_mult = 0.25
	}

	traits = {
		virtues = { 
			brave = { weight = 25 scale = 2 }
		}
	}
}
