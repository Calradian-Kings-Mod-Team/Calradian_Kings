		ai_rationality = 30 # remove penalty if you get do catch it

	}

	icon = core_tenet_vlandian
	piety_cost = {

		if = {
			limit = {
				rite_has_tenet = tenet_vlandian_honor
			}
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	# is_shown = {
	# 	any_faith_rite = {
	# 		rite_has_tenet = tenet_vlandian_honor
	# 	}
	# }

	can_pick = {
		custom_description = {
			text = incompatible_tenet_pacifism_trigger
			NOT = { tenet:tenet_pacifism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = incompatible_tenet_dharmic_pacifism_trigger
			NOT = { tenet:tenet_dharmic_pacifism = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		#warmonger = yes
		invasion_cb_enabled
		clergy_can_fight
	}
	
	character_modifier = {
		ai_war_chance = 0.5
		ai_war_cooldown = -0.2
		accolade_glory_gain_mult = 0.25
		monthly_war_prestige_income_mult = 0.05
	}

	traits = {
		virtues = { 
			brave = { weight = 25 scale = 2 }
		}
	}
	can_pick_as_personal_tenet = { # check heritage_balian to prevent picking as swadian?
		culture = {
			OR = {
				has_cultural_tradition = tradition_vlandian_banneret # new hybrid
				# has_innovation = innovation_visigothic_codes # should be covered above
				culture_is_or_has_ancestor_trigger = { CULTURE = culture:balian }
				ck_can_recruit_culture_troop_vars = {CULTURE = vlandian }
			}
			NOR = { # rhodok is from kassen + togerd?
				has_cultural_tradition = tradition_southern_balion_military
				# ck_can_recruit_culture_troop_vars = {CULTURE = rhoyd }
				# ck_can_recruit_culture_troop_vars = {CULTURE = rhodok }
			}
		}
	}
	personal_tenet_modifier = {
		heavy_cavalry_maintenance_mult = -0.05
		glory_hound_ai_boldness = 10
	}
	personal_tenet_parameters = {
		# tenet_dulia_personal_sainthood_chance
		tenet_reincarnation_piety_level_courage
	}
}
