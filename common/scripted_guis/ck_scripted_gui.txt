## Companion Objections
companion_objections_gui = {
	scope = character

	is_shown = {
		is_landless_adventurer = yes
		any_courtier = {
			OR = {
				has_character_flag = ck_has_casualty_objection
				has_character_flag = ck_has_quest_objection
				has_character_flag = ck_has_food_objection
			}
		}
	}
}
companion_has_objections_gui = {
	scope = character 

	is_shown = {
		OR = {
			has_character_flag = ck_has_casualty_objection
			has_character_flag = ck_has_quest_objection
			has_character_flag = ck_has_food_objection
		}
	}
}
