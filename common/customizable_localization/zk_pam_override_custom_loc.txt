#00_compliment_custom_loc

ComplimentSentence = {
	type = character

	random_valid = yes

	text = {
		localization_key = compliment_sentence_comfortable
		trigger = {
			OR = {
				has_any_good_relationship_with_root_trigger = yes
				reverse_opinion = {
					target = root
					value >= 20
				}
			}
		}
	}

	text = { # Beauty — CK2 letter flattery register
		localization_key = compliment_sentence_beauty_poets
		trigger = {
			has_trait = beauty_good
			is_adult = yes
		}
	}

	text = { # Valor
		localization_key = compliment_sentence_valor_crown
		trigger = {
			OR = {
				has_trait = brave
				prowess >= high_skill_rating
				martial >= high_skill_rating
			}
		}
	}

	text = { # Wit / learning
		localization_key = compliment_sentence_wit_frankincense
		trigger = {
			OR = {
				has_trait = intellect_good
				has_trait = shrewd
				learning >= high_skill_rating
				diplomacy >= high_skill_rating
			}
		}
	}

	text = { # Courtly longing — warm relations
		localization_key = compliment_sentence_court_duller
		trigger = {
			OR = {
				has_any_good_relationship_with_root_trigger = yes
				reverse_opinion = {
					target = root
					value >= 40
				}
				has_relation_lover = root
				has_relation_soulmate = root
				has_relation_friend = root
			}
		}
	}

	text = { # Counsel
		localization_key = compliment_sentence_counsel_lamp
		trigger = {
			OR = {
				learning >= high_skill_rating
				stewardship >= high_skill_rating
				diplomacy >= high_skill_rating
			}
		}
	}

	text = { # Virtue — Christian complimented character (saints)
		localization_key = compliment_sentence_virtue_saints
		trigger = {
			OR = {
				has_trait = zealous
				has_trait = honest
				has_trait = chaste
				piety_level >= 3
			}
			faith = { religion_tag = calradian_religion }
		}
	}

	text = { # CK2-style rainy-sea eyes
		localization_key = compliment_sentence_eyes_sea
		trigger = {
			OR = {
				has_trait = beauty_good
				has_relation_lover = root
				has_relation_soulmate = root
			}
			is_adult = yes
		}
	}

	text = {
		localization_key = compliment_sentence_hours_crawl
		trigger = {
			OR = {
				has_relation_lover = root
				has_relation_soulmate = root
				has_relation_friend = root
				reverse_opinion = {
					target = root
					value >= 50
				}
			}
		}
	}


	text = { # Fame / renown
		localization_key = compliment_sentence_fame_horse
		trigger = {
			OR = {
				prestige_level >= 2
				has_trait = ambitious
				has_trait = arrogant
			}
		}
	}

	text = { # Courtesy / diplomacy
		localization_key = compliment_sentence_courtesy_sword
		trigger = {
			OR = {
				diplomacy >= high_skill_rating
				has_trait = gregarious
				has_trait = charming
			}
		}
	}

	text = { # Generosity
		localization_key = compliment_sentence_gold_pales
		trigger = {
			OR = {
				has_trait = generous
				ai_greed <= medium_negative_ai_value
			}
		}
	}

	text = { # Letters as relics — warm bond, Christian complimented
		localization_key = compliment_sentence_letters_relics
		trigger = {
			faith = { religion_tag = calradian_religion }
			OR = {
				has_relation_lover = root
				has_relation_soulmate = root
				has_relation_friend = root
				reverse_opinion = {
					target = root
					value >= 50
				}
			}
		}
	}

	text = { # Steadfast honor
		localization_key = compliment_sentence_oath_iron
		trigger = {
			OR = {
				has_trait = honest
				has_trait = just
				ai_honor >= medium_positive_ai_value
			}
		}
	}

	text = {
		fallback = yes
		localization_key = compliment_sentence_fallback_company
	}
}

	type = character

	text = {
		localization_key = sanctified_artifact_prefix_venerated
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_ancestral } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_sacred
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_natural } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_anointed
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_sacrificial } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_radiant
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_regal } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_sanctified
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_humble } }
	}

	# Zoroastrianism belongs to rf_eastern, so it has to be checked before that family below.
	text = {
		localization_key = sanctified_artifact_prefix_purified
		trigger = { faith = { religion_tag = darshi_religion } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_hallowed
		trigger = { faith = { is_in_family = rf_pagan } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_auspicious
		trigger = {
			faith = {
				OR = {
					is_in_family = rf_eastern
					is_in_family = rf_sinitic
				}
			}
		}
	}

	text = {
		localization_key = sanctified_artifact_prefix_blessed
		fallback = yes
	}
}
