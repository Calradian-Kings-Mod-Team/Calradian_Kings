			is_landless_adventurer = yes

				OR = {
					has_character_flag = ck_has_casualty_objection

					has_character_flag = ck_has_food_objection
				}
			}
		}
		localization_key = HasCompanionGrievances
	}
}

#### Pilgrimage-related Custom Localization
ck_madhi_grove_success = {
	type = character

	text = { # 
		trigger = {
			has_variable = grove_success_chance_gui
			var:grove_success_chance_gui >= 7
		}
		localization_key = ck_madhi_grove_success_crit
	}
	text = { # 
		trigger = {
			has_variable = grove_success_chance_gui
			AND = {
				var:grove_success_chance_gui < 7
				var:grove_success_chance_gui >= 4
			}
		}
		localization_key = ck_madhi_grove_success
	}
	text = {
		trigger = {
			has_variable = grove_success_chance_gui
			AND = {
				var:grove_success_chance_gui < 4
				var:grove_success_chance_gui > 0
			}
		}
		localization_key = ck_madhi_grove_success_slight
	}
	text = {
		trigger = {
			has_variable = grove_success_chance_gui
			AND = {
				var:grove_success_chance_gui <= 0
				var:grove_success_chance_gui > -3
			}
		}
		localization_key = ck_madhi_grove_success_failure
	}
	text = {
		trigger = {
			has_variable = grove_success_chance_gui
			var:grove_success_chance_gui <= -3
		}
		localization_key = ck_madhi_grove_success_failure_crit
	}
}
