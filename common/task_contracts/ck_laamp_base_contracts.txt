# Deliver Herd / Employer is need of livestock, whether to help bolster their numbers or in desperate need. Contractor will go to several locations to purchase livestock and deliver them to the employer. Hazards along the way. / If employer is steppe, they'll get a herd boost. Landed get a county modifier.
laamp_ck_deliver_herd = {
	group = laamp_contracts_transport_group
	icon = "gfx/interface/icons/character_interactions/icon_contract_escort.dds"
	travel = no
	use_diplomatic_range = no

	# Validity Triggers
	valid_to_create = {
		# Standard triggers.
		valid_laamp_basic_trigger = {
			EMPLOYER = scope:employer
			LAAMP = root
		}
		employer_has_treasury_to_offer_job_trigger = yes
		valid_laamp_sensible_start_trigger = yes
		rule_out_dramatic_laamp_employers_trigger = yes # This sort of thing is beneath an Emperor
		# Specific flavour triggers.
		scope:employer = {
			capital_province ?= {
				NOT = { # Avoid provinces in the uninhabitable parts of the Nahasa, shouldn't be a problem but just being sure.
					geographical_region = custom_nahasa_unhabitable
				}
			}
			OR = { # Low development, income, or modifiers relating to livestock
				monthly_character_income < 0 # Want some herd to help bolster income - The treasury trigger (employer_has_treasury_to_offer_job_trigger) above handles the minimum gold they have to pay out for this contract. Negative income will pay out less at end of contract.
				has_character_flag = ck_stolen_herd_flag # Their herd was recently stolen (Steppe Interaction)
				any_sub_realm_county = { # Can be any of the employer's counties. If any of these are true, the 
					OR = {
						has_county_modifier = martial_chivalry_stolen_livestock_modifier # A modifier added if someone's livestock is stolen (Chivalry Lifestyle Event - martial_chivalry.4000)
						development_level <= 10 # Specifically chose the because we want development to increase
					}
				}
			}
			# any_neighboring_top_liege_realm_owner = { # We want a neighbor to pull livestock from
			# 	is_within_diplo_range = { CHARACTER = scope:employer }
			# 	# NOT = {
			# 	# 	has_character_flag = ck_stolen_herd_flag # Their herd was recently stolen (Steppe Interaction)
			# 	# }
			# 	any_sub_realm_county = {
			# 		NOT = {
			# 			has_county_modifier = martial_chivalry_stolen_livestock_modifier # A modifier added if someone's livestock is stolen (Chivalry Lifestyle Event - martial_chivalry.4000), shouldn't pull from another county with stolen livestock
			# 		}
			# 		any_county_province = {
			# 			squared_distance = {
			# 				target = scope:employer.capital_province # We want a nearby county to pull from within reasonable distance of the employer
			# 				value = 85000 # Wales + Ireland (Top-to-bottom distance, given our own map size too)
			# 			}
			# 		}
			# 	}
			# }
		}
	}
	valid_to_keep = {
		# exists = root:var:escorted_ck_herd
		# Standard triggers.
		valid_laamp_basic_trigger = {
			EMPLOYER = root.task_contract_employer
			LAAMP = root.task_contract_taker
		}
		task_contract_employer = { is_landed = yes }
	}
	valid_to_accept = {
		# Standard triggers.
		valid_laamp_basic_accept_only_trigger = yes
		valid_laamp_basic_trigger = {
			EMPLOYER = scope:employer
			LAAMP = root
		}
		scope:employer = { is_landed = yes }
	}
	valid_to_continue = {
		# exists = var:escorted_ck_herd
		# Standard triggers.
		valid_laamp_basic_trigger = {
			EMPLOYER = root.task_contract_employer
			LAAMP = root.task_contract_taker
		}
		task_contract_employer = { is_landed = yes }
	}

	weight = {
		# Standard weights.
		value = task_contract_weight_default_value
		add = laamp_contracts_weight_up_transport_value
		add = task_contract_weight_employer_contact_list_value
		# Employer Weights
		scope:employer = {
			add = {
				# Arrogant a-holes don't like asking for things
				if = {
					limit = { has_trait = arrogant }
					add = task_contract_weight_malus_employer_has_associated_traits_value
				}
				# Impatient Employers want this done FAST
				if = {
					limit = { has_trait = impatient }
					add = task_contract_weight_bonus_employer_has_associated_traits_value
				}
				# Nomads want new herds
				if = {
					limit = { has_character_flag = ck_stolen_herd_flag }
					add = task_contract_weight_bonus_employer_has_associated_traits_value
				}
				# Restrictions.
				min = task_contract_weight_employer_associated_traits_min_value
				max = task_contract_weight_employer_associated_traits_max_value
			}
			multiply = task_contract_weight_by_tier_value
			# This weight is an override, so make sure it's placed at the bottom.
			add = task_contract_weight_employer_contact_list_value # This is if they're your contact and know to trust you
		}
	}

	# On_Actions
	on_create = {
		scope:contract = {
			# Make our gold rewards static.
			grab_gold_fuzz_number_effect = yes
			save_scope_as = task_contract
			set_variable = {
				name = gold_success_substandard
				value = task_contract_taker.task_contract_success_gold_gain_half_value
			}
			set_variable = {
				name = gold_success_standard
				value = task_contract_taker.task_contract_success_gold_gain_full_value
			}
			# Make sure critical values are a bit higher.
			save_scope_value_as = { name = gold_safety_margin value = flag:yes }
			set_variable = {
				name = gold_success_critical
				value = task_contract_taker.task_contract_success_gold_gain_full_value
			}

		}
	}

	on_accepted = {
		task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_accept_contract" }
		# For the intro event.
		save_scope_as = task_contract
		task_contract_employer = {
			save_scope_as = task_contract_employer
			save_scope_as = employer
		}
		# Then we start the travel.
		task_contract_taker = {
			save_scope_as = task_contract_taker
		}
		scope:task_contract_taker = {
			trigger_event = ck_contract_events.0001
		}
		# Track the livestock
		increment_variable_effect = {
			VAR = ck_livestock_tally
			VAL = 0
		}
		debug_log = "LIVESTOCK_TALLY_SET_LOG"
	}
	
	on_invalidated = {
		if = {
			limit = {
				root.task_contract_taker = {
					any_character_active_contract = {
						this = root
					}
				}
			}
			task_contract_employer = {
				save_scope_as = task_contract_employer
			}
			save_scope_value_as = {
				name = task_contract_tier
				value = root.task_contract_tier
			}
			var:task_contract_destination = {
				save_scope_as = task_contract_destination
			}
			var:task_contract_target = {
				save_scope_as = task_contract_target
			}
			task_contract_taker = {
				set_variable = {
					name = invalidated_task_contract_type
					value = flag:laamp_ck_deliver_herd
				}
			}
			if = {
				limit = {
					task_contract_employer = {
						is_alive = no
					}
				}
				task_contract_employer.primary_title.holder ?= {
					save_scope_as = employer_heir
				}
				task_contract_taker = {
					save_scope_value_as = {
						name = task_contract_invalidation_reason
						value = flag:employer_dead
					}
					trigger_event = ep3_contract_event.0012
				
				}
			}
			else_if = {
				limit = {
					scope:task_contract_target = {
						is_alive = no
					}
				}
				task_contract_taker = {
					save_scope_value_as = {
						name = task_contract_invalidation_reason
						value = flag:target_dead
					}
					trigger_event = ep3_contract_event.0012
				}
			}
		}
		# Remove the peasants so we get new ones next time around
		hidden_effect = {
			scope:destination_1_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_2_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_3_character ?= {
				silent_disappearance_effect = yes
			}
			scope:destination_4_character ?= {
				silent_disappearance_effect = yes
			}
		}
	}
	# Rewards
	task_contract_reward = {
		success_critical = { # More livestock than desired
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				save_scope_value_as = { name = extra_reward value = flag:no }
				# Critical Rewards.
				laamp_rewards_disburse_ordinary_currency_effect = {
					GOLD = scope:task_contract.var:gold_success_critical
					PRESTIGE = task_contract_success_prestige_gain_full_value
					PIETY = 0
					PROVISIONS = task_contract_success_provisions_gain_medium_value
					OPINION = task_contract_opinion_excelled_reward_value
					OPINION_TYPE = critically_succeeded_task_contract_opinion
					CONTACT = yes
					CONTACT_HOOK = yes
					EXTRA_REWARD = scope:extra_reward
				}
			}
		}
		success_standard = { # Desired amount of livestock
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				save_scope_value_as = { name = extra_reward value = flag:no }
				# Standard Rewards.
				laamp_rewards_disburse_ordinary_currency_effect = {
					GOLD = scope:task_contract.var:gold_success_standard
					PRESTIGE = task_contract_success_prestige_gain_half_value
					PIETY = 0
					PROVISIONS = task_contract_success_provisions_gain_minor_value
					OPINION = task_contract_opinion_standard_reward_value
					OPINION_TYPE = succeeded_task_contract_opinion
					CONTACT = yes
					CONTACT_HOOK = no
					EXTRA_REWARD = scope:extra_reward
				}
			}
		}
		success_substandard = { # Less livestock than desired
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				save_scope_value_as = { name = extra_reward value = flag:no }
				# Substandard Rewards.
				laamp_rewards_disburse_ordinary_currency_effect = {
					GOLD = scope:task_contract.var:gold_success_substandard
					PRESTIGE = task_contract_success_prestige_gain_quarter_value
					PIETY = 0
					PROVISIONS = 0
					OPINION = task_contract_opinion_reduced_reward_value
					OPINION_TYPE = succeeded_task_contract_opinion
					CONTACT = yes
					CONTACT_HOOK = no
					EXTRA_REWARD = scope:extra_reward
				}
			}

		}
		failure_standard = { # No livestock remaining by end of contract
			positive = no
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract" }
				save_scope_value_as = { name = extra_reward value = flag:no }
				# Standard Penalties.
				laamp_rewards_disburse_ordinary_currency_effect = {
					GOLD = 0
					PRESTIGE = task_contract_failure_prestige_loss_full_value
					PIETY = 0
					PROVISIONS = 0
					OPINION = task_contract_opinion_standard_failure_value
					OPINION_TYPE = failed_task_contract_opinion
					CONTACT = no
					CONTACT_HOOK = no
					EXTRA_REWARD = scope:extra_reward
				}
			}

		}
		failure_stole_herd = { # Stole the herd for provisions
			visible = no
			positive = no
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract" }
				save_scope_value_as = { name = extra_reward value = flag:no }
				save_scope_value_as = {
					name = stolen_livestock_provisions
					value = {
						value = 100
						multiply = scope:task_contract_taker.var:ck_livestock_tally
					}
				}
				# Real Big Penalties.
				laamp_rewards_disburse_ordinary_currency_effect = {
					GOLD = 0
					PRESTIGE = task_contract_failure_prestige_loss_full_value
					PIETY = 0
					PROVISIONS = scope:stolen_livestock_provisions
					OPINION = task_contract_failure_prestige_loss_double_value
					OPINION_TYPE = failed_task_contract_opinion
					CONTACT = no
					CONTACT_HOOK = no
					EXTRA_REWARD = scope:extra_reward
				}
				task_contract_employer = {
					add_legitimacy = medium_legitimacy_loss
					add_prestige = task_contract_opinion_standard_failure_value
					add_stress = medium_stress_gain
				}
				# Update gallowsbait score.
				task_contract_taker = {
					laamp_rewards_apply_criminal_xp_effect = {
						TRACK = thief
						XP_MAX = gallowsbait_xp_major_gain
						XP_MIN = gallowsbait_xp_medium_gain
					}
				}
			}

		}
	}

}
