# Order of Conquest
ho_calradian_order_rose = {
	type = military
	government = holy_order_government
	title = d_order_of_the_rose
	name = "holy_order_order_of_the_rose_tamali"
	coat_of_arms = "holy_order_order_of_the_rose_tamali_coa"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.county ?= title:c_saverip
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
			}
		}
	}

	weight = {
		value = 10
		#Should be founded in saverip, the Great Citadel of Geroia
		if = {
			limit = {
				scope:province.county ?= title:c_saverip
			}
			add = 1000
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = region_geroia
				}
			}
			add = 100
		}
		#More likely if first holy order
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							leader ?= {
								government_has_flag = government_is_military_holy_order
							}
						}
					}
				}
			}
			add = 100
		}
		
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				
				scope:province.county ?= title:c_saverip
			}
			add = 1000
		}
	}

	men_at_arms = { geroian_crossbowmen_2 }
}

#knight brothers
ho_generic_military_culture_knight_brothers = {

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = military_order_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			scope:founder = {
				religion = religion:calradian_religion
				faith = {
					has_doctrine_parameter = is_western_christian_faith
					NOT = {
						any_faith_holy_order = {
							title = {
								has_variable = military_culture_brothers
								var:military_culture_brothers = scope:founder.culture
							}
						}
					}
				}
				culture = {
					has_cultural_parameter = martial_custom_male_only_combatant
				}
			}
		}
	}

	weight = {
		value = 1
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							title = {
								has_variable = military_culture_brothers
							}
						}
					}
				}
			}
			multiply = 3
		}
	}

	effect = {
		set_variable = {
			name = military_culture_brothers
			value = scope:founder.culture
		}
	}

	name = holy_order_culture_knight_brothers
	government = holy_order_government
	type = military
}

ho_calradic_ulvardites = {
	type = monastic
	government = monastic_holy_order_government
	title = d_ho_ulvardites
	name = "holy_order_ulvardites"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 905
			}
			scope:province.county ?= title:c_galend ##  CK_120 CK_TODO - Change to the mountain church between Swadia and Rhodok
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
				location ?= {
					geographical_region = region_vlandia
				}
			}
		}
	}

	weight = {
		value = 1
		#Should be founded in Italy
		if = {
			limit = {
				scope:province.county ?= title:c_galend
			}
			add = 5
		}
		#Trying to put ulvardites here on game start
		if = {
			limit = {
				years_from_game_start < 1
				current_year >= 867
				
				scope:province.county ?= title:c_galend
			}
			add = 10000
		}
		if = {
			limit = {
				scope:province = {
					OR = {
						kingdom = title:k_rhodok
						kingdom = title:k_vlandia
					}
				}
			}
			add = 5
		}
	}

	effect = {
		if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 1178
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_simony }
		}
		else_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 1066
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_dulia }
		}
		else_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 867
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_ora_et_labora }
		}
	}
}
